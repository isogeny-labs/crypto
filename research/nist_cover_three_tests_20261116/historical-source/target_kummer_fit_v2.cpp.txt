std::vector<GoodPoint> ek_input_points(TargetWeil& target,const Json& descent,const Json& receipt){
    auto matrix=good_matrix(descent.at("change_of_coordinates"));T24 slope=target.infinity2[1];std::vector<GoodPoint> out;int index=0;
    for(auto& row:receipt.at("points").arr){Line base{good_fp(row.at("x")),good_fp(row.at("y")),T24(1)},old{};
        for(int i=0;i<3;++i)for(int j=0;j<3;++j)old[i]=old[i]+matrix[i][j]*base[j];
        require(target.f.eval(old)==T24(0),"Kummer input point curve equation");
        if(old[2]!=T24(0)){T24 x=old[0]/old[2],y=old[1]/old[2],l=y-slope*x;out.push_back({Line{x,y,T24(1)},Values8{T24(1),x,x*x,y,y*y,x*y,x*l*l,y*l*l},index});}++index;
    }return out;
}
TVector ek_good_values(const std::vector<GoodPoint>& points,const Triple& triple,const std::vector<Triple>& monomials,const TSpace& functions){
    const auto& a=points.at(size_t(triple[0]));const auto& b=points.at(size_t(triple[1]));const auto& c=points.at(size_t(triple[2]));
    Matrix3<T24> m{a.point,b.point,c.point};T24 omega=determinant3(m);require(omega!=T24(0),"noncollinear Kummer fit support");auto values=symmetric_values(monomials,a.basis,b.basis,c.basis);TVector out;for(auto& f:functions)out.push_back(good_dot(f,values)/(omega*omega));return out;
}
TSpace ek_solve(TSpace left,const TSpace& right){
    require(left.size()==8&&right.size()==8,"eight fit equations");for(int i=0;i<8;++i){require(left[i].size()==8&&right[i].size()==8,"eight fit columns");left[i].insert(left[i].end(),right[i].begin(),right[i].end());}
    for(int col=0;col<8;++col){int pick=col;while(pick<8&&left[pick][col]==T24(0))++pick;require(pick<8,"Kummer interpolation rank8");std::swap(left[pick],left[col]);T24 inv=left[col][col].inv();for(int j=col;j<16;++j)left[col][j]=left[col][j]*inv;
        for(int row=0;row<8;++row)if(row!=col){T24 c=left[row][col];if(c!=T24(0))for(int j=col;j<16;++j)left[row][j]=left[row][j]-c*left[col][j];}
    }
    TSpace map(8,TVector(8,T24(0)));for(int j=0;j<8;++j)for(int i=0;i<8;++i)map[j][i]=left[i][8+j];return map;
}
bool ek_fit_equal(const TSpace& matrix,const Values8& k,const TVector& h){TVector values(k.begin(),k.end());for(int i=0;i<8;++i)if(good_dot(matrix[i],values)!=h[i])return false;return true;}
