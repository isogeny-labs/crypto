struct Echelon {
    int width;
    std::map<int,TVector> rows;
    explicit Echelon(int n):width(n){}
    bool insert(TVector row){
        require(int(row.size())==width,"good-function matrix width");
        for(auto& [pivot,r]:rows){T24 c=row[pivot];if(c!=T24(0))for(int j=pivot;j<width;++j)if(r[j]!=T24(0))row[j]=row[j]-c*r[j];}
        int pivot=0;while(pivot<width&&row[pivot]==T24(0))++pivot;if(pivot==width)return false;
        T24 inv=row[pivot].inv();for(int j=pivot;j<width;++j)row[j]=row[j]*inv;rows.emplace(pivot,row);return true;
    }
    TSpace kernel()const{
        TSpace out;
        for(int free=0;free<width;++free)if(!rows.count(free)){TVector v(size_t(width),T24(0));v[free]=T24(1);
            for(auto it=rows.rbegin();it!=rows.rend();++it){T24 sum;for(int j=it->first+1;j<width;++j)if(v[j]!=T24(0)&&it->second[j]!=T24(0))sum=sum+it->second[j]*v[j];v[it->first]=-sum;}
            out.push_back(v);
        }return out;
    }
};
T24 good_dot(const TVector& a,const TVector& b){require(a.size()==b.size(),"good-function evaluation shape");T24 value;for(size_t i=0;i<a.size();++i)if(a[i]!=T24(0)&&b[i]!=T24(0))value=value+a[i]*b[i];return value;}

TVector symmetric_values(const std::vector<Triple>& monomials,const Values8& a,const Values8& b,const Values8& c){
    TVector values;values.reserve(monomials.size());

    return values;
}
