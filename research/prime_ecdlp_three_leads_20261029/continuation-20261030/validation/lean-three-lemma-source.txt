import ThreeLeads
import Mathlib.GroupTheory.SpecificGroups.Cyclic

/-! Logical contracts of the native scalar certificates. No solving-cost or
polynomial-system theorem is asserted. The cyclic group/cardinality hypotheses
and original row equations are explicit; Rust parsing/arithmetic are not proved.
-/
namespace ThreeLeads.PKMScalarCertificates
open scoped BigOperators

/-- Construct the scalar coordinate using Mathlib's cyclic-group isomorphism,
rather than assuming an injective field-to-point map. The finite point group
must be generated by P and have the stated nonzero cardinality. -/
theorem cyclic_point_relation {G I : Type*} [AddCommGroup G] [Fintype I]
    {n : ℕ} [NeZero n] (P Q : G)
    (generates : ∀ x, x ∈ AddSubgroup.zmultiples P) (cardinality : Nat.card G = n)
    (B : I → G) (c : I → ZMod n) (a b : ZMod n)
    (relation : (∑ i, (c i).val • B i) = a.val • P + b.val • Q) :
    let log := (zmodAddEquivOfGenerator generates cardinality).symm
    (∑ i, c i * log (B i)) - b * log Q = a := by
  let log := (zmodAddEquivOfGenerator generates cardinality).symm
  have h := congrArg log relation
  have hp : log P = 1 := zmodAddEquivOfGenerator_symm_apply_generator generates cardinality
  have equation : (∑ i, c i * log (B i)) = a + b * log Q := by
    simpa only [map_sum, map_add, map_nsmul, nsmul_eq_mul, ZMod.natCast_zmod_val,
      hp, mul_one] using h
  change (∑ i, c i * log (B i)) - b * log Q = a
  rw [equation]
  ring

/-- A checked combination of retained original rows isolates one coordinate.
This is the exact contract of the replay certificate's recovery weights. -/
theorem recovery_combination {K I J : Type*} [Field K] [Fintype I] [Fintype J] [DecidableEq I]
    (M : J → I → K) (z : I → K) (rhs weights : J → K) (target : I) (value : K)
    (rows : ∀ j, ThreeLeads.rowEval (M j) z = rhs j)
    (certificate : ∀ i, (∑ j, weights j * M j i) = if i = target then 1 else 0)
    (rightHandSide : (∑ j, weights j * rhs j) = value) : z target = value := by
  classical
  have h := ThreeLeads.row_combination M z rhs weights rows
  have evaluation : ThreeLeads.rowEval (fun i => ∑ j, weights j * M j i) z = z target := by
    unfold ThreeLeads.rowEval
    simp_rw [certificate]
    simp
  rw [evaluation, rightHandSide] at h
  exact h

/-- A target-normalized kernel witness certifies an unresolved coordinate for
any consistent retained system. No probabilistic full-rank assumption is used. -/
theorem target_ambiguity {K I J : Type*} [Field K] [Fintype I]
    (M : J → I → K) (z h : I → K) (rhs : J → K) (target : I)
    (rows : ∀ j, ThreeLeads.rowEval (M j) z = rhs j)
    (kernel : ∀ j, ThreeLeads.rowEval (M j) h = 0) (changes : h target = 1) :
    (∀ j, ThreeLeads.rowEval (M j) (z + h) = rhs j) ∧ (z + h) target ≠ z target := by
  have nonzero : h target ≠ 0 := by rw [changes]; exact one_ne_zero
  obtain ⟨preserves, different⟩ := ThreeLeads.kernel_ambiguity M z h kernel target nonzero
  exact ⟨fun j => (preserves j).trans (rows j), different⟩

end ThreeLeads.PKMScalarCertificates
