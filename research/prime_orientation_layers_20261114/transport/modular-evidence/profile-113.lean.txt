module
public import CryptoLean.StrictTriangular
public import OrientationTransport
import all OrientationAnchor
import all OrientationLeading
public section
noncomputable section
namespace OrientationConstructedProfile
open Matrix Finset OrientationAnchor OrientationLeading

def penaltyMatrix (u : ℕ) : Matrix (Fin u) (Fin u) ℕ := fun i j => j.val-i.val

theorem penalty_nilpotent (u : ℕ) : (penaltyMatrix u)^u=0 := by
  have H := CryptoLean.StrictTriangular.nat_pow_card_zero (penaltyMatrix u)
    (fun i j hij => Nat.sub_eq_zero_of_le (by exact hij))
  simpa only [Fintype.card_fin] using H

def resolvent (u : ℕ) : Matrix (Fin u) (Fin u) ℕ := ∑ r ∈ range u,(penaltyMatrix u)^r

theorem resolvent_fixed (u : ℕ) : resolvent u=penaltyMatrix u*resolvent u+1 :=
  CryptoLean.StrictTriangular.finite_geometric_fixed_point _ _ (penalty_nilpotent u)

def vector (u : ℕ) : Fin u → ℕ := resolvent u *ᵥ (fun _ => 1)

theorem vector_fixed (u : ℕ) : vector u=penaltyMatrix u *ᵥ vector u+(fun _ => 1) := by
  unfold vector
  conv_lhs => rw [resolvent_fixed]
  rw [Matrix.add_mulVec,← Matrix.mulVec_mulVec,Matrix.one_mulVec]

theorem vector_row (u : ℕ) (i : Fin u) :
    vector u i=(∑ j : Fin u,(j.val-i.val)*vector u j)+1 := by
  have H := congrFun (vector_fixed u) i
  simpa only [Pi.add_apply,Matrix.mulVec,dotProduct,penaltyMatrix] using H

def profile (u j : ℕ) : ℕ := if h : j < u then vector u ⟨j,h⟩ else 0

theorem profile_at (u i : ℕ) (hi : i < u) : profile u i=vector u ⟨i,hi⟩ := by
  simp only [profile,dif_pos hi]

theorem profile_row (u i : ℕ) (hi : i < u) :
    profile u i=(∑ j ∈ range u,(j-i)*profile u j)+1 := by
  have hs : (∑ j : Fin u,(j.val-i)*vector u j)=(∑ j ∈ range u,(j-i)*profile u j) := by
    rw [← Fin.sum_univ_eq_sum_range]
    apply Finset.sum_congr rfl
    intro j _
    rw [profile_at u j.val j.isLt]
  rw [profile_at u i hi]
  exact (vector_row u ⟨i,hi⟩).trans (congrArg (fun x : ℕ => x+1) hs)

theorem profile_positive (u i : ℕ) (hi : i < u) : 0 < profile u i := by
  rw [profile_row u i hi]
  omega

theorem profile_schedule (u i : ℕ) (hi : i < u) : schedule u i (profile u)=1 := by
  have H := congrArg (fun x : ℕ => (x:ℤ)) (profile_row u i hi)
  push_cast at H
  unfold schedule
  omega

theorem four_profile_schedule (u i : ℕ) (hi : i < u) :
    schedule u i (fun j => 4*profile u j)=4 := by
  have H := profile_schedule u i hi
  unfold schedule at H ⊢
  simp only [Nat.cast_mul,Nat.cast_ofNat]
  have he : (∑ j ∈ range u,((j-i:ℕ):ℤ)*(4*(profile u j:ℤ)))=
      4*(∑ j ∈ range u,((j-i:ℕ):ℤ)*(profile u j:ℤ)) := by
    rw [Finset.mul_sum]
    apply Finset.sum_congr rfl
    intro j _
    ring
  rw [he]
  linarith

theorem actual_weight_nonnegative (a u i : ℕ) (hi : i < u) (hu : u ≤ a) :
    0 ≤ jetWeight a u i (fun j => 4*profile u j) := by
  rw [jetWeight,four_profile_schedule u i hi]
  omega

theorem four_profile_mass (u : ℕ) :
    (∑ j ∈ range u,4*profile u j)=4*(∑ j ∈ range u,profile u j) :=
  (Finset.mul_sum _ _ _).symm

end OrientationConstructedProfile
