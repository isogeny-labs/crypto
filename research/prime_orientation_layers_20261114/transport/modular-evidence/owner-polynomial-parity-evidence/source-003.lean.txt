module
public import Mathlib.Algebra.Polynomial.Expand
public import Mathlib.RingTheory.Ideal.Quotient.Basic
public import Mathlib.Tactic

public section
namespace CryptoLean.PolynomialParity
open Polynomial

section CommSemiring
variable {R : Type*} [CommSemiring R]

theorem fourth_linear_expand_two (s : R) (h4 : (4:R)=0) :
    (1+C s*X : R[X])^4 =
      expand R 2 (1+C (6*s^2)*X+C (s^4)*X^2) := by
  have h4P : (4:R[X])=0 := by
    have H := congrArg (fun r : R => (C r : R[X])) h4
    simpa only [map_ofNat,map_zero] using H
  simp only [map_add,map_one,map_mul,expand_C,expand_X,map_pow,map_ofNat]
  calc
    _ = (1+(6:R[X])*(C s)^2*X^2+(C s)^4*X^4)+
        (4:R[X])*(C s*X+(C s)^3*X^3) := by ring
    _ = _ := by
      rw [h4P,zero_mul,add_zero]
      ring

end CommSemiring

section CommRing
variable {R : Type*} [CommRing R]

theorem dvd_odd_coeff_of_quotient_expand (d : R) (P : R[X])
    (Q : (R ⧸ Ideal.span ({d}:Set R))[X])
    (h : P.map (Ideal.Quotient.mk (Ideal.span ({d}:Set R)))=
      expand (R ⧸ Ideal.span ({d}:Set R)) 2 Q) (b : ℕ) :
    d ∣ P.coeff (2*b+1) := by
  apply (Ideal.Quotient.eq_zero_iff_dvd d (P.coeff (2*b+1))).mp
  have H := congrArg (fun A : (R ⧸ Ideal.span ({d}:Set R))[X] => A.coeff (2*b+1)) h
  rw [coeff_map,coeff_expand (by decide : 0 < 2)] at H
  have hodd : ¬ 2 ∣ 2*b+1 := by
    rintro ⟨k,hk⟩
    omega
  simpa only [ite_eq_right hodd] using H

theorem quartic_geometric_odd_coeff_dvd {ι : Type*} (d A B N : R)
    (S : Finset ι) (s : ι → R) (n : ι → ℕ) (k b : ℕ)
    (hd4 : d ∣ (4:R)) (hdB : d ∣ B) :
    d ∣ (C N * ∑ r ∈ Finset.range k,
      (1-((C A+C B*X)*∏ j ∈ S,(1+C (s j)*X)^(4*n j)))^r : R[X]).coeff (2*b+1) := by
  classical
  let I : Ideal R := Ideal.span ({d}:Set R)
  let K := R ⧸ I
  let f : R →+* K := Ideal.Quotient.mk I
  let even4 (x : K) : K[X] := 1+C (6*x^2)*X+C (x^4)*X^2
  let D : R[X] := (C A+C B*X)*∏ j ∈ S,(1+C (s j)*X)^(4*n j)
  let E : K[X] := C (f A)*∏ j ∈ S,(even4 (f (s j)))^(n j)
  have hf4 : (4:K)=0 := by
    have H := (Ideal.Quotient.eq_zero_iff_dvd d (4:R)).mpr hd4
    simpa only [map_ofNat] using H
  have hfB : f B=0 := (Ideal.Quotient.eq_zero_iff_dvd d B).mpr hdB
  have hfour : ∀ x : K,(1+C x*X)^4=expand K 2 (even4 x) :=
    fun x => fourth_linear_expand_two x hf4
  have hD : D.map f=expand K 2 E := by
    simp only [D,E,Polynomial.map_mul,Polynomial.map_add,Polynomial.map_C,
      Polynomial.map_X,Polynomial.map_prod,Polynomial.map_pow,Polynomial.map_one,
      hfB,map_zero,zero_mul,add_zero,pow_mul,hfour,
      map_mul,map_prod,map_pow,expand_C]
  change d ∣ (C N * ∑ r ∈ Finset.range k,(1-D)^r).coeff (2*b+1)
  refine dvd_odd_coeff_of_quotient_expand d _
    (C (f N)*∑ r ∈ Finset.range k,(1-E)^r) ?_ b
  change (C N * ∑ r ∈ Finset.range k,(1-D)^r).map f=
    expand K 2 (C (f N)*∑ r ∈ Finset.range k,(1-E)^r)
  simp only [Polynomial.map_mul,Polynomial.map_C,Polynomial.map_sum,
    Polynomial.map_pow,Polynomial.map_sub,Polynomial.map_one,hD,
    map_mul,map_sum,map_pow,map_sub,map_one,expand_C]

end CommRing
end CryptoLean.PolynomialParity
