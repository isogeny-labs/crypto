def Bits (b : ℕ) (q : ℚ) : Prop := max q.num.natAbs q.den ≤ 2^b

theorem bits_iff_height (b : ℕ) (q : ℚ) : Bits b q ↔ mulHeight₁ q ≤ (2:ℝ)^b := by
  rw [Rat.mulHeight₁_eq_max]
  unfold Bits
  norm_cast

theorem bits_zero : Bits 0 (0:ℚ) := by norm_num [Bits]
theorem bits_one : Bits 0 (1:ℚ) := by norm_num [Bits]
theorem bits_two : Bits 1 (2:ℚ) := by norm_num [Bits]

theorem Bits.mono {b c : ℕ} {q : ℚ} (h : Bits b q) (hbc : b ≤ c) : Bits c q :=
  h.trans (Nat.pow_le_pow_right (by decide) hbc)

theorem Bits.neg {b : ℕ} {q : ℚ} (h : Bits b q) : Bits b (-q) := by
  rw [bits_iff_height] at h ⊢
  simpa only [mulHeight₁_neg] using h

theorem Bits.inv {b : ℕ} {q : ℚ} (h : Bits b q) : Bits b q⁻¹ := by
  rw [bits_iff_height] at h ⊢
  simpa only [mulHeight₁_inv] using h

theorem Bits.mul {b c : ℕ} {q r : ℚ} (hq : Bits b q) (hr : Bits c r) :
    Bits (b+c) (q*r) := by
  rw [bits_iff_height] at hq hr ⊢
  calc
    mulHeight₁ (q*r) ≤ mulHeight₁ q*mulHeight₁ r := mulHeight₁_mul_le q r
    _ ≤ (2:ℝ)^b*2^c := mul_le_mul hq hr (by positivity) (by positivity)
    _ = (2:ℝ)^(b+c) := (pow_add _ _ _).symm

theorem Bits.div {b c : ℕ} {q r : ℚ} (hq : Bits b q) (hr : Bits c r) :
    Bits (b+c) (q/r) := by simpa only [div_eq_mul_inv] using hq.mul hr.inv

theorem Bits.add {b c : ℕ} {q r : ℚ} (hq : Bits b q) (hr : Bits c r) :
    Bits (b+c+1) (q+r) := by
  rw [bits_iff_height] at hq hr ⊢
  have H : mulHeight₁ (q+r) ≤ 2*mulHeight₁ q*mulHeight₁ r := by
    simpa only [NumberField.totalWeight_eq_finrank,Module.finrank_self,pow_one] using
      mulHeight₁_add_le q r
  calc
    mulHeight₁ (q+r) ≤ 2*(mulHeight₁ q*mulHeight₁ r) := by simpa only [mul_assoc] using H
    _ ≤ 2*((2:ℝ)^b*2^c) := mul_le_mul_of_nonneg_left
      (mul_le_mul hq hr (by positivity) (by positivity)) (by norm_num)
    _ = (2:ℝ)^(b+c+1) := by rw [pow_add,pow_add,pow_one]; ring

theorem Bits.sub {b c : ℕ} {q r : ℚ} (hq : Bits b q) (hr : Bits c r) :
    Bits (b+c+1) (q-r) := by simpa only [sub_eq_add_neg] using hq.add hr.neg

theorem Bits.pow {b : ℕ} {q : ℚ} (h : Bits b q) (n : ℕ) : Bits (b*n) (q^n) := by
  rw [bits_iff_height] at h ⊢
  rw [mulHeight₁_pow,pow_mul]
  exact pow_le_pow_left₀ (by positivity) h n

theorem bits_int (b : ℕ) (z : ℤ) (hz : z.natAbs ≤ 2^b) : Bits b (z:ℚ) := by
  unfold Bits
  simp only [Rat.num_intCast,Rat.den_intCast]
  exact max_le hz Nat.one_le_two_pow

theorem bits_two_pow (n : ℕ) : Bits n ((2:ℚ)^n) := by
  simpa only [one_mul] using bits_two.pow n

theorem bits_prod {I : Type*} (S : Finset I) (q : I → ℚ) (b : I → ℕ)
    (h : ∀ i ∈ S,Bits (b i) (q i)) : Bits (∑ i ∈ S,b i) (∏ i ∈ S,q i) := by
  classical
  induction S using Finset.induction_on with
  | empty => simpa using bits_one
  | @insert i S hi ih =>
    rw [sum_insert hi,prod_insert hi]
    exact (h i (mem_insert_self _ _)).mul (ih (by intro j hj; exact h j (mem_insert_of_mem hj)))

theorem bits_sum {I : Type*} (S : Finset I) (q : I → ℚ) (b : I → ℕ)
    (h : ∀ i ∈ S,Bits (b i) (q i)) : Bits (S.card+∑ i ∈ S,b i) (∑ i ∈ S,q i) := by
  classical
  induction S using Finset.induction_on with
  | empty => simpa using bits_zero
  | @insert i S hi ih =>
    have H := (h i (mem_insert_self _ _)).add (ih (by intro j hj; exact h j (mem_insert_of_mem hj)))
    have he : (insert i S).card+(∑ j ∈ insert i S,b j)=b i+(S.card+∑ j ∈ S,b j)+1 := by
      rw [card_insert_of_notMem hi,sum_insert hi]
      omega
    rw [he,sum_insert hi]
    exact H

theorem grid_natAbs_bound (a : ℕ) (t : ℤ) (ht : (t:ℚ) ∈ grid a) : t.natAbs ≤ 2^a := by
  rcases mem_image.mp ht with ⟨z,hz,he⟩
  have hzt : z=t := by exact_mod_cast he
  subst z
  have h := mem_Ico.mp hz
  have H : |t| ≤ (2:ℤ)^a := abs_le.mpr ⟨h.1,le_of_lt h.2⟩
  have HC : (t.natAbs:ℤ) ≤ (2:ℤ)^a := by
    simpa only [Int.natCast_natAbs] using H
  exact_mod_cast HC

theorem actual_grid_bits (a : ℕ) (t : ℤ) (ht : (t:ℚ) ∈ grid a) : Bits a (t:ℚ) :=
  bits_int a t (grid_natAbs_bound a t ht)

theorem actual_delta_bits (a r y : ℕ) (hr : r < 2^a) (hy : y < 2^a) : Bits a (delta r y) := by
  have H : ((y:ℤ)-r).natAbs ≤ 2^a := by
    have hrZ : (r:ℤ) < (2:ℤ)^a := by exact_mod_cast hr
    have hyZ : (y:ℤ) < (2:ℤ)^a := by exact_mod_cast hy
    have h : |(y:ℤ)-r| ≤ (2:ℤ)^a := abs_le.mpr ⟨by omega,by omega⟩
    have HC : (((y:ℤ)-r).natAbs:ℤ) ≤ (2:ℤ)^a := by
      simpa only [Int.natCast_natAbs] using h
    exact_mod_cast HC
  simpa only [Int.cast_sub,Int.cast_natCast,delta] using bits_int a ((y:ℤ)-r) H

theorem actual_slope_bits (a u b x y : ℕ) (hu : u ≤ a) (hy : y < 2^a)
    (q : ℚ) (hq : Bits b q) : Bits (u+(b+a+1)) (slope u (rowResidue u x) y q) := by
  exact bits_two_pow u |>.div (hq.sub (actual_delta_bits a _ y (row_source_bound a u x hu) hy))

theorem actual_chart_slope_bits (a u x y : ℕ) (t : ℤ) (hu : u ≤ a)
    (hy : y < 2^a) (ht : (t:ℚ) ∈ grid a) : Bits ((a+1)+(a+a+1)) (chartSlope a u x y t) := by
  exact (bits_two_pow (a+1)).div
    ((actual_grid_bits a t ht).sub (actual_delta_bits a _ y (row_source_bound a u x hu) hy))

theorem actual_anchor_slope_bits (a u j x y : ℕ) (hu : u ≤ a) (hy : y < 2^a) :
    Bits (u+(j+a+1)) (slope u (rowResidue u x) y ((2:ℚ)^j)) :=
  actual_slope_bits a u j x y hu hy _ (bits_two_pow j)

end OrientationRationalSize

theorem bits_nat (b n : ℕ) (hn : n ≤ 2^b) : Bits b (n:ℚ) := by
  simpa only [Int.cast_natCast,Int.natAbs_natCast] using bits_int b (n:ℤ) hn

theorem actual_grid_member_bits (a : ℕ) (q : ℚ) (hq : q ∈ grid a) : Bits a q := by
  rcases mem_image.mp hq with ⟨z,hz,rfl⟩
  exact actual_grid_bits a z (mem_image.mpr ⟨z,hz,rfl⟩)

def derivativeSize (a : ℕ) : ℕ := (2*2^a)*(a+a+1)
def scalarSize (a : ℕ) : ℕ := (a+1)+derivativeSize a+derivativeSize a
def correctionSize (u i : ℕ) (m : ℕ → ℕ) : ℕ := ∑ j ∈ range u,(j+i+1+j)*m j
def diagonalSize (a u i : ℕ) (m : ℕ → ℕ) : ℕ := scalarSize a+i+correctionSize u i m

theorem actual_derivative_bits (a : ℕ) (q : ℚ) (hq : q ∈ grid a) :
    Bits (derivativeSize a) ((nodal (grid a)).derivative.eval q) := by
  rw [nodal_derivative _ _ hq,puncture,eval_prod]
  have H := bits_prod ((grid a).erase q) (fun s => q-s) (fun _ => a+a+1) (by
    intro s hs
    exact (actual_grid_member_bits a q hq).sub
      (actual_grid_member_bits a s (mem_erase.mp hs).2))
  have hb : ((grid a).erase q).card ≤ 2*2^a := by
    rw [← grid_card]
    exact card_erase_le
  have hs : (∑ _ ∈ (grid a).erase q,(a+a+1))=((grid a).erase q).card*(a+a+1) := by
    simp
  rw [hs] at H
  simpa only [derivativeSize,eval_sub,eval_X,eval_C] using H.mono (Nat.mul_le_mul_right _ hb)

theorem actual_normalizer_bits (a : ℕ) : Bits (derivativeSize a) (normalizer (grid a)) :=
  actual_derivative_bits a 0 (zero_in_grid a)

theorem actual_jet_scalar_bits (a i : ℕ) (hi : i ≤ a) : Bits (scalarSize a) (jetScalar a i) := by
  unfold jetScalar scalarSize
  exact ((bits_two_pow (a+1)).mul (actual_normalizer_bits a).inv).mul
    (actual_derivative_bits a (poleQ i) (pole_member a i hi))

theorem actual_diagonal_value_bits (u i : ℕ) (m : ℕ → ℕ) :
    Bits (correctionSize u i m) (diagonalValue u i m) := by
  rw [diagonal_value_product]
  unfold correctionSize
  apply bits_prod
  intro j _
  exact (((bits_two_pow j).add (bits_two_pow i)).div (bits_two_pow j)).pow (m j)

theorem actual_diagonal_bits (a u i : ℕ) (m : ℕ → ℕ) (hi : i ≤ a) :
    Bits (diagonalSize a u i m) (diagonalJet a u i m) := by
  rw [diagonalJet_formula a u i m hi]
  exact ((actual_jet_scalar_bits a i hi).div (bits_two_pow i).neg).mul
    (actual_diagonal_value_bits u i m)

theorem actual_moment_left_bits (a u k : ℕ) (x : Ports a) (rb : MomentLabels u k) :
    Bits (rb.2.val+a*rb.2.val) (momentLeft a u k x rb) := by
  unfold momentLeft
  split_ifs
  · have hb : rowQuotient u x.val ≤ 2^a :=
      (Nat.div_le_self _ _).trans (Nat.le_of_lt x.isLt)
    exact (bits_two_pow rb.2.val).mul ((bits_nat a _ hb).pow rb.2.val)
  · exact bits_zero.mono (by omega)

theorem actual_absolute_left_coefficient_bits (a u i p : ℕ) (m : ℕ → ℕ)
    (hi : i ≤ a) (x : Ports a) (rb : LabelsAt a u i p m) (n : ℕ) :
    Bits (diagonalSize a u i m+(rb.2.val+a*rb.2.val))
      ((absoluteLeft a u i p m x rb).coeff n) := by
  have H := (actual_diagonal_bits a u i m hi).mul
    (actual_moment_left_bits a u _ x rb)
  unfold absoluteLeft OrientationBivariate.weightedLeft OrientationBivariate.polyLeft
  rw [← map_mul]
  by_cases hn : n=0
  · subst n
    simpa only [coeff_C_zero] using H
  · simp only [coeff_C,hn,ite_false]
    exact bits_zero.mono (by omega)

end OrientationRationalSize

def geometric {R : Type*} [CommRing R] (x : R) (k : ℕ) : R := ∑ j ∈ range k,(1-x)^j

def clearedGeometric {R : Type*} [CommRing R] (d A : R) : ℕ → R
  | 0 => 0
  | k+1 => d*clearedGeometric d A k+d*(d-A)^k

theorem geometric_succ {R : Type*} [CommRing R] (x : R) (k : ℕ) :
    geometric x (k+1)=geometric x k+(1-x)^k := sum_range_succ _ _

theorem cleared_identity {R : Type*} [CommRing R] (d e A : R) (hde : d*e=1) (k : ℕ) :
    clearedGeometric d A k=d^k*geometric (e*A) k := by
  induction k with
  | zero => simp [clearedGeometric,geometric]
  | succ k ih =>
    rw [clearedGeometric,ih,geometric_succ]
    have he : d*(1-e*A)=d-A := by rw [mul_sub,mul_one,← mul_assoc,hde,one_mul]
    have hp : (d-A)^k=d^k*(1-e*A)^k := by rw [← he,mul_pow]
    rw [hp,pow_succ]
    ring

theorem cleared_map {R S : Type*} [CommRing R] [CommRing S]
    (f : R →+* S) (d A : R) (k : ℕ) :
    f (clearedGeometric d A k)=clearedGeometric (f d) (f A) k := by
  induction k with
  | zero => simp [clearedGeometric]
  | succ k ih => simp only [clearedGeometric,map_add,map_mul,map_pow,map_sub,ih]

def integerMap : (ℤ[X])[X] →+* (ℚ[X])[X] :=
  Polynomial.mapRingHom (Polynomial.mapRingHom (Int.castRingHom ℚ))

def integerNumerator (d : ℤ) (A : (ℤ[X])[X]) (k : ℕ) : (ℤ[X])[X] :=
  clearedGeometric (C (C d)) A k

theorem integerMap_constant (d : ℤ) : integerMap (C (C d))=C (C (d:ℚ)) := by
  simp [integerMap]

theorem integer_bivariate_clearing (d : ℤ) (hd : d ≠ 0) (A : (ℤ[X])[X]) (k : ℕ) :
    C (C ((d:ℚ)^k))*geometricRows (C (C ((d:ℚ)⁻¹))*integerMap A) k=
      integerMap (integerNumerator d A k) := by
  have hq : (d:ℚ) ≠ 0 := by exact_mod_cast hd
  have hu : (C (C (d:ℚ)) : (ℚ[X])[X])*C (C ((d:ℚ)⁻¹))=1 := by
    rw [← map_mul,← map_mul,mul_inv_cancel₀ hq,map_one,map_one]
  have H := cleared_identity (C (C (d:ℚ))) (C (C ((d:ℚ)⁻¹))) (integerMap A) hu k
  rw [integerNumerator,cleared_map,integerMap_constant]
  simpa only [geometric,geometricRows,map_pow] using H.symm

end OrientationCommonDenominator

structure Rep (P : (ℚ[X])[X]) where
  den : ℕ
  positive : 0 < den
  num : (ℤ[X])[X]
  formula : integerMap num=C (C (den:ℚ))*P

def Rep.rational (q : ℚ) : Rep (C (C q)) where
  den := q.den
  positive := q.den_pos
  num := C (C q.num)
  formula := by
    rw [integerMap_constant,← map_mul,← map_mul]
    apply congrArg (fun r : ℚ => (C (C r) : (ℚ[X])[X]))
    simpa only [mul_comm] using (Rat.mul_den_eq_num q).symm

def Rep.one : Rep (1 : (ℚ[X])[X]) where
  den := 1
  positive := by decide
  num := 1
  formula := by simp

def Rep.outerX : Rep (X : (ℚ[X])[X]) where
  den := 1
  positive := by decide
  num := X
  formula := by simp [integerMap]

def Rep.innerX : Rep (C X : (ℚ[X])[X]) where
  den := 1
  positive := by decide
  num := C X
  formula := by simp [integerMap]

def Rep.mul {P Q : (ℚ[X])[X]} (hP : Rep P) (hQ : Rep Q) : Rep (P*Q) where
  den := hP.den*hQ.den
  positive := Nat.mul_pos hP.positive hQ.positive
  num := hP.num*hQ.num
  formula := by
    rw [map_mul,hP.formula,hQ.formula]
    simp only [Nat.cast_mul,map_mul]
    ring

def Rep.add {P Q : (ℚ[X])[X]} (hP : Rep P) (hQ : Rep Q) : Rep (P+Q) where
  den := hP.den*hQ.den
  positive := Nat.mul_pos hP.positive hQ.positive
  num := C (C (hQ.den:ℤ))*hP.num+C (C (hP.den:ℤ))*hQ.num
  formula := by
    simp only [map_add,map_mul,integerMap_constant,Int.cast_natCast,
      hP.formula,hQ.formula,Nat.cast_mul]
    ring

def Rep.pow {P : (ℚ[X])[X]} (hP : Rep P) (k : ℕ) : Rep (P^k) where
  den := hP.den^k
  positive := pow_pos hP.positive k
  num := hP.num^k
  formula := by
    rw [map_pow,hP.formula]
    simp only [Nat.cast_pow,map_pow,mul_pow]

def Rep.prod {I : Type*} (S : Finset I) (P : I → (ℚ[X])[X])
    (hP : ∀ i,Rep (P i)) : Rep (∏ i ∈ S,P i) where
  den := ∏ i ∈ S,(hP i).den
  positive := prod_pos (by intro i _; exact (hP i).positive)
  num := ∏ i ∈ S,(hP i).num
  formula := by
    rw [map_prod]
    simp_rw [(hP _).formula]
    rw [prod_mul_distrib]
    simp only [Nat.cast_prod,map_prod]

def Rep.geometric {P : (ℚ[X])[X]} (hP : Rep P) (k : ℕ) : Rep (geometricRows P k) where
  den := hP.den^k
  positive := pow_pos hP.positive k
  num := integerNumerator (hP.den:ℤ) hP.num k
  formula := by
    have hd : (hP.den:ℤ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hP.positive)
    have hq : (hP.den:ℚ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hP.positive)
    have hu : (C (C ((hP.den:ℚ)⁻¹)) : (ℚ[X])[X])*C (C (hP.den:ℚ))=1 := by
      rw [← map_mul,← map_mul,inv_mul_cancel₀ hq,map_one,map_one]
    have he : C (C ((hP.den:ℚ)⁻¹))*integerMap hP.num=P := by
      rw [hP.formula,← mul_assoc,hu,one_mul]
    have H := integer_bivariate_clearing (hP.den:ℤ) hd hP.num k
    simpa only [Int.cast_natCast,Nat.cast_pow,he] using H.symm

theorem geometric_denominator_exact {P : (ℚ[X])[X]} (hP : Rep P) (k : ℕ) :
    (hP.geometric k).den=hP.den^k := by
  unfold Rep.geometric
  rfl

end OrientationCommonDenominator

def Rep.cast {P Q : (ℚ[X])[X]} (r : Rep P) (h : P=Q) : Rep Q where
  den := r.den
  positive := r.positive
  num := r.num
  formula := by rw [← h]; exact r.formula

def actualDenomRep (a u x y : ℕ) (t : ℤ) (m : ℕ → ℕ) : Rep (denomRows a u x y t m) := by
  let h0 := (Rep.one.add ((Rep.rational (chartSlope a u x y t)).mul Rep.innerX)).add
    ((Rep.rational (slope u (rowResidue u x) y (t:ℚ))).mul Rep.outerX)
  let hA := Rep.prod (mask u x y)
    (fun j => (1+C (C (slope u (rowResidue u x) y ((2:ℚ)^j)))*X)^m j)
    (fun j => (Rep.one.add ((Rep.rational (slope u (rowResidue u x) y ((2:ℚ)^j))).mul Rep.outerX)).pow (m j))
  have he : ((1+C (C (chartSlope a u x y t))*C X)+
      C (C (slope u (rowResidue u x) y (t:ℚ)))*X)*
      (∏ j ∈ mask u x y,(1+C (C (slope u (rowResidue u x) y ((2:ℚ)^j)))*X)^m j)=
      denomRows a u x y t m := by
    simp only [denomRows,anchorRows,map_add,map_one,map_mul]
  exact (h0.mul hA).cast he

theorem actualDenomRep_den (a u x y : ℕ) (t : ℤ) (m : ℕ → ℕ) :
    (actualDenomRep a u x y t m).den=
      (chartSlope a u x y t).den*(slope u (rowResidue u x) y (t:ℚ)).den*
      ∏ j ∈ mask u x y,(slope u (rowResidue u x) y ((2:ℚ)^j)).den^m j := by
  dsimp [actualDenomRep,Rep.cast,Rep.mul,Rep.add,Rep.one,Rep.rational,Rep.innerX,
    Rep.outerX,Rep.prod,Rep.pow]
  simp only [Nat.one_mul,Nat.mul_one]

theorem bits_nat_iff (b n : ℕ) : Bits b (n:ℚ) ↔ n ≤ 2^b := by
  unfold Bits
  simp only [Rat.num_natCast,Rat.den_natCast,Int.natAbs_natCast]
  exact ⟨fun h => (le_max_left _ _).trans h,fun h => max_le h Nat.one_le_two_pow⟩

theorem den_bound {b : ℕ} {q : ℚ} (h : Bits b q) : q.den ≤ 2^b :=
  (le_max_right _ _).trans h

def denominatorSize (a u x y : ℕ) (m : ℕ → ℕ) : ℕ :=
  ((a+1)+(a+a+1))+(u+(a+a+1))+
    ∑ j ∈ mask u x y,(u+(j+a+1))*m j

theorem actualDenomRep_bound (a u x y : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (hu : u ≤ a) (hy : y < 2^a) (ht : (t:ℚ) ∈ grid a) :
    (actualDenomRep a u x y t m).den ≤ 2^(denominatorSize a u x y m) := by
  have hL := bits_nat _ _ (den_bound (actual_chart_slope_bits a u x y t hu hy ht))
  have hE := bits_nat _ _ (den_bound (actual_slope_bits a u a x y hu hy _ (actual_grid_bits a t ht)))
  have hP := bits_prod (mask u x y)
    (fun j => ((slope u (rowResidue u x) y ((2:ℚ)^j)).den:ℚ)^m j)
    (fun j => (u+(j+a+1))*m j) (by
      intro j _
      exact (bits_nat _ _ (den_bound (actual_anchor_slope_bits a u j x y hu hy))).pow (m j))
  have H := (hL.mul hE).mul hP
  apply (bits_nat_iff _ _).mp
  rw [actualDenomRep_den]
  simpa only [denominatorSize,Nat.cast_mul,Nat.cast_prod,Nat.cast_pow] using H

def extra (d : ℕ) : ℕ := padicValNat 2 d
def oddPart (d : ℕ) : ℕ := d.divMaxPow 2

theorem denominator_factor (d : ℕ) : 2^(extra d)*oddPart d=d :=
  Nat.pow_padicValNat_mul_divMaxPow 2 d

theorem oddPart_not_dvd (d : ℕ) (hd : 0 < d) : ¬ 2 ∣ oddPart d :=
  Nat.not_dvd_divMaxPow (by decide) (Nat.ne_of_gt hd)

theorem oddPart_unit (d p : ℕ) (hd : 0 < d) : IsUnit (oddPart d : ZMod (2^p)) := by
  have hc : (oddPart d).Coprime 2 := Nat.coprime_comm.mpr
    ((Nat.prime_two.coprime_iff_not_dvd).mpr (oddPart_not_dvd d hd))
  exact (ZMod.isUnit_iff_coprime _ _).mpr (hc.pow_right p)

theorem extra_le_bound (d B : ℕ) (hd : 0 < d) (hb : d ≤ 2^B) : extra d ≤ B := by
  have hv : 2^(extra d) ≤ d := Nat.le_of_dvd hd pow_padicValNat_dvd
  by_contra! h
  have hp : 2^B < 2^(extra d) := Nat.pow_lt_pow_right (by decide) h
  omega

theorem numerator_divisible (n : ℤ) (d : ℕ) (hd : 0 < d) (q : integralRationals)
    (hrep : (n:ℚ)=(d:ℚ)*q.val) : (2:ℤ)^(extra d) ∣ n := by
  by_cases hn : n=0
  · subst n; exact dvd_zero _
  have hdQ : (d:ℚ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hd)
  have hnQ : (n:ℚ) ≠ 0 := by exact_mod_cast hn
  have hq : q.val ≠ 0 := by intro h; simp only [h,mul_zero] at hrep; exact hnQ hrep
  have hval := q.property.resolve_left hq
  have hv : (extra d:ℤ) ≤ padicValInt 2 n := by
    have H : padicValRat 2 (n:ℚ)=padicValRat 2 (d:ℚ)+padicValRat 2 q.val := by
      rw [hrep,padicValRat.mul hdQ hq]
    rw [padicValRat.of_int,padicValRat.of_nat] at H
    dsimp [extra]
    omega
  have hi : extra d ≤ padicValNat 2 n.natAbs := by
    change (extra d:ℤ) ≤ (padicValInt 2 n:ℤ) at hv
    exact_mod_cast hv
  have hN : 2^(extra d) ∣ n.natAbs :=
    (Nat.pow_dvd_iff_le_padicValNat (by decide) (Int.natAbs_ne_zero.mpr hn)).mpr hi
  have HD : ((2^(extra d):ℕ):ℤ) ∣ n := Int.natCast_dvd.mpr hN
  simpa only [Nat.cast_pow,Nat.cast_ofNat] using HD

def quotientResidue (n : ℤ) (d p : ℕ) : ZMod (2^p) :=
  ((n % (2:ℤ)^(p+extra d))/(2:ℤ)^(extra d) : ℤ)

theorem quotientResidue_eq (n : ℤ) (d p : ℕ) (hn : (2:ℤ)^(extra d) ∣ n) :
    quotientResidue n d p=((n/(2:ℤ)^(extra d):ℤ):ZMod (2^p)) := by
  let c : ℤ := 2^(extra d)
  let M : ℤ := 2^p
  have hnc : c ∣ n := hn
  have hc : c ≠ 0 := by dsimp [c]; positivity
  have hN : n % (c*M) ≡ n [ZMOD c*M] := by
    change n % (c*M) % (c*M)=n % (c*M)
    exact Int.emod_emod _ _
  have hDiff : c ∣ n-n%(c*M) := (dvd_mul_right c M).trans (Int.modEq_iff_dvd.mp hN)
  have hr : c ∣ n%(c*M) := by
    have H := Int.dvd_sub hnc hDiff
    simpa only [sub_sub_cancel] using H
  have hh : c*(n%(c*M)/c) ≡ c*(n/c) [ZMOD c*M] := by
    simpa only [Int.mul_ediv_cancel' hr,Int.mul_ediv_cancel' hnc] using hN
  have he := Int.ModEq.mul_left_cancel' hc hh
  apply (ZMod.intCast_eq_intCast_iff _ _ _).mpr
  have hpow : (2:ℤ)^(p+extra d)=c*M := by dsimp [c,M]; rw [pow_add]; ring
  simpa only [quotientResidue,hpow,c,M,Nat.cast_pow,Nat.cast_ofNat] using he

def reduceCleared (n : ℤ) (d p : ℕ) : ZMod (2^p) :=
  quotientResidue n d p*(oddPart d : ZMod (2^p))⁻¹

theorem quotient_representation (n : ℤ) (d : ℕ) (hd : 0 < d) (q : integralRationals)
    (hrep : (n:ℚ)=(d:ℚ)*q.val) :
    ((n/(2:ℤ)^(extra d):ℤ):ℚ)=(oddPart d:ℚ)*q.val := by
  have hcQ : ((2:ℤ)^(extra d):ℚ) ≠ 0 := by positivity
  apply mul_left_cancel₀ hcQ
  have hdiv := Int.mul_ediv_cancel' (numerator_divisible n d hd q hrep)
  have hfactor : ((2:ℤ)^(extra d):ℚ)*(oddPart d:ℚ)=(d:ℚ) := by
    exact_mod_cast denominator_factor d
  calc
    _ = (n:ℚ) := by exact_mod_cast hdiv
    _ = (d:ℚ)*q.val := hrep
    _ = _ := by rw [← hfactor]; ring

theorem reduceCleared_correct (n : ℤ) (d p : ℕ) (hd : 0 < d) (q : integralRationals)
    (hrep : (n:ℚ)=(d:ℚ)*q.val) : reduceCleared n d p=reduce p q := by
  have H : ((n/(2:ℤ)^(extra d):ℤ):integralRationals)=(oddPart d:integralRationals)*q := by
    apply Subtype.ext
    exact quotient_representation n d hd q hrep
  have he := congrArg (reduce p) H
  simp only [map_intCast,map_mul,map_natCast] at he
  unfold reduceCleared
  rw [quotientResidue_eq n d p (numerator_divisible n d hd q hrep),he]
  calc
    _ = reduce p q*((oddPart d:ZMod (2^p))*(oddPart d:ZMod (2^p))⁻¹) := by ring
    _ = _ := by rw [ZMod.mul_inv_of_unit _ (oddPart_unit d p hd),mul_one]

end OrientationModularReduction

open Polynomial Finset

abbrev Box (R : Type*) (h z : ℕ) := Fin h → Fin z → R

def sample {R : Type*} [CommRing R] (h z : ℕ) (P : (R[X])[X]) : Box R h z :=
  fun b n => (P.coeff b.val).coeff n.val

def extend {R : Type*} [Zero R] {h z : ℕ} (A : Box R h z) (b n : ℕ) : R :=
  if hb : b<h then if hn : n<z then A ⟨b,hb⟩ ⟨n,hn⟩ else 0 else 0

def add {R : Type*} [Add R] {h z : ℕ} (A B : Box R h z) : Box R h z :=
  fun b n => A b n+B b n

def neg {R : Type*} [Neg R] {h z : ℕ} (A : Box R h z) : Box R h z :=
  fun b n => -A b n

def mul {R : Type*} [CommRing R] {h z : ℕ} (A B : Box R h z) : Box R h z :=
  fun b n => ∑ ij ∈ antidiagonal b.val,∑ rs ∈ antidiagonal n.val,
    extend A ij.1 rs.1*extend B ij.2 rs.2

theorem extend_sample {R : Type*} [CommRing R] (h z : ℕ) (P : (R[X])[X])
    (b n : ℕ) (hb : b<h) (hn : n<z) :
    extend (sample h z P) b n=(P.coeff b).coeff n := by
  simp only [extend,hb,hn,dite_true,sample]

theorem sample_add {R : Type*} [CommRing R] (h z : ℕ) (P Q : (R[X])[X]) :
    sample h z (P+Q)=add (sample h z P) (sample h z Q) := by
  funext b n
  simp only [sample,add,coeff_add]

theorem sample_neg {R : Type*} [CommRing R] (h z : ℕ) (P : (R[X])[X]) :
    sample h z (-P)=neg (sample h z P) := by
  funext b n
  simp only [sample,neg,coeff_neg]

theorem sample_mul {R : Type*} [CommRing R] (h z : ℕ) (P Q : (R[X])[X]) :
    sample h z (P*Q)=mul (sample h z P) (sample h z Q) := by
  funext b n
  simp only [sample,mul,coeff_mul,finsetSum_coeff]
  apply sum_congr rfl
  intro ij hij
  apply sum_congr rfl
  intro rs hrs
  have hij' : ij.1+ij.2=b.val := mem_antidiagonal.mp hij
  have hrs' : rs.1+rs.2=n.val := mem_antidiagonal.mp hrs
  rw [extend_sample h z P ij.1 rs.1 (by omega) (by omega)]
  rw [extend_sample h z Q ij.2 rs.2 (by omega) (by omega)]

inductive Expr where
  | literal (value : ℤ)
  | inner
  | outer
  | add (a b : Expr)
  | neg (a : Expr)
  | mul (a b : Expr)

def fullEval {R : Type*} [CommRing R] : Expr → (R[X])[X]
  | .literal v => C (C (v:R))
  | .inner => C X
  | .outer => X
  | .add a b => fullEval a+fullEval b
  | .neg a => -fullEval a
  | .mul a b => fullEval a*fullEval b

def boxEval {R : Type*} [CommRing R] (h z : ℕ) : Expr → Box R h z
  | .literal v => sample h z (C (C (v:R)))
  | .inner => sample h z (C X)
  | .outer => sample h z X
  | .add a b => add (boxEval h z a) (boxEval h z b)
  | .neg a => neg (boxEval h z a)
  | .mul a b => mul (boxEval h z a) (boxEval h z b)

theorem boxEval_correct {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    boxEval h z e=sample h z (fullEval e : (R[X])[X]) := by
  induction e with
  | literal v => rfl
  | inner => rfl
  | outer => rfl
  | add a b ha hb => rw [boxEval,ha,hb,fullEval,sample_add]
  | neg a ha => rw [boxEval,ha,fullEval,sample_neg]
  | mul a b ha hb => rw [boxEval,ha,hb,fullEval,sample_mul]

theorem fullEval_map {R S : Type*} [CommRing R] [CommRing S]
    (f : R →+* S) (e : Expr) :
    (Polynomial.mapRingHom (Polynomial.mapRingHom f)) (fullEval e)=
      (fullEval e : (S[X])[X]) := by
  induction e with
  | literal v => simp only [fullEval,map_intCast]
  | inner => simp only [fullEval,Polynomial.coe_mapRingHom,map_C,map_X]
  | outer => simp only [fullEval,Polynomial.coe_mapRingHom,map_X]
  | add a b ha hb => simp only [fullEval,map_add,ha,hb]
  | neg a ha => simp only [fullEval,map_neg,ha]
  | mul a b ha hb => simp only [fullEval,map_mul,ha,hb]

theorem modular_box_coefficient (q h z : ℕ) (e : Expr) (b : Fin h) (n : Fin z) :
    boxEval (R:=ZMod (2^q)) h z e b n=
      ((((fullEval e : (ℤ[X])[X]).coeff b.val).coeff n.val:ℤ):ZMod (2^q)) := by
  rw [boxEval_correct]
  unfold sample
  rw [← fullEval_map (Int.castRingHom (ZMod (2^q))) e]
  simp only [Polynomial.coe_mapRingHom,coeff_map,Int.coe_castRingHom]

def multiplicationProducts (h z : ℕ) : ℕ :=
  ∑ b ∈ range h,∑ n ∈ range z,(b+1)*(n+1)

theorem multiplicationProducts_bound (h z : ℕ) : multiplicationProducts h z≤(h*z)^2 := by
  unfold multiplicationProducts
  calc
    _ ≤ ∑ b ∈ range h,∑ n ∈ range z,h*z := by
      apply sum_le_sum
      intro b hb
      apply sum_le_sum
      intro n hn
      exact Nat.mul_le_mul (Nat.succ_le_of_lt (mem_range.mp hb))
        (Nat.succ_le_of_lt (mem_range.mp hn))
    _ = (h*z)^2 := by simp; ring

theorem coefficient_slots (h z : ℕ) : Fintype.card (Fin h × Fin z)=h*z := by simp

end OrientationCoefficientBox

def exprPow (e : Expr) : ℕ → Expr
  | 0 => .literal 1
  | k+1 => .mul (exprPow e k) e

theorem exprPow_correct {R : Type*} [CommRing R] (e : Expr) (k : ℕ) :
    (fullEval (exprPow e k) : (R[X])[X])=(fullEval e)^k := by
  induction k with
  | zero => simp only [exprPow,fullEval,Int.cast_one,map_one,pow_zero]
  | succ k ih => simp only [exprPow,fullEval,ih,pow_succ]

structure Code {P : (ℚ[X])[X]} (r : Rep P) where
  expr : Expr
  sound : (fullEval expr : (ℤ[X])[X])=r.num

def Code.rational (q : ℚ) : Code (Rep.rational q) :=
  ⟨.literal q.num,rfl⟩
def Code.one : Code Rep.one := ⟨.literal 1,by simp only [fullEval,Int.cast_one,map_one]; rfl⟩
def Code.zero : Code Rep.zero := ⟨.literal 0,by simp only [fullEval,Int.cast_zero,map_zero]; rfl⟩
def Code.innerX : Code Rep.innerX := ⟨.inner,rfl⟩
def Code.outerX : Code Rep.outerX := ⟨.outer,rfl⟩

def Code.cast {P Q : (ℚ[X])[X]} {r : Rep P} (c : Code r) (h : P=Q) : Code (r.cast h) :=
  ⟨c.expr,c.sound⟩

def Code.mul {P Q : (ℚ[X])[X]} {r : Rep P} {s : Rep Q}
    (c : Code r) (d : Code s) : Code (r.mul s) :=
  ⟨.mul c.expr d.expr,by simp only [fullEval,c.sound,d.sound]; rfl⟩

def Code.add {P Q : (ℚ[X])[X]} {r : Rep P} {s : Rep Q}
    (c : Code r) (d : Code s) : Code (r.add s) :=
  ⟨.add (.mul (.literal (s.den:ℤ)) c.expr) (.mul (.literal (r.den:ℤ)) d.expr),by
    simp only [fullEval,c.sound,d.sound,Int.cast_id]; rfl⟩

def Code.neg {P : (ℚ[X])[X]} {r : Rep P} (c : Code r) : Code r.neg :=
  ⟨.neg c.expr,by simp only [fullEval,c.sound]; rfl⟩

def Code.sub {P Q : (ℚ[X])[X]} {r : Rep P} {s : Rep Q}
    (c : Code r) (d : Code s) : Code (r.sub s) :=
  (c.add d.neg).cast (by rw [sub_eq_add_neg])

def Code.pow {P : (ℚ[X])[X]} {r : Rep P} (c : Code r) (k : ℕ) : Code (r.pow k) :=
  ⟨exprPow c.expr k,by rw [exprPow_correct,c.sound]; rfl⟩

def exprProduct : List Expr → Expr
  | [] => .literal 1
  | e::es => .mul e (exprProduct es)

theorem exprProduct_correct (es : List Expr) :
    (fullEval (exprProduct es) : (ℤ[X])[X])=(es.map fullEval).prod := by
  induction es with
  | nil => simp only [exprProduct,fullEval,Int.cast_one,map_one,List.map_nil,List.prod_nil]
  | cons e es ih => simp only [exprProduct,fullEval,ih,List.map_cons,List.prod_cons]

def Code.prod {I : Type*} [LinearOrder I] (S : Finset I) (P : I → (ℚ[X])[X])
    (r : ∀ i,Rep (P i)) (c : ∀ i,Code (r i)) : Code (Rep.prod S P r) := by
  refine ⟨exprProduct ((S.sort (·≤·)).map (fun i => (c i).expr)),?_⟩
  rw [exprProduct_correct]
  rw [List.map_map]
  change ((S.sort (·≤·)).map (fun i => (fullEval (c i).expr : (ℤ[X])[X]))).prod=_
  simp_rw [(c _).sound]
  change ((S.sort (·≤·)).map (fun i => (r i).num)).prod=∏ i ∈ S,(r i).num
  calc
    _ = (S.toList.map (fun i => (r i).num)).prod :=
      ((S.sort_perm_toList (·≤·)).map (fun i => (r i).num)).prod_eq
    _ = _ := S.prod_map_toList _

def clearedExpr (d : ℤ) (A : Expr) : ℕ → Expr

/-- Runtime payload has only scalar denominator and syntax, no polynomial. -/
structure Packed where
  den : ℕ
  expr : Expr

def Packed.rational (q : ℚ) : Packed := ⟨q.den,.literal q.num⟩
def Packed.one : Packed := ⟨1,.literal 1⟩
def Packed.zero : Packed := ⟨1,.literal 0⟩
def Packed.innerX : Packed := ⟨1,.inner⟩
def Packed.outerX : Packed := ⟨1,.outer⟩
def Packed.mul (s t : Packed) : Packed := ⟨s.den*t.den,.mul s.expr t.expr⟩
def Packed.add (s t : Packed) : Packed :=
  ⟨s.den*t.den,.add (.mul (.literal (t.den:ℤ)) s.expr) (.mul (.literal (s.den:ℤ)) t.expr)⟩
def Packed.neg (s : Packed) : Packed := ⟨s.den,.neg s.expr⟩
def Packed.sub (s t : Packed) : Packed := s.add t.neg
def Packed.pow (s : Packed) (k : ℕ) : Packed := ⟨s.den^k,exprPow s.expr k⟩
def Packed.prod {I : Type*} [LinearOrder I] (S : Finset I) (s : I → Packed) : Packed :=
  ⟨∏ i ∈ S,(s i).den,exprProduct ((S.sort (·≤·)).map (fun i => (s i).expr))⟩
def Packed.geometric (s : Packed) (k : ℕ) : Packed :=
  ⟨s.den^k,clearedExpr (s.den:ℤ) s.expr k⟩

/-- This relation is a proof interface, not runtime Packed input. -/
def Matches {P : (ℚ[X])[X]} (s : Packed) (r : Rep P) : Prop :=
  s.den=r.den ∧ (fullEval s.expr : (ℤ[X])[X])=r.num

theorem matches_rational (q : ℚ) : Matches (Packed.rational q) (Rep.rational q) := ⟨rfl,rfl⟩
theorem matches_one : Matches Packed.one Rep.one := by
  constructor
  · rfl
  · simp only [Packed.one,fullEval,Int.cast_one,map_one]; rfl

theorem matches_zero : Matches Packed.zero Rep.zero := by
  constructor
  · rfl
  · simp only [Packed.zero,fullEval,Int.cast_zero,map_zero]; rfl

theorem matches_innerX : Matches Packed.innerX Rep.innerX := ⟨rfl,rfl⟩
theorem matches_outerX : Matches Packed.outerX Rep.outerX := ⟨rfl,rfl⟩

theorem Matches.cast {P Q : (ℚ[X])[X]} {s : Packed} {r : Rep P}
    (h : Matches s r) (hPQ : P=Q) : Matches s (r.cast hPQ) := h

theorem Matches.mul {P Q : (ℚ[X])[X]} {s t : Packed} {r : Rep P} {v : Rep Q}
    (hs : Matches s r) (ht : Matches t v) : Matches (s.mul t) (r.mul v) := by
  constructor
  · exact congrArg₂ (·*·) hs.1 ht.1
  · simp only [Packed.mul,fullEval,hs.2,ht.2]; rfl

theorem Matches.add {P Q : (ℚ[X])[X]} {s t : Packed} {r : Rep P} {v : Rep Q}
    (hs : Matches s r) (ht : Matches t v) : Matches (s.add t) (r.add v) := by
  constructor
  · exact congrArg₂ (·*·) hs.1 ht.1
  · simp only [Packed.add,fullEval,hs.1,ht.1,hs.2,ht.2,Int.cast_id]; rfl

theorem Matches.neg {P : (ℚ[X])[X]} {s : Packed} {r : Rep P}
    (h : Matches s r) : Matches s.neg r.neg := by
  constructor
  · exact h.1
  · simp only [Packed.neg,fullEval,h.2]; rfl

theorem Matches.sub {P Q : (ℚ[X])[X]} {s t : Packed} {r : Rep P} {v : Rep Q}
    (hs : Matches s r) (ht : Matches t v) : Matches (s.sub t) (r.sub v) :=
  (hs.add ht.neg).cast (by rw [sub_eq_add_neg])

theorem Matches.pow {P : (ℚ[X])[X]} {s : Packed} {r : Rep P}
    (h : Matches s r) (k : ℕ) : Matches (s.pow k) (r.pow k) := by
  constructor
  · exact congrArg (fun n : ℕ => n^k) h.1
  · simp only [Packed.pow,exprPow_correct,h.2]; rfl

theorem matches_prod {I : Type*} [LinearOrder I] (S : Finset I) (P : I → (ℚ[X])[X])
    (s : I → Packed) (r : ∀ i,Rep (P i)) (h : ∀ i,Matches (s i) (r i)) :
    Matches (Packed.prod S s) (Rep.prod S P r) := by
  constructor
  · dsimp [Packed.prod,Rep.prod]
    apply prod_congr rfl
    intro i _
    exact (h i).1
  · rw [Packed.prod,exprProduct_correct,List.map_map]
    change ((S.sort (·≤·)).map (fun i => (fullEval (s i).expr : (ℤ[X])[X]))).prod=_
    simp_rw [(h _).2]
    change ((S.sort (·≤·)).map (fun i => (r i).num)).prod=∏ i ∈ S,(r i).num
    calc
      _ = (S.toList.map (fun i => (r i).num)).prod :=
        ((S.sort_perm_toList (·≤·)).map (fun i => (r i).num)).prod_eq
      _ = _ := S.prod_map_toList _

theorem Matches.geometric {P : (ℚ[X])[X]} {s : Packed} {r : Rep P}
    (h : Matches s r) (k : ℕ) : Matches (s.geometric k) (r.geometric k) := by
  constructor
  · exact congrArg (fun n : ℕ => n^k) h.1
  · simp only [Packed.geometric,clearedExpr_correct,h.1,h.2]; rfl

theorem Matches.positive {P : (ℚ[X])[X]} {s : Packed} {r : Rep P}
    (h : Matches s r) : 0<s.den := by rw [h.1]; exact r.positive

end OrientationScalarCode

def nodes : Expr → ℕ
  | .literal _ => 1
  | .inner => 1
  | .outer => 1
  | .add a b => 1+nodes a+nodes b
  | .neg a => 1+nodes a
  | .mul a b => 1+nodes a+nodes b

theorem exprPow_nodes (e : Expr) (k : ℕ) : nodes (exprPow e k)=k*(nodes e+1)+1 := by
  induction k with
  | zero => simp only [exprPow,nodes,Nat.zero_mul,Nat.zero_add]
  | succ k ih => simp only [exprPow,nodes,ih,Nat.succ_mul]; ring

theorem exprProduct_nodes (es : List Expr) :
    nodes (exprProduct es)=1+(es.map (fun e => nodes e+1)).sum := by
  induction es with
  | nil => rfl
  | cons e es ih => simp only [exprProduct,nodes,ih,List.map_cons,List.sum_cons]; omega

    nlinarith

theorem packed_prod_nodes {I : Type*} [LinearOrder I] (S : Finset I) (s : I → Packed) :
    nodes (Packed.prod S s).expr=1+∑ i ∈ S,(nodes (s i).expr+1) := by
  dsimp only [Packed.prod]
  rw [exprProduct_nodes,List.map_map]
  change 1+((S.sort (·≤·)).map (fun i => nodes (s i).expr+1)).sum=_
  apply congrArg (1+·)
  calc
    _ = (S.toList.map (fun i => nodes (s i).expr+1)).sum :=
      ((S.sort_perm_toList (·≤·)).map (fun i => nodes (s i).expr+1)).sum_eq
    _ = _ := S.sum_map_toList _

def mass (u : ℕ) (m : ℕ → ℕ) : ℕ := ∑ j ∈ range u,m j

theorem mask_charge_bound (u x y c d : ℕ) (m : ℕ → ℕ) :
    (∑ j ∈ mask u x y,(c*m j+d))≤c*mass u m+d*u := by
  have hS : mask u x y⊆range u := by unfold mask; exact filter_subset _ _
  have H := sum_le_sum_of_subset (f:=fun j => c*m j+d) hS
  refine H.trans_eq ?_
  rw [sum_add_distrib,← mul_sum]
  simp only [mass,sum_const,card_range,nsmul_eq_mul,Nat.cast_id]
  ring

def rowNodeBound (a u : ℕ) (m : ℕ → ℕ) : ℕ := 17+20*2^a+13*mass u m+2*u

def denomNodeBound (u : ℕ) (m : ℕ → ℕ) : ℕ := 19+10*mass u m+2*u

theorem packedRow_nodes_bound (a u i x y : ℕ) (t : ℤ) (m : ℕ → ℕ) :
    nodes (packedRow a u i x y t m).expr≤rowNodeBound a u m := by
  rw [packedRow_nodes]
  have hC : ((grid a).erase (delta (rowResidue u x) y)).card≤2*2^a := by
    rw [← grid_card]
    exact card_erase_le
  have H := mask_charge_bound u (rowResidue u x) y 13 2 m
  unfold rowNodeBound
  omega

    nodes (packedDenom a u x y t m).expr≤denomNodeBound u m := by
  rw [packedDenom_nodes]
  have H := mask_charge_bound u x y 10 2 m
  unfold denomNodeBound
  omega

def approximationNodeBound (a u k : ℕ) (m : ℕ → ℕ) : ℕ :=
  1+rowNodeBound a u m+(denomNodeBound u m+10)*(k+1)^2

  have hR := packedRow_nodes_bound a u i x y t m

  have hG := clearedExpr_nodes_bound ((packedDenom a u x y t m).den:ℤ)
    (packedDenom a u x y t m).expr k
  simp only [packedApprox,Packed.mul,Packed.geometric,nodes,approximationNodeBound]
  have H := Nat.mul_le_mul_right ((k+1)^2) (Nat.add_le_add_right hD 10)
  omega

  | .literal _ => 0
  | .inner => 0
  | .outer => 0
  | .add a b => scalarProducts h z a+scalarProducts h z b
  | .neg a => scalarProducts h z a
  | .mul a b => scalarProducts h z a+scalarProducts h z b+multiplicationProducts h z

theorem scalarProducts_bound (h z : ℕ) (e : Expr) :
    scalarProducts h z e≤nodes e*(h*z)^2 := by

  | inner => simp only [scalarProducts,nodes]; positivity
  | outer => simp only [scalarProducts,nodes]; positivity
  | add a b ha hb => simp only [scalarProducts,nodes]; nlinarith
  | neg a ha => simp only [scalarProducts,nodes]; nlinarith
  | mul a b ha hb => simp only [scalarProducts,nodes]; nlinarith

  fun b n => A[b.val][n.val]

def evaluate {R : Type*} [CommRing R] (h z : ℕ) : Expr → Stored R h z
  | .literal v => materialize (sample h z (C (C (v:R))))
  | .inner => materialize (sample h z (C X))
  | .outer => materialize (sample h z X)
  | .add a b =>
      let A := evaluate h z a
      let B := evaluate h z b
      materialize (OrientationCoefficientBox.add (view A) (view B))
  | .neg a =>
      let A := evaluate h z a
      materialize (OrientationCoefficientBox.neg (view A))
  | .mul a b =>
      let A := evaluate h z a
      let B := evaluate h z b
      materialize (OrientationCoefficientBox.mul (view A) (view B))

theorem evaluate_correct {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    view (evaluate h z e)=sample h z (fullEval e : (R[X])[X]) := by
  induction e with
  | literal v => exact view_materialize _
  | inner => exact view_materialize _
  | outer => exact view_materialize _
  | add a b ha hb => simp only [evaluate,view_materialize,ha,hb,fullEval,sample_add]
  | neg a ha => simp only [evaluate,view_materialize,ha,fullEval,sample_neg]
  | mul a b ha hb => simp only [evaluate,view_materialize,ha,hb,fullEval,sample_mul]

def dotLogged {R : Type*} [CommRing R] : List (R × R) → Logged R
  | [] => ⟨0,[]⟩
  | xy::rest =>
      let p := emitMultiply xy.1 xy.2
      let t := dotLogged rest
      let s := emitAdd p.value t.value
      ⟨s.value,p.events++t.events++s.events⟩

def emitNegate {R : Type*} [Neg R] (x : R) : Logged R := ⟨-x,[.negate]⟩

theorem multiplyLogged_cast {R : Type*} [CommRing R] {h z : ℕ} (A B : Stored R h z) :
    (multiplyLogged A B).events.count .cast=0 := by
  simp only [multiplyLogged,materializeLogged_count,convolutionLogged,dotLogged_cast,sum_const_zero]

def evaluateLogged {R : Type*} [CommRing R] (h z : ℕ) : Expr → Logged (Stored R h z)
  | .literal v =>
      let c := emitCast (R:=R) v
      let A := materializeLogged (fun b n => (⟨sample h z (C (C c.value)) b n,[]⟩ : Logged R))
      ⟨A.value,c.events++A.events⟩
  | .inner => ⟨materialize (sample h z (C X)),[]⟩
  | .outer => ⟨materialize (sample h z X),[]⟩
  | .add a b =>
      let A := evaluateLogged h z a
      let B := evaluateLogged h z b
      let C := materializeLogged (fun r n => emitAdd (view A.value r n) (view B.value r n))
      ⟨C.value,A.events++B.events++C.events⟩
  | .neg a =>
      let A := evaluateLogged h z a
      let B := materializeLogged (fun r n => emitNegate (view A.value r n))
      ⟨B.value,A.events++B.events⟩
  | .mul a b =>
      let A := evaluateLogged h z a
      let B := evaluateLogged h z b
      let C := multiplyLogged A.value B.value
      ⟨C.value,A.events++B.events++C.events⟩

theorem evaluateLogged_value {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    view (evaluateLogged h z e).value=sample h z (fullEval e : (R[X])[X]) := by
  induction e with
  | literal v => simp only [evaluateLogged,emitCast,materializeLogged_value,fullEval]
  | inner => exact view_materialize _
  | outer => exact view_materialize _
  | add a b ha hb =>
    simp only [evaluateLogged,materializeLogged_value,emitAdd,ha,hb,fullEval,sample_add]
    unfold OrientationCoefficientBox.add
    rfl
  | neg a ha =>
    simp only [evaluateLogged,materializeLogged_value,emitNegate,ha,fullEval,sample_neg]
    unfold OrientationCoefficientBox.neg
    rfl
  | mul a b ha hb =>
    simp only [evaluateLogged,multiplyLogged_value,ha,hb,fullEval,sample_mul]

  | mul a b ha hb => simp only [evaluateLogged,List.count_append,multiplyLogged_multiply,ha,hb,scalarProducts]

def LiteralBound (B : ℕ) : Expr → Prop
  | .literal n => n.natAbs≤2^B
  | .inner => True
  | .outer => True
  | .add a b => LiteralBound B a ∧ LiteralBound B b
  | .neg a => LiteralBound B a
  | .mul a b => LiteralBound B a ∧ LiteralBound B b

theorem LiteralBound.mono {B C : ℕ} {e : Expr} (h : LiteralBound B e) (hBC : B≤C) :
    LiteralBound C e := by
  revert h
  induction e with
  | literal n =>
    intro h
    exact h.trans (Nat.pow_le_pow_right (by decide) hBC)
  | inner => intro _; trivial
  | outer => intro _; trivial
  | add a b ha hb => intro h; exact ⟨ha h.1,hb h.2⟩
  | neg a ha => intro h; exact ha h
  | mul a b ha hb => intro h; exact ⟨ha h.1,hb h.2⟩

theorem literal_one (B : ℕ) : LiteralBound B (.literal 1) := by
  change 1≤2^B
  exact Nat.one_le_two_pow

theorem literal_zero (B : ℕ) : LiteralBound B (.literal 0) := by
  change 0≤2^B
  exact Nat.zero_le _

theorem LiteralBound.power_uniform {B : ℕ} {e : Expr} (h : LiteralBound B e) (k : ℕ) :
    LiteralBound B (exprPow e k) := by
  induction k with
  | zero => exact literal_one B
  | succ k ih => exact ⟨ih,h⟩

theorem LiteralBound.power {B : ℕ} {e : Expr} (h : LiteralBound B e) (k : ℕ) :
    LiteralBound (B*k) (exprPow e k) := by
  by_cases hk : k=0
  · subst k
    exact literal_one 0
  · have hBk : B≤B*k := by
      calc
        B = B*1 := (Nat.mul_one B).symm
        _ ≤ B*k := Nat.mul_le_mul_left B (Nat.one_le_iff_ne_zero.mpr hk)
    exact (h.power_uniform k).mono hBk

theorem LiteralBound.product {B : ℕ} (es : List Expr)
    (h : ∀ e∈es,LiteralBound B e) : LiteralBound B (exprProduct es) := by
  induction es with
  | nil => exact literal_one B
  | cons e es ih =>
    exact ⟨h e (List.mem_cons_self),ih (by intro f hf; exact h f (List.mem_cons_of_mem _ hf))⟩

theorem LiteralBound.cleared {B : ℕ} {d : ℤ} {A : Expr}
    (hd : d.natAbs≤2^B) (hA : LiteralBound B A) (k : ℕ) :
    LiteralBound B (clearedExpr d A k) := by
  induction k with
  | zero => exact literal_zero B
  | succ k ih =>
    exact ⟨⟨hd,ih⟩,⟨hd,(LiteralBound.power_uniform (e:=.add (.literal d) (.neg A)) ⟨hd,hA⟩ k)⟩⟩

def Budget (B : ℕ) (s : Packed) : Prop :=
  0<s.den ∧ s.den≤2^B ∧ LiteralBound B s.expr

theorem Budget.mono {B C : ℕ} {s : Packed} (h : Budget B s) (hBC : B≤C) : Budget C s :=
  ⟨h.1,h.2.1.trans (Nat.pow_le_pow_right (by decide) hBC),h.2.2.mono hBC⟩

theorem budget_rational {B : ℕ} {q : ℚ} (h : Bits B q) : Budget B (Packed.rational q) :=
  ⟨q.den_pos,(le_max_right _ _).trans h,(le_max_left _ _).trans h⟩

theorem budget_one : Budget 0 Packed.one := by
  norm_num [Budget,Packed.one,LiteralBound]

theorem budget_zero : Budget 0 Packed.zero := by
  norm_num [Budget,Packed.zero,LiteralBound]

theorem budget_innerX : Budget 0 Packed.innerX := by
  norm_num [Budget,Packed.innerX,LiteralBound]

theorem budget_outerX : Budget 0 Packed.outerX := by
  norm_num [Budget,Packed.outerX,LiteralBound]

theorem Budget.mul {B C : ℕ} {s t : Packed} (hs : Budget B s) (ht : Budget C t) :
    Budget (B+C) (s.mul t) := by
  refine ⟨Nat.mul_pos hs.1 ht.1,
    (Nat.mul_le_mul hs.2.1 ht.2.1).trans_eq (pow_add _ _ _).symm,?_⟩
  exact ⟨hs.2.2.mono (by omega),ht.2.2.mono (by omega)⟩

theorem Budget.add {B C : ℕ} {s t : Packed} (hs : Budget B s) (ht : Budget C t) :
    Budget (B+C) (s.add t) := by
  have hs' := hs.mono (show B≤B+C by omega)
  have ht' := ht.mono (show C≤B+C by omega)
  refine ⟨Nat.mul_pos hs.1 ht.1,
    (Nat.mul_le_mul hs.2.1 ht.2.1).trans_eq (pow_add _ _ _).symm,?_⟩
  change (((t.den:ℤ).natAbs≤2^(B+C) ∧ LiteralBound (B+C) s.expr) ∧
    ((s.den:ℤ).natAbs≤2^(B+C) ∧ LiteralBound (B+C) t.expr))
  simpa only [Int.natAbs_natCast] using ⟨⟨ht'.2.1,hs'.2.2⟩,⟨hs'.2.1,ht'.2.2⟩⟩

theorem Budget.neg {B : ℕ} {s : Packed} (h : Budget B s) : Budget B s.neg := h

theorem Budget.sub {B C : ℕ} {s t : Packed} (hs : Budget B s) (ht : Budget C t) :
    Budget (B+C) (s.sub t) := hs.add ht.neg

theorem Budget.power {B : ℕ} {s : Packed} (h : Budget B s) (k : ℕ) : Budget (B*k) (s.pow k) := by
  refine ⟨pow_pos h.1 k,?_,h.2.2.power k⟩
  exact (pow_le_pow_left₀ (Nat.zero_le _) h.2.1 k).trans_eq (pow_mul _ _ _).symm

theorem budget_product {I : Type*} [LinearOrder I] (S : Finset I)
    (s : I → Packed) (B : I → ℕ) (h : ∀ i∈S,Budget (B i) (s i)) :
    Budget (∑ i∈S,B i) (Packed.prod S s) := by
  refine ⟨prod_pos (by intro i hi; exact (h i hi).1),?_,?_⟩
  · apply (bits_nat_iff _ _).mp
    have H := bits_prod S (fun i => ((s i).den:ℚ)) B
      (by intro i hi; exact bits_nat _ _ (h i hi).2.1)
    simpa only [Packed.prod,Nat.cast_prod] using H
  · apply LiteralBound.product
    intro e he
    rcases List.mem_map.mp he with ⟨i,hi,rfl⟩
    have hiS : i∈S := (Finset.mem_sort (·≤·)).mp hi
    exact (h i hiS).2.2.mono (single_le_sum (by intro j _; exact Nat.zero_le (B j)) hiS)

theorem Budget.geometric {B : ℕ} {s : Packed} (h : Budget B s) (k : ℕ) :
    Budget (B*k) (s.geometric k) := by
  refine ⟨pow_pos h.1 k,
    (pow_le_pow_left₀ (Nat.zero_le _) h.2.1 k).trans_eq (pow_mul _ _ _).symm,?_⟩
  by_cases hk : k=0
  · subst k
    exact literal_zero 0
  · have hBk : B≤B*k := by
      calc
        B = B*1 := (Nat.mul_one B).symm
        _ ≤ B*k := Nat.mul_le_mul_left B (Nat.one_le_iff_ne_zero.mpr hk)
    have hd : (s.den:ℤ).natAbs≤2^B := by simpa only [Int.natAbs_natCast] using h.2.1
    exact (LiteralBound.cleared hd h.2.2 k).mono hBk

end OrientationLiteralBudget

def affineBudget (a : ℕ) : ℕ := (a+a+1)+(a+1)
def factorBudget (a j : ℕ) : ℕ := ((j+a+1)+(a+1))+(j+a+1)
def punctureBudget (a : ℕ) : ℕ := (2*2^a)*affineBudget a
def correctionBudget (a u : ℕ) (m : ℕ → ℕ) : ℕ := ∑ j∈range u,factorBudget a j*m j
def completedBudget (a u : ℕ) (m : ℕ → ℕ) : ℕ :=
  (derivativeSize a+punctureBudget a)+correctionBudget a u m
def normalizedBudget (a u i : ℕ) (m : ℕ → ℕ) : ℕ := diagonalSize a u i m+completedBudget a u m
def rowBudget (a u i : ℕ) (m : ℕ → ℕ) : ℕ := ((a+1)+(a+a+1))+normalizedBudget a u i m

theorem actual_root_budget (a : ℕ) (t : ℤ) (s : ℚ)
    (ht : (t:ℚ)∈grid a) (hs : s∈grid a) :
    Budget (affineBudget a) (packedChartRoot a t s) := by
  have H := (budget_rational ((actual_grid_bits a t ht).sub (actual_grid_member_bits a s hs))).add
    ((budget_rational (bits_two_pow (a+1))).mul budget_innerX)
  simpa only [affineBudget,packedChartRoot,Nat.add_zero] using H

theorem actual_factor_budget (a j x y : ℕ) (t : ℤ)
    (ht : (t:ℚ)∈grid a) (hx : x<2^a) (hy : y<2^a) :
    Budget (factorBudget a j) (packedChartFactor a j t (delta x y)) := by
  have H := (((budget_rational ((bits_two_pow j).sub (actual_grid_bits a t ht))).sub
    ((budget_rational (bits_two_pow (a+1))).mul budget_innerX))).mul
    (budget_rational (((bits_two_pow j).sub (actual_delta_bits a x y hx hy)).inv))
  simpa only [factorBudget,packedChartFactor,Nat.add_zero] using H

theorem actual_puncture_budget (a : ℕ) (t : ℤ) (d : ℚ) (ht : (t:ℚ)∈grid a) :
    Budget (punctureBudget a) (packedPuncture (grid a) d a t) := by
  have H := budget_product ((grid a).erase d) (packedChartRoot a t) (fun _ => affineBudget a)
    (by intro s hs; exact actual_root_budget a t s ht (mem_erase.mp hs).2)
  have hC : ((grid a).erase d).card≤2*2^a := by rw [← grid_card]; exact card_erase_le
  have hB : (∑ _∈(grid a).erase d,affineBudget a)≤punctureBudget a := by
    simpa only [sum_const,nsmul_eq_mul,Nat.cast_id,punctureBudget] using
      Nat.mul_le_mul_right (affineBudget a) hC
  exact H.mono hB

theorem actual_correction_budget (a u x y : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (ht : (t:ℚ)∈grid a) (hx : x<2^a) (hy : y<2^a) :
    Budget (correctionBudget a u m) (packedCorrection a u x y t m) := by
  have H := budget_product (mask u x y)
    (fun j => (packedChartFactor a j t (delta x y)).pow (m j))
    (fun j => factorBudget a j*m j)
    (by intro j _; exact (actual_factor_budget a j x y t ht hx hy).power (m j))
  have hS : mask u x y⊆range u := by unfold mask; exact filter_subset _ _
  exact H.mono (sum_le_sum_of_subset hS)

theorem actual_completed_budget (a u x y : ℕ) (t : ℤ) (m : ℕ → ℕ)

theorem actual_normalized_budget (a u i x y : ℕ) (t : ℤ) (m : ℕ → ℕ)

def literals : Expr → List ℤ
  | .literal n => [n]
  | .inner => []
  | .outer => []
  | .add a b => literals a++literals b
  | .neg a => literals a
  | .mul a b => literals a++literals b

theorem literals_length (e : Expr) : (literals e).length=castLeaves e := by
  induction e with
  | literal n => rfl
  | inner => rfl
  | outer => rfl
  | add a b ha hb => simp only [literals,castLeaves,List.length_append,ha,hb]
  | neg a ha => simp only [literals,castLeaves,ha]
  | mul a b ha hb => simp only [literals,castLeaves,List.length_append,ha,hb]

theorem LiteralBound.members {B : ℕ} {e : Expr} (h : LiteralBound B e) :
    ∀ n∈literals e,n.natAbs≤2^B := by
  revert h
  induction e with
  | literal m =>
    intro h n hn
    have he : n=m := List.mem_singleton.mp hn
    simpa only [he,LiteralBound] using h
  | inner => intro _ n hn; simp only [literals,List.not_mem_nil] at hn
  | outer => intro _ n hn; simp only [literals,List.not_mem_nil] at hn
  | add a b ha hb =>
    intro h n hn
    rcases List.mem_append.mp hn with hn | hn
    · exact ha h.1 n hn
    · exact hb h.2 n hn
  | neg a ha => intro h n hn; exact ha h n hn
  | mul a b ha hb =>
    intro h n hn
    rcases List.mem_append.mp hn with hn | hn
    · exact ha h.1 n hn
    · exact hb h.2 n hn

def signedWidth (n : ℤ) : ℕ := Nat.size n.natAbs+1

theorem signedWidth_bound {B : ℕ} {n : ℤ} (h : n.natAbs≤2^B) : signedWidth n≤B+2 := by
  have H := Nat.size_le_size h
  rw [Nat.size_pow] at H
  unfold signedWidth
  omega

theorem list_volume_bound {I : Type*} (xs : List I) (f : I → ℕ) (B : ℕ)
    (h : ∀ x∈xs,f x≤B) : (xs.map f).sum≤xs.length*B := by
  induction xs with
  | nil => simp only [List.map_nil,List.sum_nil,List.length_nil,Nat.zero_mul,Nat.le_refl]
  | cons x xs ih =>
    have hx := h x (List.mem_cons_self)
    have H := ih (by intro y hy; exact h y (List.mem_cons_of_mem _ hy))
    simp only [List.map_cons,List.sum_cons,List.length_cons,Nat.succ_mul]
    omega

def inputVolume (e : Expr) : ℕ := ((literals e).map signedWidth).sum

theorem LiteralBound.volume {B : ℕ} {e : Expr} (h : LiteralBound B e) :
    inputVolume e≤castLeaves e*(B+2) := by
  have H := list_volume_bound (literals e) signedWidth (B+2)
    (by intro n hn; exact signedWidth_bound (h.members n hn))
  simpa only [inputVolume,literals_length] using H

theorem LiteralBound.volume_nodes {B : ℕ} {e : Expr} (h : LiteralBound B e) :
    inputVolume e≤nodes e*(B+2) :=
  h.volume.trans (Nat.mul_le_mul_right _ (castLeaves_bound e))

theorem actual_inputVolume_bound (a u i x y k : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (ht : (t:ℚ)∈grid a) (hi : i≤a) (hu : u≤a) (hy : y<2^a) :
    inputVolume (packedApprox a u i x y k t m).expr≤
      approximationNodeBound a u k m*(approximationBudget a u i x y k m+2) :=
  ((actual_approximation_literals a u i x y k t m ht hi hu hy).volume_nodes).trans
    (Nat.mul_le_mul_right _ (packedApprox_nodes_bound a u i x y k t m))

def rightBudget (a u i r y k b : ℕ) (m : ℕ → ℕ) : ℕ :=
  b+approximationBudget a u i r y k m

theorem actual_right_den_budget (a u i k : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (rb : MomentLabels u k) (y : Ports a) (ht : (t:ℚ)∈grid a) (hi : i≤a) (hu : u≤a) :
    0<packedRightDen a u i k t m rb y ∧
    packedRightDen a u i k t m rb y≤2^(rightBudget a u i rb.1.val y.val k rb.2.val m) := by
  have hA := actual_approximation_budget a u i rb.1.val y.val k t m ht hi hu y.isLt
  have hQ := budget_rational (bits_two_pow rb.2.val).inv
  by_cases hc : rb.1.val.testBit i=y.val.testBit i
  · simp only [packedRightDen,hc,ite_true]
    exact ⟨Nat.mul_pos hQ.1 hA.1,
      (Nat.mul_le_mul hQ.2.1 hA.2.1).trans_eq (pow_add _ _ _).symm⟩
  · simp only [packedRightDen,hc,ite_false]
    exact ⟨by decide,Nat.one_le_two_pow⟩

theorem actual_right_work_precision (a u i p : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (rb : LabelsAt a u i p m) (y : Ports a) (ht : (t:ℚ)∈grid a) (hi : i≤a) (hu : u≤a) :
    p+extra (packedRightDen a u i (creationIndex p (jetWeight a u i m)) t m rb y)≤
      p+rightBudget a u i rb.1.val y.val (creationIndex p (jetWeight a u i m)) rb.2.val m := by
  have H := actual_right_den_budget a u i _ t m rb y ht hi hu
  exact Nat.add_le_add_left (extra_le_bound _ _ H.1 H.2) p

/-- Explicit integer-cast payloads supplement, rather than replace, primitive events. -/
structure WithInputs (R : Type*) where
  outcome : Logged R
  inputs : List ℤ

def emitInteger {R : Type*} [CommRing R] (n : ℤ) : WithInputs R :=
  ⟨emitCast n,[n]⟩

def evaluateWithInputs {R : Type*} [CommRing R] (h z : ℕ) :
    Expr → WithInputs (Stored R h z)
  | .literal v =>
      let c := emitInteger (R:=R) v
      let A := materializeLogged
        (fun b n => (⟨sample h z (C (C c.outcome.value)) b n,[]⟩ : Logged R))
      ⟨⟨A.value,c.outcome.events++A.events⟩,c.inputs⟩
  | .inner => ⟨⟨materialize (sample h z (C X)),[]⟩,[]⟩
  | .outer => ⟨⟨materialize (sample h z X),[]⟩,[]⟩
  | .add a b =>
      let A := evaluateWithInputs h z a
      let B := evaluateWithInputs h z b
      let C := materializeLogged
        (fun r n => emitAdd (view A.outcome.value r n) (view B.outcome.value r n))
      ⟨⟨C.value,A.outcome.events++B.outcome.events++C.events⟩,A.inputs++B.inputs⟩
  | .neg a =>
      let A := evaluateWithInputs h z a
      let B := materializeLogged (fun r n => emitNegate (view A.outcome.value r n))
      ⟨⟨B.value,A.outcome.events++B.events⟩,A.inputs⟩
  | .mul a b =>
      let A := evaluateWithInputs h z a
      let B := evaluateWithInputs h z b
      let C := multiplyLogged A.outcome.value B.outcome.value
      ⟨⟨C.value,A.outcome.events++B.outcome.events++C.events⟩,A.inputs++B.inputs⟩

theorem evaluateWithInputs_erasure {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    (evaluateWithInputs (R:=R) h z e).outcome=evaluateLogged h z e := by
  induction e with
  | literal v => rfl
  | inner => rfl
  | outer => rfl
  | add a b ha hb => simp only [evaluateWithInputs,ha,hb,evaluateLogged]
  | neg a ha => simp only [evaluateWithInputs,ha,evaluateLogged]
  | mul a b ha hb => simp only [evaluateWithInputs,ha,hb,evaluateLogged]

theorem evaluateWithInputs_inputs {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    (evaluateWithInputs (R:=R) h z e).inputs=literals e := by
  induction e with
  | literal v => rfl
  | inner => rfl
  | outer => rfl
  | add a b ha hb => simp only [evaluateWithInputs,literals,ha,hb]
  | neg a ha => simp only [evaluateWithInputs,literals,ha]
  | mul a b ha hb => simp only [evaluateWithInputs,literals,ha,hb]

theorem evaluateWithInputs_value {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    view (evaluateWithInputs (R:=R) h z e).outcome.value=
      sample h z (fullEval e : (R[X])[X]) := by
  rw [evaluateWithInputs_erasure]
  exact evaluateLogged_value h z e

theorem evaluateWithInputs_casts {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    (evaluateWithInputs (R:=R) h z e).inputs.length=
      (evaluateWithInputs (R:=R) h z e).outcome.events.count .cast := by
  rw [evaluateWithInputs_inputs,evaluateWithInputs_erasure,literals_length,evaluateLogged_cast]

theorem evaluateWithInputs_volume {R : Type*} [CommRing R] (h z : ℕ) (e : Expr) :
    (((evaluateWithInputs (R:=R) h z e).inputs).map signedWidth).sum=inputVolume e := by
  rw [evaluateWithInputs_inputs]
  rfl

theorem bounded_recorded_volume {R : Type*} [CommRing R] (h z B : ℕ) (e : Expr)
    (he : LiteralBound B e) :
    (((evaluateWithInputs (R:=R) h z e).inputs).map signedWidth).sum≤nodes e*(B+2) := by
  rw [evaluateWithInputs_volume]
  exact he.volume_nodes

theorem actual_recorded_volume {R : Type*} [CommRing R]
    (a u i x y k h z : ℕ) (t : ℤ) (m : ℕ → ℕ)
    (ht : (t:ℚ)∈OrientationAnchor.grid a) (hi : i≤a) (hu : u≤a) (hy : y<2^a) :
    (((evaluateWithInputs (R:=R) h z (packedApprox a u i x y k t m).expr).inputs).map signedWidth).sum≤
      approximationNodeBound a u k m*(approximationBudget a u i x y k m+2) := by
  rw [evaluateWithInputs_volume]
  exact actual_inputVolume_bound a u i x y k t m ht hi hu hy

end OrientationPayloadTrace

/-- Reads concern an already available magnitude bit list; construction is separate. -/
structure Run (R : Type*) where
  value : R
  events : List Event
  reads : List Bool

def foldBits {R : Type*} [CommRing R] : List Bool → Run R
  | [] => ⟨0,[],[]⟩
  | b::rest =>
      let t := foldBits rest
      let d := emitAdd t.value t.value
      let s := emitAdd d.value (if b then 1 else 0)
      ⟨s.value,t.events++d.events++s.events,t.reads++[b]⟩

theorem foldBits_reads {R : Type*} [CommRing R] (bs : List Bool) :
    (foldBits (R:=R) bs).reads=bs.reverse := by
  induction bs with
  | nil => rfl
  | cons b bs ih => simp only [foldBits,ih,List.reverse_cons]

theorem foldBits_adds {R : Type*} [CommRing R] (bs : List Bool) :
    (foldBits (R:=R) bs).events.count .add=2*bs.length := by
  induction bs with
  | nil => rfl
  | cons b bs ih => simp [foldBits,emitAdd,ih]; omega

theorem foldBits_negates {R : Type*} [CommRing R] (bs : List Bool) :
    (foldBits (R:=R) bs).events.count .negate=0 := by
  induction bs with
  | nil => rfl
  | cons b bs ih => simp [foldBits,emitAdd,ih]

theorem foldBits_correct {R : Type*} [CommRing R] (n : ℕ) :
    (foldBits (R:=R) n.bits).value=(n:R) := by
  induction n using Nat.binaryRec' with
  | zero => simp only [Nat.zero_bits,foldBits,Nat.cast_zero]
  | bit b n h ih =>
    rw [Nat.bits_append_bit n b h]
    simp only [foldBits,emitAdd,ih,Nat.bit_val,Nat.cast_add,Nat.cast_mul,Nat.cast_ofNat]
    cases b <;> simp <;> ring

def signed {R : Type*} [CommRing R] : ℤ → Run R
  | .ofNat n => foldBits n.bits
  | .negSucc n =>
      let A := foldBits (n+1).bits
      let B := emitNegate A.value
      ⟨B.value,A.events++B.events,A.reads⟩

theorem signed_correct {R : Type*} [CommRing R] (n : ℤ) :
    (signed (R:=R) n).value=(n:R) := by
  cases n with
  | ofNat n => simpa only [signed,Int.ofNat_eq_natCast,Int.cast_natCast] using (foldBits_correct (R:=R) n)
  | negSucc n => simp [signed,emitNegate,foldBits_correct,Int.negSucc_eq]

theorem signed_reads_length {R : Type*} [CommRing R] (n : ℤ) :
    (signed (R:=R) n).reads.length=Nat.size n.natAbs := by
  cases n <;> simp [signed,foldBits_reads,Nat.size_eq_bits_len]

theorem signed_adds {R : Type*} [CommRing R] (n : ℤ) :
    (signed (R:=R) n).events.count .add=2*Nat.size n.natAbs := by
  cases n <;> simp [signed,emitNegate,foldBits_adds,Nat.size_eq_bits_len]

theorem signed_negates {R : Type*} [CommRing R] (n : ℤ) :
    (signed (R:=R) n).events.count .negate≤1 := by
  cases n <;> simp [signed,emitNegate,foldBits_negates]

theorem bounded_signed_reads {R : Type*} [CommRing R] (B : ℕ) (n : ℤ)
    (hn : n.natAbs≤2^B) :
    (signed (R:=R) n).reads.length≤B+1 := by
  rw [signed_reads_length]
  have H := Nat.size_le_size hn
  simpa only [Nat.size_pow] using H

theorem bounded_signed_adds {R : Type*} [CommRing R] (B : ℕ) (n : ℤ)
    (hn : n.natAbs≤2^B) :
    (signed (R:=R) n).events.count .add≤2*(B+1) := by
  rw [signed_adds]
  exact Nat.mul_le_mul_left 2 (by
    have H := Nat.size_le_size hn
    simpa only [Nat.size_pow] using H)

end OrientationBinaryInput

namespace OrientationBooleanAdder
inductive Gate where
  | bxor
  | band
  | bor
  deriving DecidableEq

structure Gated (A : Type*) where
  value : A
  events : List Gate

def emitXor (a b : Bool) : Gated Bool := ⟨a ^^ b,[.bxor]⟩
def emitAnd (a b : Bool) : Gated Bool := ⟨a && b,[.band]⟩
def emitOr (a b : Bool) : Gated Bool := ⟨a || b,[.bor]⟩

def fullAdder (a b c : Bool) : Gated (Bool × Bool) :=
  let p := emitXor a b
  let s := emitXor p.value c
  let g := emitAnd a b
  let h := emitAnd p.value c
  let k := emitOr g.value h.value
  ⟨(s.value,k.value),p.events++s.events++g.events++h.events++k.events⟩

theorem fullAdder_identity (a b c : Bool) :
    a.toNat+b.toNat+c.toNat=
      (fullAdder a b c).value.1.toNat+2*(fullAdder a b c).value.2.toNat := by
  cases a <;> cases b <;> cases c <;> decide

def gateMultiplicity : Gate → ℕ
  | .bxor => 2
  | .band => 2
  | .bor => 1

theorem fullAdder_count (a b c : Bool) (g : Gate) :
    (fullAdder a b c).events.count g=gateMultiplicity g := by
  cases g <;> rfl

theorem fullAdder_events_length (a b c : Bool) :
    (fullAdder a b c).events.length=5 := by
  simp [fullAdder,emitXor,emitAnd,emitOr]

def decode : List Bool → ℕ
  | [] => 0
  | b::bs => b.toNat+2*decode bs

theorem decode_bound (bs : List Bool) : decode bs<2^bs.length := by
  induction bs with
  | nil => decide
  | cons b bs ih =>
    cases b <;> simp only [decode,Bool.toNat_false,Bool.toNat_true,List.length_cons,pow_succ]
    all_goals omega

/-- Only equal-length inputs are admitted by the arithmetic/count theorems. -/
def ripple : Bool → List Bool → List Bool → Gated (List Bool × Bool)
  | c,[],[] => ⟨([],c),[]⟩
  | c,a::as,b::bs =>
      let f := fullAdder a b c
      let t := ripple f.value.2 as bs
      ⟨(f.value.1::t.value.1,t.value.2),f.events++t.events⟩
  | c,_,_ => ⟨([],c),[]⟩

theorem ripple_length (c : Bool) (xs ys : List Bool) (h : xs.length=ys.length) :
    (ripple c xs ys).value.1.length=xs.length := by
  induction xs generalizing ys c with
  | nil =>
    cases ys with
    | nil => rfl
    | cons b bs => simp at h
  | cons a as ih =>
    cases ys with
    | nil => simp at h
    | cons b bs =>
      have hab : as.length=bs.length := Nat.succ.inj h
      have I := ih (c := (fullAdder a b c).value.2) (ys := bs) hab
      simpa only [ripple,List.length_cons] using congrArg Nat.succ I

theorem ripple_identity (c : Bool) (xs ys : List Bool) (h : xs.length=ys.length) :
    decode (ripple c xs ys).value.1+2^xs.length*(ripple c xs ys).value.2.toNat=
      decode xs+decode ys+c.toNat := by
  induction xs generalizing ys c with
  | nil =>
    cases ys with
    | nil => simp [ripple,decode]
    | cons b bs => simp at h
  | cons a as ih =>
    cases ys with
    | nil => simp at h
    | cons b bs =>
      have hab : as.length=bs.length := Nat.succ.inj h
      have I := ih (c := (fullAdder a b c).value.2) (ys := bs) hab
      have F := fullAdder_identity a b c
      simp only [ripple,decode,List.length_cons,pow_succ]
      nlinarith [I,F]

theorem ripple_count (g : Gate) (c : Bool) (xs ys : List Bool) (h : xs.length=ys.length) :
    (ripple c xs ys).events.count g=gateMultiplicity g*xs.length := by
  induction xs generalizing ys c with
  | nil =>
    cases ys with
    | nil => simp [ripple]
    | cons b bs => simp at h
  | cons a as ih =>
    cases ys with
    | nil => simp at h
    | cons b bs =>
      have hab : as.length=bs.length := Nat.succ.inj h
      have I := ih (c := (fullAdder a b c).value.2) (ys := bs) hab
      simp only [ripple,List.count_append,fullAdder_count,I,List.length_cons,Nat.mul_succ]
      omega

theorem ripple_events_length (c : Bool) (xs ys : List Bool) (h : xs.length=ys.length) :
    (ripple c xs ys).events.length=5*xs.length := by
  induction xs generalizing ys c with
  | nil =>
    cases ys with
    | nil => rfl
    | cons b bs => simp at h
  | cons a as ih =>
    cases ys with
    | nil => simp at h
    | cons b bs =>
      have hab : as.length=bs.length := Nat.succ.inj h
      have I := ih (c := (fullAdder a b c).value.2) (ys := bs) hab
      simp only [ripple,List.length_append,fullAdder_events_length,I,List.length_cons]
      omega

theorem ripple_residue (c : Bool) (xs ys : List Bool) (h : xs.length=ys.length) :
    decode (ripple c xs ys).value.1=
      (decode xs+decode ys+c.toNat)%(2^xs.length) := by
  have H := ripple_identity c xs ys h
  have B := decode_bound (ripple c xs ys).value.1
  rw [ripple_length c xs ys h] at B
  calc
    decode (ripple c xs ys).value.1=
        (decode (ripple c xs ys).value.1+2^xs.length*(ripple c xs ys).value.2.toNat)%(2^xs.length) := by
      simp [Nat.add_mod,Nat.mod_eq_of_lt B]
    _ = (decode xs+decode ys+c.toNat)%(2^xs.length) := congrArg (fun n => n%(2^xs.length)) H

end OrientationBooleanAdder

def encode : ℕ → ℕ → List Bool
  | 0,_ => []
  | q+1,n => n.bodd::encode q (n/2)

theorem encode_length (q n : ℕ) : (encode q n).length=q := by
  induction q generalizing n with
  | zero => rfl
  | succ q ih => simp only [encode,List.length_cons,ih]

theorem encode_quotient (q n : ℕ) :
    decode (encode q n)+2^q*(n/(2^q))=n := by
  induction q generalizing n with
  | zero => simp [encode,decode]
  | succ q ih =>
    have I := ih (n/2)
    have D : n/2/(2^q)=n/(2^(q+1)) := by
      rw [Nat.div_div_eq_div_mul,pow_succ,Nat.mul_comm 2 (2^q)]
    rw [D] at I
    rw [pow_succ] at I
    have T := Nat.mod_add_div n 2
    simp only [encode,decode]
    rw [← Nat.mod_two_of_bodd]
    rw [pow_succ]
    nlinarith [I,T]

theorem encode_mod (q n : ℕ) : decode (encode q n)=n%(2^q) := by
  have H := encode_quotient q n
  have B := decode_bound (encode q n)
  rw [encode_length] at B
  calc
    decode (encode q n)=(decode (encode q n)+2^q*(n/(2^q)))%(2^q) := by
      simp [Nat.add_mod,Nat.mod_eq_of_lt B]
    _ = n%(2^q) := congrArg (fun v => v%(2^q)) H

abbrev Word (q : ℕ) := {bs : List Bool // bs.length=q}

def ofNatWord (q n : ℕ) : Word q := ⟨encode q n,encode_length q n⟩

def wordValue {q : ℕ} (w : Word q) : ZMod (2^q) := (decode w.val : ZMod (2^q))

theorem ofNatWord_value (q n : ℕ) : wordValue (ofNatWord q n)=(n:ZMod (2^q)) := by
  unfold wordValue ofNatWord
  rw [encode_mod,ZMod.natCast_mod]

def addWord {q : ℕ} (x y : Word q) : Gated (Word q) :=
  let f := ripple false x.val y.val
  ⟨⟨f.value.1,(ripple_length false x.val y.val (x.property.trans y.property.symm)).trans x.property⟩,
    f.events⟩

theorem addWord_value {q : ℕ} (x y : Word q) :
    wordValue (addWord x y).value=wordValue x+wordValue y := by
  have H := ripple_residue false x.val y.val (x.property.trans y.property.symm)
  simp only [x.property,Bool.toNat_false,Nat.add_zero] at H
  unfold addWord wordValue
  change ((decode (ripple false x.val y.val).value.1 : ℕ) : ZMod (2^q))=
    (decode x.val : ZMod (2^q))+(decode y.val : ZMod (2^q))
  rw [H,ZMod.natCast_mod,Nat.cast_add]

theorem addWord_count {q : ℕ} (x y : Word q) (g : Gate) :
    (addWord x y).events.count g=gateMultiplicity g*q := by
  simpa only [addWord,x.property] using
    ripple_count g false x.val y.val (x.property.trans y.property.symm)

theorem addWord_events_length {q : ℕ} (x y : Word q) :
    (addWord x y).events.length=5*q := by
  simpa only [addWord,x.property] using
    ripple_events_length false x.val y.val (x.property.trans y.property.symm)

theorem addWord_ofNat_value (q a b : ℕ) :
    wordValue (addWord (ofNatWord q a) (ofNatWord q b)).value=
      ((a+b:ℕ):ZMod (2^q)) := by
  rw [addWord_value,ofNatWord_value,ofNatWord_value,Nat.cast_add]

end OrientationFixedWord

def invertBits : List Bool → Gated (List Bool)

theorem invertBits_length (bs : List Bool) : (invertBits bs).value.length=bs.length := by

theorem invertBits_sum (bs : List Bool) :

    cases b with
    | false =>
      change (1+2*decode (invertBits bs).value)+(0+2*decode bs)+1=2^(bs.length+1)
      rw [pow_succ]
      omega
    | true =>
      change (0+2*decode (invertBits bs).value)+(1+2*decode bs)+1=2^(bs.length+1)
      rw [pow_succ]
      omega

theorem invertBits_count (bs : List Bool) (g : Gate) :
    (invertBits bs).events.count g=if g=.bxor then bs.length else 0 := by
  induction bs with
  | nil => cases g <;> rfl
  | cons b bs ih => cases g <;> simp [invertBits,emitXor,ih]

theorem invertBits_events_length (bs : List Bool) :
    (invertBits bs).events.length=bs.length := by
  induction bs with
  | nil => rfl
  | cons b bs ih => simp [invertBits,emitXor,ih]

def invertWord {q : ℕ} (w : Word q) : Gated (Word q) :=
  let f := invertBits w.val
  ⟨⟨f.value,(invertBits_length w.val).trans w.property⟩,f.events⟩

theorem invertWord_value {q : ℕ} (w : Word q) :
    wordValue (invertWord w).value= -wordValue w-1 := by
  have H := congrArg (fun n : ℕ => (n:ZMod (2^q))) (invertBits_sum w.val)
  simp only [w.property,Nat.cast_add,Nat.cast_one,ZMod.natCast_self] at H
  unfold invertWord wordValue
  change (decode (invertBits w.val).value : ZMod (2^q))= -(decode w.val : ZMod (2^q))-1
  linear_combination H

theorem invertWord_events_length {q : ℕ} (w : Word q) :
    (invertWord w).events.length=q := by
  simpa only [invertWord,w.property] using invertBits_events_length w.val

def negateWord {q : ℕ} (w : Word q) : Gated (Word q) :=
  let A := invertWord w
  let B := addWord A.value (ofNatWord q 1)
  ⟨B.value,A.events++B.events⟩

theorem negateWord_value {q : ℕ} (w : Word q) :
    wordValue (negateWord w).value= -wordValue w := by
  unfold negateWord
  rw [addWord_value,invertWord_value,ofNatWord_value]
  simp only [Nat.cast_one]
  ring

theorem negateWord_events_length {q : ℕ} (w : Word q) :
    (negateWord w).events.length=6*q := by
  simp only [negateWord,List.length_append,invertWord_events_length,addWord_events_length]
  omega

def ofIntWord (q : ℕ) : ℤ → Gated (Word q)
  | .ofNat n => ⟨ofNatWord q n,[]⟩
  | .negSucc n => negateWord (ofNatWord q (n+1))

theorem ofIntWord_value (q : ℕ) (n : ℤ) :
    wordValue (ofIntWord q n).value=(n:ZMod (2^q)) := by
  cases n with
  | ofNat n =>
    simpa only [ofIntWord,Int.ofNat_eq_natCast,Int.cast_natCast] using ofNatWord_value q n
  | negSucc n => simp [ofIntWord,negateWord_value,ofNatWord_value,Int.negSucc_eq]

theorem ofIntWord_events_bound (q : ℕ) (n : ℤ) :
    (ofIntWord q n).events.length≤6*q := by
  cases n <;> simp [ofIntWord,negateWord_events_length]

end OrientationSignedWord

namespace OrientationWordMultiply

theorem decode_append (xs ys : List Bool) :
    decode (xs++ys)=decode xs+2^xs.length*decode ys := by
  induction xs with
  | nil => simp [decode]
  | cons b bs ih =>
    simp only [List.cons_append,decode,List.length_cons,pow_succ,ih]
    ring

def shiftWord : {q : ℕ} → Word q → Word q
  | 0,_ => ⟨[],rfl⟩
  | q+1,w => ⟨false::w.val.take q,by
      simp only [List.length_cons,List.length_take,w.property,
        Nat.min_eq_left (Nat.le_succ q)]⟩

theorem shiftWord_value {q : ℕ} (w : Word q) :
    wordValue (shiftWord w)=2*wordValue w := by
  cases q with
  | zero =>
    exact @Subsingleton.elim (ZMod (2^0))
      (ZMod.subsingleton_iff.mpr (by decide)) _ _
  | succ q =>
    have P : (w.val.take q).length=q := by
      rw [List.length_take,w.property]
      exact Nat.min_eq_left (Nat.le_succ q)
    have E := decode_append (w.val.take q) (w.val.drop q)
    rw [List.take_append_drop,P] at E
    have H : 2*decode w.val=
        decode (shiftWord w).val+2^(q+1)*decode (w.val.drop q) := by
      simp only [shiftWord,decode,Bool.toNat_false,Nat.zero_add]
      rw [pow_succ]
      nlinarith [E]
    have HC := congrArg (fun n : ℕ => (n:ZMod (2^(q+1)))) H
    simp only [Nat.cast_mul,Nat.cast_ofNat,Nat.cast_add,ZMod.natCast_self,
      zero_mul,add_zero] at HC
    simpa only [wordValue] using HC.symm

def multiplyBits {q : ℕ} (w : Word q) : List Bool → Gated (Word q)
  | [] => ⟨ofNatWord q 0,[]⟩
  | b::bs =>
      let t := multiplyBits (shiftWord w) bs
      let selected := if b then w else ofNatWord q 0
      let s := addWord selected t.value
      ⟨s.value,t.events++s.events⟩

theorem multiplyBits_value {q : ℕ} (w : Word q) (bs : List Bool) :
    wordValue (multiplyBits w bs).value=wordValue w*(decode bs:ZMod (2^q)) := by
  induction bs generalizing w with
  | nil => simp only [multiplyBits,ofNatWord_value,decode,Nat.cast_zero,mul_zero]
  | cons b bs ih =>
    simp only [multiplyBits,addWord_value,ih,shiftWord_value,decode,
      Nat.cast_add,Nat.cast_mul,Nat.cast_ofNat]
    cases b <;> simp [ofNatWord_value] <;> ring

theorem multiplyBits_events_length {q : ℕ} (w : Word q) (bs : List Bool) :
    (multiplyBits w bs).events.length=5*q*bs.length := by
  induction bs generalizing w with
  | nil => rfl
  | cons b bs ih =>
    simp only [multiplyBits,List.length_append,addWord_events_length,ih,
      List.length_cons,Nat.mul_succ]

def multiplyWord {q : ℕ} (x y : Word q) : Gated (Word q) := multiplyBits x y.val

theorem multiplyWord_value {q : ℕ} (x y : Word q) :
    wordValue (multiplyWord x y).value=wordValue x*wordValue y := by
  simpa only [multiplyWord,wordValue] using multiplyBits_value x y.val

theorem multiplyWord_events_length {q : ℕ} (x y : Word q) :
    (multiplyWord x y).events.length=5*q^2 := by
  simpa only [multiplyWord,y.property,pow_two,Nat.mul_assoc] using
    multiplyBits_events_length x y.val

end OrientationWordMultiply

def wordView {q h z : ℕ} (A : Stored (Word q) h z) : Box (ZMod (2^q)) h z :=
  fun b n => wordValue (view A b n)

def wordExtend {q h z : ℕ} (A : Stored (Word q) h z) (b n : ℕ) : Word q :=
  if hb : b<h then if hn : n<z then view A ⟨b,hb⟩ ⟨n,hn⟩ else ofNatWord q 0 else ofNatWord q 0

theorem wordExtend_value {q h z : ℕ} (A : Stored (Word q) h z) (b n : ℕ) :
    wordValue (wordExtend A b n)=extend (wordView A) b n := by
  by_cases hb : b<h <;> by_cases hn : n<z <;>
    simp [wordExtend,extend,wordView,hb,hn,ofNatWord_value]

def dotWord {q : ℕ} : List (Word q × Word q) → Gated (Word q)
  | [] => ⟨ofNatWord q 0,[]⟩
  | xy::rest =>
      let p := multiplyWord xy.1 xy.2
      let t := dotWord rest
      let s := addWord p.value t.value
      ⟨s.value,p.events++t.events++s.events⟩

theorem dotWord_value {q : ℕ} (xs : List (Word q × Word q)) :
    wordValue (dotWord xs).value=(xs.map (fun xy => wordValue xy.1*wordValue xy.2)).sum := by
  induction xs with
  | nil => simp only [dotWord,ofNatWord_value,Nat.cast_zero,List.map_nil,List.sum_nil]
  | cons xy xs ih =>
    simp only [dotWord,addWord_value,multiplyWord_value,ih,List.map_cons,List.sum_cons]

theorem dotWord_events_length {q : ℕ} (xs : List (Word q × Word q)) :
    (dotWord xs).events.length=(5*q^2+5*q)*xs.length := by
  induction xs with
  | nil => rfl
  | cons xy xs ih =>
    simp only [dotWord,List.length_append,multiplyWord_events_length,
      addWord_events_length,ih,List.length_cons]
    ring

def wordPairs {q h z : ℕ} (A B : Stored (Word q) h z) (b n : ℕ) : List (Word q × Word q) :=
  ((antidiagonal b).toList).flatMap (fun ij => ((antidiagonal n).toList).map
    (fun rs => (wordExtend A ij.1 rs.1,wordExtend B ij.2 rs.2)))

theorem wordPairs_length {q h z : ℕ} (A B : Stored (Word q) h z) (b n : ℕ) :
    (wordPairs A B b n).length=(b+1)*(n+1) := by
  simp only [wordPairs,List.length_flatMap,List.length_map,Finset.length_toList,
    Finset.Nat.card_antidiagonal]
  rw [nat_sum_constant]
  simp only [Finset.length_toList,Finset.Nat.card_antidiagonal]

def wordConvolution {q h z : ℕ} (A B : Stored (Word q) h z) (b : Fin h) (n : Fin z) :
    Gated (Word q) := dotWord (wordPairs A B b.val n.val)

theorem wordConvolution_value {q h z : ℕ}
    (A B : Stored (Word q) h z) (b : Fin h) (n : Fin z) :
    wordValue (wordConvolution A B b n).value=
      OrientationCoefficientBox.mul (wordView A) (wordView B) b n := by
  rw [wordConvolution,dotWord_value,wordPairs,map_flatMap_sum]
  simp only [List.map_map,Function.comp_apply,Finset.sum_map_toList,wordExtend_value,
    OrientationCoefficientBox.mul]

theorem wordConvolution_events_length {q h z : ℕ}
    (A B : Stored (Word q) h z) (b : Fin h) (n : Fin z) :
    (wordConvolution A B b n).events.length=(5*q^2+5*q)*((b.val+1)*(n.val+1)) := by
  rw [wordConvolution,dotWord_events_length,wordPairs_length]

end OrientationWordConvolution

def materializeGated {A : Type*} {h z : ℕ} (f : Fin h → Fin z → Gated A) :
    Gated (Stored A h z) :=
  let cells : Stored (Gated A) h z := materialize f
  ⟨cells.map (fun row => row.map (fun cell => cell.value)),
    cells.toList.flatMap (fun row => row.toList.flatMap (fun cell => cell.events))⟩

theorem materializeGated_view {A : Type*} {h z : ℕ} (f : Fin h → Fin z → Gated A) :
    view (materializeGated f).value=(fun b n => (f b n).value) := by
  funext b n
  simp only [materializeGated,materialize,view,Vector.getElem_map,Vector.getElem_ofFn]

theorem materializeGated_events_length {A : Type*} {h z : ℕ}
    (f : Fin h → Fin z → Gated A) :
    (materializeGated f).events.length=∑ b : Fin h,∑ n : Fin z,(f b n).events.length := by
  simp only [materializeGated,materialize,Vector.toList_ofFn,List.length_flatMap]
  simp only [Vector.toList_ofFn,List.map_ofFn,Function.comp_def,Fin.sum_ofFn]

def addWords {q h z : ℕ} (A B : Stored (Word q) h z) : Gated (Stored (Word q) h z) :=
  materializeGated (fun b n => addWord (view A b n) (view B b n))

def negateWords {q h z : ℕ} (A : Stored (Word q) h z) : Gated (Stored (Word q) h z) :=
  materializeGated (fun b n => negateWord (view A b n))

def multiplyWords {q h z : ℕ} (A B : Stored (Word q) h z) : Gated (Stored (Word q) h z) :=
  materializeGated (fun b n => wordConvolution A B b n)

theorem addWords_value {q h z : ℕ} (A B : Stored (Word q) h z) :
    wordView (addWords A B).value=OrientationCoefficientBox.add (wordView A) (wordView B) := by
  unfold addWords wordView OrientationCoefficientBox.add
  rw [materializeGated_view]
  funext b n
  exact addWord_value (view A b n) (view B b n)

theorem negateWords_value {q h z : ℕ} (A : Stored (Word q) h z) :
    wordView (negateWords A).value=OrientationCoefficientBox.neg (wordView A) := by
  unfold negateWords wordView OrientationCoefficientBox.neg
  rw [materializeGated_view]
  funext b n
  exact negateWord_value (view A b n)

theorem multiplyWords_value {q h z : ℕ} (A B : Stored (Word q) h z) :
    wordView (multiplyWords A B).value=OrientationCoefficientBox.mul (wordView A) (wordView B) := by
  unfold multiplyWords wordView
  rw [materializeGated_view]
  funext b n
  exact wordConvolution_value A B b n

theorem addWords_events_length {q h z : ℕ} (A B : Stored (Word q) h z) :
    (addWords A B).events.length=5*q*h*z := by
  simp only [addWords,materializeGated_events_length,addWord_events_length]
  simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,Nat.cast_id]
  ring

theorem negateWords_events_length {q h z : ℕ} (A : Stored (Word q) h z) :
    (negateWords A).events.length=6*q*h*z := by
  simp only [negateWords,materializeGated_events_length,negateWord_events_length]
  simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,Nat.cast_id]
  ring

theorem multiplyWords_events_length {q h z : ℕ} (A B : Stored (Word q) h z) :
    (multiplyWords A B).events.length=(5*q^2+5*q)*OrientationCoefficientBox.multiplicationProducts h z := by
  simp only [multiplyWords,materializeGated_events_length,wordConvolution_events_length]
  rw [← fin_products_sum]
  simp only [Finset.mul_sum]

theorem multiplyWords_events_bound {q h z : ℕ} (A B : Stored (Word q) h z) :
    (multiplyWords A B).events.length≤(5*q^2+5*q)*(h*z)^2 := by
  rw [multiplyWords_events_length]
  exact Nat.mul_le_mul_left _ (OrientationCoefficientBox.multiplicationProducts_bound h z)

def basisWordArray {q : ℕ} (h z r s : ℕ) (v : Word q) : Stored (Word q) h z :=
  materialize (fun b n => if b.val=r ∧ n.val=s then v else ofNatWord q 0)

theorem basisWordArray_value {q : ℕ} (h z r s : ℕ) (v : Word q) :
    wordView (basisWordArray h z r s v)=
      (fun b n => if b.val=r ∧ n.val=s then wordValue v else 0) := by
  funext b n
  simp only [wordView,basisWordArray,view_materialize]
  split_ifs <;> simp [ofNatWord_value]

theorem basisWordArray_literal {q : ℕ} (h z : ℕ) (v : Word q) :
    wordView (basisWordArray h z 0 0 v)=sample h z (C (C (wordValue v))) := by
  rw [basisWordArray_value]
  funext b n
  by_cases hb : b.val=0 <;> by_cases hn : n.val=0 <;>
    simp [sample,Polynomial.coeff_C,hb,hn]

theorem basisWordArray_inner (q h z : ℕ) :
    wordView (basisWordArray h z 0 1 (ofNatWord q 1))=
      sample h z (C X : ((ZMod (2^q))[X])[X]) := by
  rw [basisWordArray_value,ofNatWord_value]
  funext b n
  by_cases hb : b.val=0 <;> by_cases hn : n.val=1 <;>
    simp [sample,Polynomial.coeff_C,Polynomial.coeff_X,eq_comm,hb,hn]

theorem basisWordArray_outer (q h z : ℕ) :
    wordView (basisWordArray h z 1 0 (ofNatWord q 1))=
      sample h z (X : ((ZMod (2^q))[X])[X]) := by
  rw [basisWordArray_value,ofNatWord_value]
  funext b n
  by_cases hb : b.val=1 <;> by_cases hn : n.val=0 <;>
    simp [sample,Polynomial.coeff_X,Polynomial.coeff_one,eq_comm,hb,hn]

def evaluateWord (q h z : ℕ) : Expr → Gated (Stored (Word q) h z)
  | .literal v =>
      let c := ofIntWord q v
      ⟨basisWordArray h z 0 0 c.value,c.events⟩
  | .inner => ⟨basisWordArray h z 0 1 (ofNatWord q 1),[]⟩
  | .outer => ⟨basisWordArray h z 1 0 (ofNatWord q 1),[]⟩
  | .add a b =>
      let A := evaluateWord q h z a
      let B := evaluateWord q h z b
      let C := addWords A.value B.value
      ⟨C.value,A.events++B.events++C.events⟩
  | .neg a =>
      let A := evaluateWord q h z a
      let B := negateWords A.value
      ⟨B.value,A.events++B.events⟩
  | .mul a b =>
      let A := evaluateWord q h z a
      let B := evaluateWord q h z b
      let C := multiplyWords A.value B.value
      ⟨C.value,A.events++B.events++C.events⟩

theorem evaluateWord_value (q h z : ℕ) (e : Expr) :
    wordView (evaluateWord q h z e).value=sample h z (fullEval e : ((ZMod (2^q))[X])[X]) := by
  induction e with
  | literal v => simp only [evaluateWord,basisWordArray_literal,ofIntWord_value,fullEval]
  | inner => exact basisWordArray_inner q h z
  | outer => exact basisWordArray_outer q h z
  | add a b ha hb => simp only [evaluateWord,addWords_value,ha,hb,fullEval,sample_add]
  | neg a ha => simp only [evaluateWord,negateWords_value,ha,fullEval,sample_neg]
  | mul a b ha hb => simp only [evaluateWord,multiplyWords_value,ha,hb,fullEval,sample_mul]

theorem evaluateWord_value_eq (q h z : ℕ) (e : Expr) :
    wordView (evaluateWord q h z e).value=
      view (OrientationStoredBox.evaluate (R:=ZMod (2^q)) h z e) :=
  (evaluateWord_value q h z e).trans (OrientationStoredBox.evaluate_correct h z e).symm

def coefficientWord (q a u i x y k b n : ℕ) (t : ℤ) (m : ℕ → ℕ) : Gated (Word q) :=
  let A := evaluateWord q (b+1) (n+1) (packedApprox a u i x y k t m).expr
  ⟨view A.value ⟨b,Nat.lt_succ_self b⟩ ⟨n,Nat.lt_succ_self n⟩,A.events⟩

theorem coefficientWord_value (q a u i x y k b n : ℕ) (t : ℤ) (m : ℕ → ℕ) :
    wordValue (coefficientWord q a u i x y k b n t m).value=
      OrientationStoredBox.coefficient q a u i x y k b n t m := by
  unfold coefficientWord OrientationStoredBox.coefficient
  change wordView (evaluateWord q (b+1) (n+1) (packedApprox a u i x y k t m).expr).value
    ⟨b,Nat.lt_succ_self b⟩ ⟨n,Nat.lt_succ_self n⟩=_
  rw [evaluateWord_value_eq]

end OrientationWordEvaluator

namespace OrientationWordGateCost
open OrientationCoefficientBox OrientationScalarCost OrientationScalarCode OrientationBooleanAdder
  OrientationFixedWord OrientationSignedWord OrientationWordArray OrientationWordEvaluator

def wordGateUnit (q : ℕ) : ℕ := 10*(q+1)^2

def wordGateBudget (q h z : ℕ) : ℕ := wordGateUnit q*(h*z+1)^2

theorem unit_scalar (q : ℕ) : 6*q≤wordGateUnit q := by
  unfold wordGateUnit
  nlinarith

theorem unit_tensor (q : ℕ) : 5*q^2+5*q≤wordGateUnit q := by
  unfold wordGateUnit
  nlinarith

theorem shape_one (h z : ℕ) : 1≤(h*z+1)^2 := by
  have H : 0<(h*z+1)^2 := by positivity
  omega

theorem shape_linear (h z : ℕ) : h*z≤(h*z+1)^2 := by
  nlinarith [Nat.zero_le (h*z),Nat.zero_le ((h*z)^2)]

theorem shape_square (h z : ℕ) : (h*z)^2≤(h*z+1)^2 := by
  nlinarith [Nat.zero_le (h*z)]

theorem scalar_gate_cap (q h z : ℕ) : 6*q≤wordGateBudget q h z := by
  simpa only [wordGateBudget,Nat.mul_one] using
    Nat.mul_le_mul (unit_scalar q) (shape_one h z)

theorem addWords_gate_cap {q h z : ℕ}
    (A B : OrientationStoredBox.Stored (Word q) h z) :
    (addWords A B).events.length≤wordGateBudget q h z := by
  rw [addWords_events_length]
  have Hq : 5*q≤wordGateUnit q := by have H := unit_scalar q; omega
  simpa only [wordGateBudget,Nat.mul_assoc] using
    Nat.mul_le_mul Hq (shape_linear h z)

theorem negateWords_gate_cap {q h z : ℕ}
    (A : OrientationStoredBox.Stored (Word q) h z) :
    (negateWords A).events.length≤wordGateBudget q h z := by
  rw [negateWords_events_length]
  simpa only [wordGateBudget,Nat.mul_assoc] using
    Nat.mul_le_mul (unit_scalar q) (shape_linear h z)

theorem multiplyWords_gate_cap {q h z : ℕ}
    (A B : OrientationStoredBox.Stored (Word q) h z) :
    (multiplyWords A B).events.length≤wordGateBudget q h z := by
  rw [multiplyWords_events_length]
  have HP := (multiplicationProducts_bound h z).trans (shape_square h z)
  simpa only [wordGateBudget] using Nat.mul_le_mul (unit_tensor q) HP

theorem evaluateWord_events_bound (q h z : ℕ) (e : Expr) :
    (evaluateWord q h z e).events.length≤wordGateBudget q h z*nodes e := by
  induction e with
  | literal v =>
    simp only [evaluateWord,nodes,Nat.mul_one]
    exact (ofIntWord_events_bound q v).trans (scalar_gate_cap q h z)
  | inner =>
    simp only [evaluateWord,nodes,List.length_nil,Nat.mul_one]
    exact Nat.zero_le _
  | outer =>
    simp only [evaluateWord,nodes,List.length_nil,Nat.mul_one]
    exact Nat.zero_le _
  | add a b ha hb =>
    simp only [evaluateWord,List.length_append,nodes]
    have H := addWords_gate_cap (evaluateWord q h z a).value (evaluateWord q h z b).value
    nlinarith [ha,hb,H]
  | neg a ha =>
    simp only [evaluateWord,List.length_append,nodes]
    have H := negateWords_gate_cap (evaluateWord q h z a).value
    nlinarith [ha,H]
  | mul a b ha hb =>
    simp only [evaluateWord,List.length_append,nodes]
    have H := multiplyWords_gate_cap (evaluateWord q h z a).value (evaluateWord q h z b).value
    nlinarith [ha,hb,H]

theorem actualApprox_gate_bound (q a u i x y k h z : ℕ) (t : ℤ) (m : ℕ → ℕ) :

structure DropLog where
  value : List Bool
  skipped : List Bool

def tracedDrop : ℕ → List Bool → DropLog
  | 0,xs => ⟨xs,[]⟩
  | _+1,[] => ⟨[],[]⟩
  | k+1,b::bs =>
      let t := tracedDrop k bs
      ⟨t.value,b::t.skipped⟩

theorem tracedDrop_value (k : ℕ) (xs : List Bool) : (tracedDrop k xs).value=xs.drop k := by
  induction k generalizing xs with
  | zero => rfl
  | succ k ih => cases xs <;> simp [tracedDrop,ih]

theorem tracedDrop_skipped (k : ℕ) (xs : List Bool) : (tracedDrop k xs).skipped=xs.take k := by
  induction k generalizing xs with
  | zero => rfl
  | succ k ih => cases xs <;> simp [tracedDrop,ih]

theorem tracedDrop_skipped_length (k : ℕ) (xs : List Bool) :
    (tracedDrop k xs).skipped.length=min k xs.length := by
  rw [tracedDrop_skipped,List.length_take]

theorem decode_drop_of_le (k : ℕ) (xs : List Bool) (hk : k≤xs.length) :
    OrientationBooleanAdder.decode (xs.drop k)=OrientationBooleanAdder.decode xs/(2^k) := by
  have P : (xs.take k).length=k := by rw [List.length_take,Nat.min_eq_left hk]
  have H := decode_append (xs.take k) (xs.drop k)
  rw [List.take_append_drop,P] at H
  have B := OrientationBooleanAdder.decode_bound (xs.take k)
  rw [P] at B
  rw [H,Nat.add_mul_div_left _ _ (by positivity),Nat.div_eq_of_lt B,Nat.zero_add]

theorem wordValue_val {q : ℕ} (w : Word q) :
    (wordValue w).val=OrientationBooleanAdder.decode w.val := by
  unfold wordValue
  exact ZMod.val_natCast_of_lt (by
    simpa only [w.property] using OrientationBooleanAdder.decode_bound w.val)

def quotientWord (p E : ℕ) (w : Word (p+E)) : Word p :=
  let d := tracedDrop E w.val
  ⟨d.value,by rw [tracedDrop_value,List.length_drop,w.property]; omega⟩

    wordValue (quotientWord p E w)=(((wordValue w).val/(2^E):ℕ):ZMod (2^p)) := by
  rw [wordValue_val]
  dsimp only [quotientWord,wordValue]
  rw [tracedDrop_value,decode_drop_of_le E w.val (by rw [w.property]; omega)]

theorem quotientWord_skipped_length (p E : ℕ) (w : Word (p+E)) :
    (tracedDrop E w.val).skipped.length=E := by
  rw [tracedDrop_skipped_length,w.property]
  omega

theorem finiteExtract_from_word (d p : ℕ) (w : Word (p+extra d)) :
    finiteExtract d p (wordValue w)=
      wordValue (quotientWord p (extra d) w)*(oddPart d:ZMod (2^p))⁻¹ := by

  have H : (((wordValue w).val:ℤ)/(2:ℤ)^extra d:ℤ)=
      (((wordValue w).val/(2^extra d):ℕ):ℤ) := by
    simpa only [Nat.cast_pow,Nat.cast_ofNat] using

  rw [H,Int.cast_natCast]

def newtonR {R : Type*} [CommRing R] (a : R) : ℕ → R
  | 0 => 1
  | k+1 => newtonR a k*(2-a*newtonR a k)

theorem error_step {R : Type*} [CommRing R] (a y : R) :
    1-a*(y*(2-a*y))=(1-a*y)^2 := by ring

theorem newtonR_error {R : Type*} [CommRing R] (a : R) (k : ℕ) :
    1-a*newtonR a k=(1-a)^(2^k) := by
  induction k with
  | zero => simp [newtonR]
  | succ k ih =>
    calc
      1-a*newtonR a (k+1)=(1-a*newtonR a k)^2 := by rw [newtonR,error_step]
      _ = ((1-a)^(2^k))^2 := by rw [ih]
      _ = (1-a)^(2^k*2) := (pow_mul _ _ _).symm
      _ = (1-a)^(2^(k+1)) :=
        congrArg (fun n : ℕ => (1-a)^n) (pow_succ (2:ℕ) k).symm

theorem precision_le_two_pow (p : ℕ) : p≤2^p := by
  induction p with
  | zero => decide
  | succ p ih =>
    rw [pow_succ]
    have H : 0<(2:ℕ)^p := by positivity
    omega

theorem two_nilpotent (p : ℕ) : (2:ZMod (2^p))^p=0 := by
  simpa only [Nat.cast_pow,Nat.cast_ofNat] using ZMod.natCast_self (2^p)

theorem two_high_power_zero (p : ℕ) : (2:ZMod (2^p))^(2^p)=0 := by
  have H := precision_le_two_pow p
  calc
    (2:ZMod (2^p))^(2^p)=(2:ZMod (2^p))^(p+(2^p-p)) :=
      congrArg (fun e : ℕ => (2:ZMod (2^p))^e) (by omega)
    _ = (2:ZMod (2^p))^p*(2:ZMod (2^p))^(2^p-p) := pow_add _ _ _
    _ = 0 := by rw [two_nilpotent,zero_mul]

theorem actual_initial_error_even (d p : ℕ) (hd : 0<d) :
    ∃ c : ZMod (2^p),1-(oddPart d:ZMod (2^p))=2*c := by
  have Hnz : oddPart d%2≠0 := by
    intro H
    exact oddPart_not_dvd d hd (Nat.dvd_of_mod_eq_zero H)
  have Hmod : oddPart d%2=1 := by
    have H := Nat.mod_lt (oddPart d) (by decide : 0<2)
    omega
  have H := Nat.mod_add_div (oddPart d) 2
  rw [Hmod] at H
  have HC := congrArg (fun n : ℕ => (n:ZMod (2^p))) H
  simp only [Nat.cast_add,Nat.cast_one,Nat.cast_mul,Nat.cast_ofNat] at HC
  refine ⟨-((oddPart d/2:ℕ):ZMod (2^p)),?_⟩
  linear_combination HC

theorem actual_initial_error_nilpotent (d p : ℕ) (hd : 0<d) :
    (1-(oddPart d:ZMod (2^p)))^(2^p)=0 := by
  obtain ⟨c,hc⟩ := actual_initial_error_even d p hd
  rw [hc,mul_pow,two_high_power_zero,zero_mul]

theorem newtonR_actual_inverse (d p : ℕ) (hd : 0<d) :
    newtonR (oddPart d:ZMod (2^p)) p=(oddPart d:ZMod (2^p))⁻¹ := by
  have H := newtonR_error (oddPart d:ZMod (2^p)) p
  rw [actual_initial_error_nilpotent d p hd] at H
  have HP : (oddPart d:ZMod (2^p))*newtonR (oddPart d:ZMod (2^p)) p=1 :=
    (sub_eq_zero.mp H).symm
  exact (ZMod.inv_eq_of_mul_eq_one (2^p) _ _ HP).symm

theorem even_zero_control (k : ℕ) : newtonR (0:ZMod 2) k*0≠1 := by
  simpa only [mul_zero] using (zero_ne_one : (0:ZMod 2)≠1)

end OrientationNewtonKernel

theorem decode_zeros (n : ℕ) : decode (List.replicate n false)=0 := by
  induction n with
  | zero => rfl

structure LiftRun (P : ℕ) where
  value : Word P
  copied : List Bool
  padding : List Bool

def liftWord {pc : ℕ} (w : Word pc) (P : ℕ) (h : pc≤P) : LiftRun P :=
  let padding := List.replicate (P-pc) false
  ⟨⟨w.val++padding,by dsimp only [padding]; simp only [List.length_append,List.length_replicate,w.property]; omega⟩,
    w.val,padding⟩

theorem liftWord_decode {pc : ℕ} (w : Word pc) (P : ℕ) (h : pc≤P) :
    decode (liftWord w P h).value.val=decode w.val := by
  simp only [liftWord,decode_append,decode_zeros,mul_zero,add_zero]

theorem liftWord_canonical {pc : ℕ} (w : Word pc) (P : ℕ) (h : pc≤P) :
    wordValue (liftWord w P h).value=((wordValue w).val:ZMod (2^P)) := by
  dsimp only [wordValue]
  rw [liftWord_decode]
  congr 1
  exact (ZMod.val_natCast_of_lt (by simpa only [w.property] using decode_bound w.val)).symm

theorem liftWord_val {pc : ℕ} (w : Word pc) (P : ℕ) (h : pc≤P) :
    (wordValue (liftWord w P h).value).val=(wordValue w).val := by
  have H : decode w.val<2^P := lt_of_lt_of_le
    (by simpa only [w.property] using decode_bound w.val)
    (Nat.pow_le_pow_right (by decide : 0<2) h)
  dsimp only [wordValue]
  rw [liftWord_decode,ZMod.val_natCast_of_lt H]
  exact (ZMod.val_natCast_of_lt (by simpa only [w.property] using decode_bound w.val)).symm

theorem liftWord_payload {pc : ℕ} (w : Word pc) (P : ℕ) (h : pc≤P) :
    (liftWord w P h).copied.length=pc ∧
    (liftWord w P h).padding.length=P-pc ∧
    (liftWord w P h).copied.length+(liftWord w P h).padding.length=P := by
  change w.val.length=pc ∧ (List.replicate (P-pc) false).length=P-pc ∧
    w.val.length+(List.replicate (P-pc) false).length=P
  rw [w.property,List.length_replicate]
  exact ⟨rfl,rfl,by omega⟩

theorem liftWord_trans_val {pc P Q : ℕ} (w : Word pc) (h : pc≤P) (h' : P≤Q) :
    (wordValue (liftWord (liftWord w P h).value Q h').value).val=
      (wordValue (liftWord w Q (h.trans h')).value).val := by
  rw [liftWord_val,liftWord_val,liftWord_val]

def retainedPolynomial (pc : ℕ) (f : ℤ[X]) : (ZMod (2^pc))[X] :=

abbrev Extension (pc : ℕ) (f : ℤ[X]) := AdjoinRoot (retainedPolynomial pc f)
abbrev EntryRing (pc : ℕ) (f : ℤ[X]) (ell : ℕ) :=
  AdjoinRoot (X^ell : (Extension pc f)[X])

def scalarMap (pc : ℕ) (f : ℤ[X]) (ell : ℕ) : ZMod (2^pc) →+* EntryRing pc f ell :=
  (AdjoinRoot.of (X^ell : (Extension pc f)[X])).comp
    (AdjoinRoot.of (retainedPolynomial pc f))

def nilpotent (pc : ℕ) (f : ℤ[X]) (ell : ℕ) : EntryRing pc f ell :=
  AdjoinRoot.root (X^ell : (Extension pc f)[X])

theorem retained_monic (pc : ℕ) (f : ℤ[X]) (hf : f.Monic) :
    (retainedPolynomial pc f).Monic := hf.map _

theorem extension_root (pc : ℕ) (f : ℤ[X]) :
    (retainedPolynomial pc f).eval₂ (AdjoinRoot.of (retainedPolynomial pc f))
      (AdjoinRoot.root (retainedPolynomial pc f))=0 := AdjoinRoot.eval₂_root _

theorem nilpotent_power (pc : ℕ) (f : ℤ[X]) (ell : ℕ) :
    nilpotent pc f ell^ell=0 := by
  simpa only [eval₂_pow,eval₂_X,nilpotent] using
    AdjoinRoot.eval₂_root (X^ell : (Extension pc f)[X])

theorem eval_nilpotent_finite {R S : Type*} [CommRing R] [CommRing S]
    (g : R →+* S) (z : S) (ell : ℕ) (hz : z^ell=0) (P : R[X]) :
    P.eval₂ g z=∑ n ∈ range ell,g (P.coeff n)*z^n := by
  let N := max (P.natDegree+1) ell
  have hN : P.natDegree<N := lt_of_lt_of_le (Nat.lt_succ_self _) (le_max_left _ _)
  rw [eval₂_eq_sum_range' g hN z]
  symm
  apply sum_subset (range_mono (le_max_right _ _))
  intro n hn hnsmall
  have hnell : ell≤n := Nat.le_of_not_gt (by simpa only [mem_range] using hnsmall)
  rw [pow_eq_zero_of_le hnell hz,mul_zero]

def wordEntrySum (pc : ℕ) (f : ℤ[X]) (ell : ℕ) (w : ℕ → Word pc) : EntryRing pc f ell :=
  ∑ n ∈ range ell,scalarMap pc f ell (wordValue (w n))*nilpotent pc f ell^n

theorem wordEntrySum_eq (pc : ℕ) (f : ℤ[X]) (ell : ℕ)
    (w : ℕ → Word pc) (P : (ZMod (2^pc))[X])
    (hw : ∀ n,wordValue (w n)=P.coeff n) :
    wordEntrySum pc f ell w=P.eval₂ (scalarMap pc f ell) (nilpotent pc f ell) := by
  rw [eval_nilpotent_finite _ _ ell (nilpotent_power pc f ell)]
  unfold wordEntrySum
  apply sum_congr rfl
  intro n _
  rw [hw]

theorem correction_single (u x y i : ℕ) (hi : i∈mask u x y) :
    correction u x y (singleMultiplicity i)=factor (delta x y) i := by
  unfold correction
  rw [prod_eq_single i]
  · simp [singleMultiplicity]
  · intro j _ hji
    simp [singleMultiplicity,hji]
  · intro hni
    exact False.elim (hni hi)

theorem anchor_single (u x y i : ℕ) (hi : i∈mask u x y) :
    anchorRows u x y (singleMultiplicity i)=
      1+C (C (OrientationMoment.slope u (rowResidue u x) y ((2:ℚ)^i)))*X := by
  unfold anchorRows
  rw [prod_eq_single i]
  · simp [singleMultiplicity]
  · intro j _ hji
    simp [singleMultiplicity,hji]
  · intro hni
    exact False.elim (hni hi)


def portTwo (a : ℕ) (ha : 2≤a) : Ports a :=
  ⟨2,lt_of_lt_of_le (by decide : 2<2^2) (Nat.pow_le_pow_right (by decide) ha)⟩

def portFour (a : ℕ) (ha : 3≤a) : Ports a :=
  ⟨4,lt_of_lt_of_le (by decide : 4<2^3) (Nat.pow_le_pow_right (by decide) ha)⟩

theorem offdiag_node_first (a : ℕ) (ha : 2≤a) :
    (normalizedNodeEntry a 2 0 0 2 (pole 0) (singleMultiplicity 0)).coeff 1=-(1:ℚ)/3 := by
  rw [actual_node_first]
  change firstJet a 2 0 0 2 (singleMultiplicity 0)/diagonalJet a 2 0 (singleMultiplicity 0)=-(1:ℚ)/3
  have hy : 2<2^a := lt_of_lt_of_le (by decide : 2<2^2) (Nat.pow_le_pow_right (by decide) ha)
  unfold diagonalJet
  rw [firstJet_formula a 2 0 0 2 _ (by positivity) hy (Nat.zero_le a) (by decide),
    firstJet_formula a 2 0 0 0 _ (by positivity) (by positivity) (Nat.zero_le a) (by decide),
    correction_single 2 0 2 0 (by decide),correction_single 2 0 0 0 (by decide)]
  have hJ := jetScalar_ne_zero a 0 (Nat.zero_le a)
  norm_num [poleQ,delta,OrientationAnchor.factor] <;> field_simp [hJ] <;> ring

theorem offdiag_geometric_first (a : ℕ) :
    ((geometricRows (denomRows a 2 0 2 (pole 0) (singleMultiplicity 0)) 2).coeff 1).coeff 0=(16:ℚ)/3 := by

    OrientationMoment.slope,delta,pole,chartSlope]

theorem offdiag_right_first (a : ℕ) (ha : 2≤a) :
    (polyRight a 2 0 2 (pole 0) (singleMultiplicity 0)
      ((0:Labels 2),(1:Fin 2)) (portTwo a ha)).coeff 1=-(8:ℚ)/9 := by
  have hz := actual_node_constant a 2 0 0 2 (pole 0) (singleMultiplicity 0)
    (by rw [pole_cast]; exact pole_member a 0 (Nat.zero_le a)) (pole_ne_zero 0)
    (pole_valuation 0) (by decide)
  have hN0 : (rowNumerator a 2 0 0 2 (pole 0) (singleMultiplicity 0)).coeff 0=0 := by
    simp only [rowNumerator,rowResidue,Nat.zero_mod,mul_coeff_zero,hz,mul_zero]
  have hN1 : (rowNumerator a 2 0 0 2 (pole 0) (singleMultiplicity 0)).coeff 1=-(1:ℚ)/3 := by

  have hA : ((approxRows a 2 0 0 2 2 (pole 0) (singleMultiplicity 0)).coeff 1).coeff 1=-(16:ℚ)/9 := by

      polyLeft a 2 2 x ((0:Labels 2),(1:Fin 2))≠0 ∧
      polyRight a 2 0 2 (pole 0) (singleMultiplicity 0) ((0:Labels 2),(1:Fin 2)) y≠0 := by
  refine ⟨portFour a ha,portTwo a (by omega),?_,?_⟩
  · rw [offdiag_left_nonzero_value a ha]
    exact C_ne_zero.mpr (by norm_num)
  · intro hz
    have H := offdiag_right_first a (by omega)
    rw [hz,coeff_zero] at H
    norm_num at H
