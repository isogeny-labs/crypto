theorem fibonacci_four_growth (n : ℕ) : 5*Nat.fib n ≤ Nat.fib (n+4) := by
  have h := Nat.fib_add n 3
  have hm : Nat.fib n ≤ Nat.fib (n+1) := Nat.fib_le_fib_succ
  have h3 : Nat.fib 3 = 2 := by decide
  have h4 : Nat.fib 4 = 3 := by decide
  rw [h3, h4] at h
  have hi : n+3+1=n+4 := by omega
  rw [hi] at h
  omega

theorem fibonacci_geometric (k : ℕ) : 5^k ≤ Nat.fib (4*k+1) := by
  induction k with
  | zero => decide
  | succ k ih =>
    have hg := fibonacci_four_growth (4*k+1)
    have he : 4*(k+1)+1=(4*k+1)+4 := by omega
    rw [he, pow_succ]
    omega

theorem actual_terminal_geometric (a k x K : ℕ) (m : ℕ → ℕ)
    (hm : ∀ i < 2*k+1, 0 ≤ schedule (2*k+1) i m)
    (hdegree : (completed (grid a) (2*k+1) x x m).natDegree ≤
      (2*2^a-1)+K*2^a) :
    5^k*m (2*k) ≤ K*2^a := by
  have hf : 5^k ≤ Nat.fib (2*(2*k)+1) := by
    convert fibonacci_geometric k using 1 <;> congr 1 <;> omega
  exact (Nat.mul_le_mul_right (m (2*k)) hf).trans
    (actual_terminal_polynomial_budget a (2*k) x K m hm hdegree)

theorem power_split (a k : ℕ) (ha : 2*k ≤ a) :
    2^a = 4^k*2^(a-2*k) := by
  have he : 2*k+(a-2*k)=a := by omega
  calc
    2^a = 2^(2*k+(a-2*k)) := by rw [he]
    _ = 2^(2*k)*2^(a-2*k) := pow_add _ _ _
    _ = 4^k*2^(a-2*k) := by rw [pow_mul]; norm_num

theorem actual_terminal_relative_scale (a k x K : ℕ) (m : ℕ → ℕ)
    (ha : 2*k ≤ a)
    (hm : ∀ i < 2*k+1, 0 ≤ schedule (2*k+1) i m)
    (hdegree : (completed (grid a) (2*k+1) x x m).natDegree ≤
      (2*2^a-1)+K*2^a) :
    (m (2*k):ℚ)/(2:ℚ)^(a-2*k) ≤ (K:ℚ)*((4:ℚ)/5)^k := by
  have hb := actual_terminal_geometric a k x K m hm hdegree
  rw [power_split a k ha] at hb
  have hc : (5:ℚ)^k*(m (2*k):ℚ) ≤
      (K:ℚ)*(4:ℚ)^k*(2:ℚ)^(a-2*k) := by exact_mod_cast hb
  apply (div_le_iff₀ (by positivity : 0 < (2:ℚ)^(a-2*k))).mpr
  rw [div_pow, ← mul_div_assoc, div_mul_eq_mul_div]
  apply (le_div_iff₀ (by positivity : 0 < (5:ℚ)^k)).mpr
  simpa only [mul_comm] using hc

end OrientationBudgetScale
