theorem nonzero_seven_shadow_rejected {A : Type} [IntModule A]
    (a : A) (ha : (7 : Int) • a=0) (hne : a≠0) :
    (4976069 : Int) • a=0 ∧ (1244018 : Int) • a = -a ∧
    (1244018 : Int) • a≠a := by
  have hz : (4976069 : Int) • a=0 := by grind
  have hn : (1244018 : Int) • a = -a := by grind
  refine ⟨hz, hn, ?_⟩
  intro he
  have h2 : (2 : Int) • a=0 := by grind
  apply hne
  calc
    a = (7 : Int) • a - (3 : Int) • ((2 : Int) • a) := by grind
    _ = 0 := by rw [ha, h2]; grind
