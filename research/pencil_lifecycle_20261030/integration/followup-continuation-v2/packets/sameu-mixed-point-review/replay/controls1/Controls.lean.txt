import WeightedExtract
set_option autoImplicit false
open Polynomial MixedCancellation SameUMixed
-- Two distinct noninverse ordinates over the singular monic quintic X^5.
theorem singular_input1 : support (0:ℚ) 0 ∣ X^5-(ordinate (1:ℚ) 0)^2 := by
 refine ⟨X^3-1,?_⟩
 simp only [MixedCancellation.support,ordinate,C_0,C_1]
 ring
theorem singular_input2 : support (0:ℚ) 0 ∣ X^5-(ordinate (2:ℚ) 0)^2 := by
 refine ⟨X^3-(C 2)^2,?_⟩
 norm_num [MixedCancellation.support,ordinate]
 <;> ring
-- Applies the actual theorem with all its original congruence premises.
example : let x := ((0:ℚ)-0)/(1-2)
 x^2+0*x+0=0 ∧ (X-C x) ∣ X^5-(C (1*x+0))^2 := by
 exact extracted_point_valid (X^5:ℚ[X]) 0 0 1 0 2 0 singular_input1 singular_input2
  (by norm_num) (by norm_num)
-- The source-extracted point ordinate is zero: validity does not imply nonzero y.
example : (1:ℚ)*((0-0)/(1-2))+0=0 := by norm_num
-- Omitting inverse-false permits nonzero slope difference but x outside support.
theorem inverse_input1 : support (0:ℚ) 1 ∣ (X^5+X^3+X^2)-(ordinate (1:ℚ) 0)^2 := by
 refine ⟨X^3,?_⟩
 simp only [MixedCancellation.support,ordinate,C_0,C_1]
 ring
theorem inverse_input2 : support (0:ℚ) 1 ∣ (X^5+X^3+X^2)-(ordinate (-1:ℚ) 0)^2 := by
 refine ⟨X^3,?_⟩
 simp only [MixedCancellation.support,ordinate,C_0,C_1,map_neg]
 ring
example : (1:ℚ)-(-1) ≠ 0 ∧ (((0:ℚ)-0)/(1-(-1)))^2+1 ≠ 0 := by norm_num

def singularA : D2 ℚ := ⟨0,0,1,0,1,1⟩
def singularB : D2 ℚ := ⟨0,0,2,0,1,1⟩
example : (sourcePoint singularA singularB).z≠0 ∧
 (sourcePoint singularA singularB).w≠0 ∧
 validPoint (X^5:ℚ[X]) (sourcePoint singularA singularB) := by
 apply source_point_valid
 · norm_num [singularA]
 · norm_num [singularA]
 · norm_num [singularB]
 · norm_num [singularB]
 · simpa [validTwo,nq,nr,ns,nt,singularA] using singular_input1
 · simpa [validTwo,nq,nr,ns,nt,singularB] using singular_input2
 · norm_num [nq,singularA,singularB]
 · norm_num [nr,singularA,singularB]
 · norm_num [ns,singularA,singularB]
 · norm_num [ns,nt,singularA,singularB]
-- Literal following dbl returns weight 2*y, which vanishes on this singular input.
example : (sourcePoint singularA singularB).y=0 ∧
 2*(sourcePoint singularA singularB).y=0 := by
 norm_num [sourcePoint,sourceK,singularA,singularB]
