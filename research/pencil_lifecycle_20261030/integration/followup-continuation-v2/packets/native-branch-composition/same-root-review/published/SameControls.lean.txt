import SameComposition
set_option autoImplicit false
namespace SameControls
open SignedGraphAlgebra GraphFind GraphUnion GraphForest GraphSameRoot
def g : GraphFind.State ℚ 4 := ⟨fun i=>if i=2 ∨ i=3 then 1 else i,
 fun i=>if i=2 then ⟨1,2,0⟩ else if i=3 then ⟨-1,5,0⟩ else ident⟩
def s : GraphUnion.State ℚ 4 := ⟨g,fun i=>if i=0 then some (0,0) else none⟩
def val (i:Fin 4) : ℚ := if i=0 then 0 else if i=1 then 3 else if i=2 then 5 else 2
def rank (i:Fin 4) : Nat := if i=2 ∨ i=3 then 1 else 0
def p2 : Potential ℚ := compose (g.pot 2) ident
def t : GraphFind.State ℚ 4 := compress g 2 1 p2
def p3 : Potential ℚ := compose (t.pot 3) ident
def w : GraphFind.State ℚ 4 := compress t 3 1 p3

theorem valid_initial : Valid s val 2 := by
 constructor
 · intro i; fin_cases i <;> norm_num [s,g,val,ident,eval]
 · intro i a b h
   fin_cases i <;> simp [s] at h
   rcases h with ⟨rfl,rfl⟩; norm_num [val]
theorem forest_initial : Forest g rank := by intro i; fin_cases i <;> norm_num [g,rank]
theorem first_nontrivial : Run g 2 t 1 p2 := by
 apply Run.step g g 2 1 ident
 · decide
 · change Run g 1 g 1 ident
   exact Run.root g 1 (by decide)
theorem second_nontrivial : Run t 3 w 1 p3 := by
 apply Run.step t t 3 1 ident
 · decide
 · change Run t 1 t 1 ident
   exact Run.root t 1 (by decide)

theorem composed_opposite :
 Forest (afterSame s w 1 p2 p3 1 1 3 2).1.graph rank ∧
 Valid (afterSame s w 1 p2 p3 1 1 3 2).1 val 2 ∧
 Safe (afterSame s w 1 p2 p3 1 1 3 2).2 2 := by
 apply same_after_two_finds s t w 2 3 1 p2 p3 rank val 1 1 3 2 2
 · exact forest_initial
 · exact valid_initial
 · exact first_nontrivial
 · exact second_nontrivial
 · norm_num
 · left; norm_num [p2,compose,g,ident]
 · right; norm_num [p3,t,compress,g,compose,ident]
 · norm_num [val]
theorem opposite_returns : (afterSame s w 1 p2 p3 1 1 3 2).2=.answer 2 := by
 norm_num [afterSame,sameRoot,GraphSameRoot.constraint,p2,p3,t,compress,g,compose,ident]
theorem new_positive_anchor : (sameRoot s 1 1 1 4 1).1.fixed 1=some (2,1/2) := by
 norm_num [sameRoot,anchor,s]
theorem new_negative_anchor : (sameRoot s 1 (-1) (-1) (-4) (-1)).1.fixed 1=some (2,1/2) := by
 norm_num [sameRoot,anchor,s]
def fixed : GraphUnion.State ℚ 4 := ⟨g,fun i=>if i=1 then some (3,0) else s.fixed i⟩
theorem repeated_anchor_answer : (sameRoot fixed 1 1 1 4 1).2=.answer 2 := by
 norm_num [sameRoot,anchor,fixed,GraphSameRoot.constraint]
theorem consistent_zero : GraphSameRoot.constraint (0:ℚ) 0=.none := by norm_num [GraphSameRoot.constraint]
theorem inconsistent_assertion : GraphSameRoot.constraint (1:ℚ) 0=.inconsistent := by norm_num [GraphSameRoot.constraint]
theorem composed_equal :
 Forest (afterSame s w 1 p2 p3 1 (-1) 1 1).1.graph rank ∧
 Valid (afterSame s w 1 p2 p3 1 (-1) 1 1).1 val 2 ∧
 Safe (afterSame s w 1 p2 p3 1 (-1) 1 1).2 2 := by
 apply same_after_two_finds s t w 2 3 1 p2 p3 rank val 1 (-1) 1 1 2
 · exact forest_initial
 · exact valid_initial
 · exact first_nontrivial
 · exact second_nontrivial
 · norm_num
 · left; norm_num [p2,compose,g,ident]
 · left; norm_num [p3,t,compress,g,compose,ident]
 · norm_num [val]
end SameControls
