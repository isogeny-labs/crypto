noncomputable def curve : (ZMod (17:UInt64).toNat)[X] := X^5-X^4-X^3+1

theorem literal_payload : slope0 17 sample=12 ∧ slope1 17 sample=8 ∧ inverseLead 17 sample=13 := by decide

theorem curve_degree : curve.natDegree=5 := by
 have h1 : ((X:(ZMod (17:UInt64).toNat)[X])^5-X^4).natDegree=5 := by
  rw [natDegree_sub_eq_left_of_natDegree_lt (by simp only [natDegree_X_pow];decide),natDegree_X_pow]
 have h2 : ((X:(ZMod (17:UInt64).toNat)[X])^5-X^4-X^3).natDegree=5 := by
  rw [natDegree_sub_eq_left_of_natDegree_lt (by rw [h1,natDegree_X_pow];decide),h1]
 unfold curve
 rw [natDegree_add_eq_left_of_natDegree_lt (by rw [h2];simp),h2]
