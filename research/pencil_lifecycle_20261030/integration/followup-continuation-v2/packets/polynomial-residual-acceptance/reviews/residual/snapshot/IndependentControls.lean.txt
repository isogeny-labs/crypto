import Acceptance
open ResidualAcceptance ResidualWord
-- Exact square subtraction and unequal operand lengths.
theorem exact_square_residual : residual 17 [1,2,1] [1,1] = [] := by decide
theorem padded_subtraction : psub 17 [2] [1,2,3] = [1,15,14] := by decide
-- Identity-shaped input and a nonzero residual division.
theorem identity_accepts : accepts 17 0 [1] [] [3,4] 2 := by unfold accepts; decide
theorem degree_guard_rejects : ¬accepts 17 3 [1] [] [] 0 := by unfold accepts; decide
-- Algebraic validity alone does not force representation guard: last stored zero is rejected.
theorem trailing_zero_divisor_rejects : ¬accepts 17 0 [1,0] [] [] 0 := by unfold accepts; decide
-- Acceptance predicate alone omits the shape restriction on v.
theorem shape_premise_needed : accepts 17 0 [1] [1] [1] 0 := by unfold accepts; decide
