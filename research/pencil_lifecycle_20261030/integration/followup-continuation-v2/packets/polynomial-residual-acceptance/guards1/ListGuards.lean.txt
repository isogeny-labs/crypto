theorem monic_list_natDegree (p : UInt64) [Fact p.toNat.Prime] (u : List UInt64)
 (hm : u.getLast?=some 1) : (wordsPoly p u).natDegree=u.length-1 := by
 have hu : 0<u.length := List.length_pos_iff.mpr (monic_nonempty u hm)
 apply natDegree_eq_of_le_of_coeff_ne_zero
 · rw [natDegree_le_iff_coeff_eq_zero]
   intro i hi
   rw [wordsPoly_coefficient]
   simp [List.getElem?_eq_none (by omega : u.length≤i)]
 · rw [top_coefficient,hm]
   simp

theorem monic_list (p : UInt64) [Fact p.toNat.Prime] (u : List UInt64)
 (hm : u.getLast?=some 1) : (wordsPoly p u).Monic := by
 change (wordsPoly p u).coeff (wordsPoly p u).natDegree=1
 rw [monic_list_natDegree p u hm,top_coefficient,hm]
 simp

theorem shape_valid (p : UInt64) [Fact p.toNat.Prime] (d : Nat) (u v : List UInt64)
 (f : (ZMod p.toNat)[X]) (hd : d≤2) (hu : u.length=d+1) (hv : v.length≤d)
 (hm : u.getLast?=some 1) (hr : wordsPoly p u∣f-(wordsPoly p v)^2) :
 Valid f (wordsPoly p u) (wordsPoly p v) := by
 have hmon:=monic_list p u hm
 have hdeg : (wordsPoly p u).natDegree=d := by rw [monic_list_natDegree p u hm,hu]; omega
 refine ⟨hmon,by rw [hdeg]; exact hd,?_,hr⟩
 rw [degree_eq_natDegree hmon.ne_zero,hdeg]
 exact lt_of_lt_of_le (degree_lt_length p v) (by exact_mod_cast hv)
end ResidualAcceptance
