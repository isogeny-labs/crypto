-- This is the final storage boundary, conditional on the already computed remainder.
theorem final_storage (p : UInt64) [Fact p.toNat.Prime] (q r : List UInt64)
 (f : (ZMod p.toNat)[X]) (hp : 1 < p.toNat) (hb : p.toNat < 2^31)
 (hc : Canon p q) (hn : Normal q) (he : q ≠ [])
 (hd : (wordsPoly p q).natDegree ≤ 2)
 (hr : r.length < (monic p q).length)
 (hv : wordsPoly p (monic p q) ∣ f-(wordsPoly p r)^2) :
 (monic p q) ≠ [] ∧ (monic p q).length ≤ 3 ∧ r.length ≤ 2 ∧
 MumfordValidity.Valid f (wordsPoly p (monic p q)) (wordsPoly p r) := by
 have hs := monic_shape p q hp hb hc hn he
 have hlen := normal_natDegree p q hp hc hn he
 have hpos : 0 < q.length := List.length_pos_iff.mpr he
 have hml : (monic p q).length=q.length := hs.2.2.1
 have hshort : (monic p q).length ≤ 3 := by omega
 have hmne : monic p q ≠ [] := by intro hz; simp [hz] at hml; omega
 refine ⟨hmne,hshort,by omega,?_⟩
 exact ResidualAcceptance.shape_valid p ((monic p q).length-1) (monic p q) r f
   (by omega) (by omega) (by omega) hs.2.2.2.1 hv

theorem quintic_final_storage (p : UInt64) [Fact p.toNat.Prime]
 (q r : List UInt64) (f u v : (ZMod p.toNat)[X])
 (hp : 1 < p.toNat) (hb : p.toNat < 2^31)
 (hc : Canon p q) (hn : Normal q)
 (hf : f.natDegree=5) (hu : 3≤u.natDegree) (hu4 : u.natDegree≤4)
 (hv : v.natDegree<u.natDegree) (hq : f-v*v=wordsPoly p q*u)
 (hr : r.length < (monic p q).length)
 (hvalid : wordsPoly p (monic p q) ∣ f-(wordsPoly p r)^2) :
 (monic p q) ≠ [] ∧ (monic p q).length ≤ 3 ∧ r.length ≤ 2 ∧
 MumfordValidity.Valid f (wordsPoly p (monic p q)) (wordsPoly p r) := by
 have hd := quintic_quotient_bound f u v (wordsPoly p q) hf hu hu4 hv hq
 have he : q ≠ [] := by intro hz; apply hd.1; simp [hz,wordsPoly,ScaleTrim.listPoly]
 exact final_storage p q r f hp hb hc hn he hd.2 hr hvalid
end ReductionWord
