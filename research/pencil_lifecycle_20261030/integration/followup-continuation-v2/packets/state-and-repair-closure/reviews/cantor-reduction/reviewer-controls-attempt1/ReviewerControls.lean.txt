import CantorControls
set_option autoImplicit false
namespace ReviewerCantor
open Polynomial CZeroCantor CZeroInterpolation
noncomputable def u1 : ℚ[X] := X
noncomputable def u2 : ℚ[X] := X-1
noncomputable def ell : ℚ[X] := 2
noncomputable def v1 : ℚ[X] := ell+u1
noncomputable def v2 : ℚ[X] := ell+u2
noncomputable def q : ℚ[X] := X-3
noncomputable def f : ℚ[X] := ell^2+u1*u2*q
noncomputable def c1 : ℚ[X] := 1-(v1+v2)

theorem nonzero_second_bezout :
 c1*1*u1*v2+c1*(-1)*u2*v1+1*(v1*v2+f)=
 ell+u1*u2*(c1*C (1:ℚ)*(1+(-1))+1*(C (1:ℚ)^2+q)) := by
 apply full_cantor_numerator
 · dsimp [u1,u2]; ring
 · dsimp [c1]; ring
 · dsimp [v1]; simp
 · dsimp [v2]; simp
 · dsimp [f]; ring

theorem second_coefficient_nonzero : (1:ℚ[X])≠0 := one_ne_zero

theorem wrong_linear_quotient :
 support (1:ℚ) 2*support 0 1*(X-C 3) ≠ support (1:ℚ) 2*support 0 1*(X-C 4) := by
 intro h
 have he:=congrArg (fun z:ℚ[X]=>z.eval 0) h
 norm_num [support] at he

theorem ramification_factorization :
 (X^5-X : ℚ[X])=(X^4-1)*X := by ring

theorem ramification_simple_coefficient :
 (X^5-X : ℚ[X]).coeff 1 = -1 := by simp

theorem positive_and_negative_points_differ : (291:ℚ)≠ -291 := by norm_num

theorem repeated_residual_support :
 ((X-1)^2*(X+1)^2 : ℚ[X])*(X-1)=(X-1)^3*(X+1)^2 := by ring
end ReviewerCantor
