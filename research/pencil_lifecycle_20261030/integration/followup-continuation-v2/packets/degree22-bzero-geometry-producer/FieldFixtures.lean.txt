 (WeightedDoubling.curve (0:ZMod 5) (-1) 1 0).derivative := by
 refine ⟨4,3*(X^4+C 3*X^3+C 4*X^2+C 3*X+C 3),?_⟩
 norm_num [WeightedDoubling.curve,derivative_add,derivative_mul,derivative_X_pow]

  _ = (2*(X:(ZMod 5)[X])^5+3*X^4+3*X^3+2*X^2+X-2)*5+1 := by ring
  _ = 1 := by rw [h5]; ring
