import BZero
set_option autoImplicit false
namespace ScholtenCover
variable {K : Type*} [Field K]
def A (i t : K) := (1+i)*t+i
def B (i t : K) := (1-i)*t-i

theorem determinant (i : K) : (1+i)*(-i)-i*(1-i)= -2*i := by ring

theorem leading (i x : K) (hi : i^2 = -1) :
 (1+i)^2-x*(1-i)^2 = 2*i*(1+x) := by
 ring_nf
 rw [hi]
 ring

theorem fibre_coefficients (i t x : K) (hi : i^2 = -1) :
 (A i t)^2-x*(B i t)^2 =
 ((1+i)^2-x*(1-i)^2)*t^2 + (2*i*(1+i)+2*x*i*(1-i))*t + (x-1) := by
 dsimp [A,B]
 ring_nf
 rw [hi]
 ring

theorem exceptional_fibre (i t : K) (hi : i^2 = -1) :
 (A i t)^2+(B i t)^2 = -4*t-2 := by
 dsimp [A,B]
 ring_nf
 rw [hi]
 ring

theorem leading_nonzero (i x : K) (hi : i^2 = -1) (hi0 : i≠0)
 (h2 : (2:K)≠0) (hx : x≠ -1) : (1+i)^2-x*(1-i)^2 ≠ 0 := by
 rw [leading i x hi]
 apply mul_ne_zero (mul_ne_zero h2 hi0)
 intro h
 apply hx
 calc x = (1+x)-1 := by ring
      _ = -1 := by rw [h]; ring

theorem reciprocal (a b : K) (ha : a≠0) (hb : b≠0) :
 (a/b)*(b/a)=1 := by field_simp

theorem sextic_chart (a b r s sb rb : K) (hb : b≠0) :
 (r*a^6+s*a^4*b^2+sb*a^2*b^4+rb*b^6)/b^6 =
 r*(a/b)^6+s*(a/b)^4+sb*(a/b)^2+rb := by
 field_simp
 <;> ring

theorem conjugate_A (i t : K) (σ : K →+* K)
 (ht : σ t=t) (hi : σ i= -i) : σ (A i t)=B i t := by
 simp only [A,B,map_add,map_mul,map_one,ht,hi]
 ring

theorem conjugate_B (i t : K) (σ : K →+* K)
 (ht : σ t=t) (hi : σ i= -i) : σ (B i t)=A i t := by
 simp only [A,B,map_sub,map_mul,map_one,ht,hi]
 ring

theorem opposite_fibre (z u : K) :
 ((-z)^2,u*(-z)^3)=(z^2,-(u*z^3)) := by
 apply Prod.ext <;> dsimp <;> ring

theorem reciprocal_fibre (z y : K) (hz : z≠0) :
 ((-z)^2,(-y)/(-z)^3)=(z^2,y/z^3) := by
 apply Prod.ext
 · dsimp; ring
 · dsimp; field_simp; ring

theorem two_torsion_rhs (r rb d : K) :
 -r+(r+d)-(rb+d)+rb=0 := by ring
end ScholtenCover
