  refine ⟨?_,?_,?_,?_⟩
  · change (X^2-1 : ℚ[X]).leadingCoeff = 1
    rw [leadingCoeff_sub_of_degree_lt (by simp)]

  · simpa only [natDegree_pow, natDegree_X, mul_one] using (le_of_eq (natDegree_sub_C (p := (X:ℚ[X])^2) (a := (1:ℚ))))
  · simp only [degree_zero]
    exact bot_lt_iff_ne_bot.mpr (degree_ne_bot.mpr (by
      intro h
      have hn := congrArg Polynomial.natDegree h
      have he := natDegree_sub_C (p := (X:ℚ[X])^2) (a := (1:ℚ))
      norm_num at hn he
      omega))
  · exact ⟨X^3+2,by simp⟩
