import Mathlib.Algebra.Field.Basic
import Mathlib.Tactic.Ring
import Lean.Elab.Tactic.Omega
set_option autoImplicit false
namespace Solve4Stage
variable {K : Type*} [Field K]
abbrev Row (K : Type*) := Fin 5 → K
abbrev Vector (K : Type*) := Fin 4 → K
def residual (v : Row K) (x : Vector K) : K := v 0*x 0+v 1*x 1+v 2*x 2+v 3*x 3-v 4
def suffix (col : Fin 4) (v pivot : Row K) (z : K) : Row K :=
 fun j=>if col.val≤j.val then v j-z*pivot j else v j
theorem suffix_full (col : Fin 4) (v pivot : Row K) (z : K)
 (hzero : ∀ j : Fin 5,j.val<col.val→pivot j=0) :
 suffix col v pivot z=fun j=>v j-z*pivot j := by
 funext j
 by_cases h:col.val≤j.val
 · simp [suffix,h]
 · simp [suffix,h,hzero j (by omega)]
theorem residual_sub (v pivot : Row K) (z : K) (x : Vector K) :
 residual (fun j=>v j-z*pivot j) x=residual v x-z*residual pivot x := by
 simp only [residual]
 ring
theorem elimination_residual (col : Fin 4) (v pivot : Row K) (z : K) (x : Vector K)
 (hzero : ∀ j : Fin 5,j.val<col.val→pivot j=0) :
 residual (suffix col v pivot z) x=residual v x-z*residual pivot x := by
 rw [suffix_full col v pivot z hzero];exact residual_sub v pivot z x
theorem equation_preserved (col : Fin 4) (v pivot : Row K) (z : K) (x : Vector K)
 (hzero : ∀ j : Fin 5,j.val<col.val→pivot j=0) (hp : residual pivot x=0) :
 residual (suffix col v pivot z) x=0 ↔ residual v x=0 := by
 rw [elimination_residual col v pivot z x hzero,hp];simp
theorem pivot_zero (col : Fin 4) (v pivot : Row K)
 (hp : pivot col.castSucc=1) : suffix col v pivot (v col.castSucc) col.castSucc=0 := by
 simp [suffix,hp]
end Solve4Stage
