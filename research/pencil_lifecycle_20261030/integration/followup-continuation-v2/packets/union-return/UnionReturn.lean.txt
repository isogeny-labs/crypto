import UnionAfterFinds
set_option autoImplicit false
namespace UnionReturn
open SignedWord
variable {n : Nat}
def word (r : UInt64) (s : UnionState n) (child parent : Fin n) (p : Pot) : GraphSameRoot.Result UInt64 :=
 match s.fixed child,s.fixed parent with
 | some ab,some xy =>
   let moved := transportPair r p ab
   constraint r (GraphRingWord.sub r xy.1 moved.1) (GraphRingWord.sub r xy.2 moved.2)
 | _,_ => .none
def field {K : Type*} [Field K] [DecidableEq K] (s : GraphUnion.State K n)
 (child parent : Fin n) (p : SignedGraphAlgebra.Potential K) : GraphSameRoot.Result K :=
 match s.fixed child,s.fixed parent with
 | some ab,some xy =>
   let moved:=GraphUnion.transport p ab
   GraphSameRoot.constraint (xy.1-moved.1) (xy.2-moved.2)
 | _,_ => .none
theorem word_refines (r : UInt64) [Fact r.toNat.Prime] (s : UnionState n)
 (child parent : Fin n) (p : Pot) (hp : 1<r.toNat) (hr : r.toNat≤2^63)
 (hg : GoodPot r p) (hf : FixedGood r s) :
 outDecode r (word r s child parent p)=field (unionDecode r s) child parent (decPot r p) := by
 cases hc:s.fixed child with
 | none => simp [word,field,unionDecode,hc,outDecode]
 | some ab =>
   cases ht:s.fixed parent with
   | none => simp [word,field,unionDecode,hc,ht,outDecode]
   | some xy =>
     have ha:=hf child ab hc
     have hx:=hf parent xy ht
     have hm:=transportPair_correct r p ab (by omega) hr hg ha
     have h1:=GraphRingWord.sub_correct r xy.1 (transportPair r p ab).1 (by omega) hr hx.1 hm.1.1
     have h2:=GraphRingWord.sub_correct r xy.2 (transportPair r p ab).2 (by omega) hr hx.2 hm.1.2
     have hh:=constraint_refines r _ _ hp h1.1 h2.1
     have hm1:=congrArg Prod.fst hm.2
     have hm2:=congrArg Prod.snd hm.2
     simp only [anchorPairDecode,wd] at hm1 hm2
     simpa [word,field,unionDecode,hc,ht,anchorPairDecode,wd,h1.2,h2.2,hm1,hm2] using hh

theorem field_safe {K : Type*} [Field K] [DecidableEq K] (s : GraphUnion.State K n)
 (child parent : Fin n) (p : SignedGraphAlgebra.Potential K) (v : Fin n→K) (k : K)
 (hv : GraphUnion.Valid s v k) (hs : p.s*p.s=1)
 (he : v child=SignedGraphAlgebra.eval p (v parent) k) :
 GraphSameRoot.Safe (field s child parent p) k := by
 cases hc:s.fixed child with
 | none => simp [field,hc,GraphSameRoot.Safe]
 | some ab =>
   cases ht:s.fixed parent with
   | none => simp [field,hc,ht,GraphSameRoot.Safe]
   | some xy =>
     have hm:=SignedGraphAlgebra.fixed_child_transport p (v child) (v parent) ab.1 ab.2 k hs he (hv.2 child ab.1 ab.2 hc)
     have hx:=hv.2 parent xy.1 xy.2 ht
     simp only [field,hc,ht]
     apply GraphSameRoot.constraint_safe
     change xy.1-(GraphUnion.transport p ab).1+(xy.2-(GraphUnion.transport p ab).2)*k=0
     change v parent=(GraphUnion.transport p ab).1+(GraphUnion.transport p ab).2*k at hm
     calc
       _ = (xy.1+xy.2*k)-((GraphUnion.transport p ab).1+(GraphUnion.transport p ab).2*k) := by ring
       _ = 0 := by rw [←hx,←hm];ring
end UnionReturn
