theorem fixture_decoding : wordsPoly 17 fwords=FastPayloadControls.curve := by
 have h16 : (16 : ZMod (17:UInt64).toNat)= -1 := by decide
 norm_num [fwords,wordsPoly,listPoly,FastPayloadControls.curve]

 ∃ q r, FallbackWord.exactDiv 17 (ResidualWord.residual 17 fwords (v 17 small)) (u 17 small)=some q ∧
 FallbackGeometry.remainder 17 (ResidualWord.pneg 17 (v 17 small)) (scaled 17 small q)=some r ∧
 Canon 17 q ∧ MonicZeroTest.Normal q ∧ Canon 17 (scaled 17 small q) ∧ MonicZeroTest.Normal (scaled 17 small q) ∧

 (scaled 17 small q).getLast?=some 1 ∧ r.length≤2 ∧
 MumfordValidity.Valid (wordsPoly 17 fwords) (wordsPoly 17 (scaled 17 small q)) (wordsPoly 17 r) := by
 apply fast_output_exists 17 small fwords (by decide) (by decide)
  (by unfold Inputs.canonical;decide) (by decide) (by decide)
  (by simp [Canon,fwords];decide)
  (by rw [fixture_decoding];exact FastPayloadControls.curve_degree)
 · refine ⟨X^3-X^2-X,?_⟩
   rw [fixture_decoding]
   norm_num [small,FastCRT.support,FastCRT.delta,FastPayloadControls.curve]
   ring
 · refine ⟨X^3-X^2-2*X,?_⟩
   rw [fixture_decoding]
   norm_num [small,FastCRT.support,FastCRT.delta,FastPayloadControls.curve]
   ring
