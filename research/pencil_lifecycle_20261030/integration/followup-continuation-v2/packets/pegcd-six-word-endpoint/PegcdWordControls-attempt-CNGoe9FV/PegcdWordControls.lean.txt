import PegcdWordOutput
set_option autoImplicit false
namespace PegcdWord
private def z0 := initial [16,0,1] [16,1]
private def z1 := advance 17 z0 [1,1] []
theorem control_exact_run : Run 17 z0 z1 :=
 Run.next z0 1 [1,1] [] z1 (by decide) (by decide) (Run.stop z1 (by decide))
theorem control_exact_output : finish 17 z1=some ([16,1],[],[1]) := by decide +kernel
private def w0 := initial [1,0,1] [1,1]
private def w1 := advance 17 w0 [16,1] [2]
private def w2 := advance 17 w1 [9,9] []
theorem control_two_step_run : Run 17 w0 w2 :=
 Run.next w0 1 [16,1] [2] w2 (by decide) (by decide)
   (Run.next w1 2 [9,9] [] w2 (by decide) (by decide) (Run.stop w2 (by decide)))
private theorem qprod : ProductPolynomial.product 17 [16,1] [1]=[16,1] := by decide
private theorem qempty : ProductPolynomial.product 17 [16,1] []=[] := by decide
private theorem negq : ResidualWord.pneg 17 [16,1]=[1,16] := by decide
private theorem svalue : w1.s1=[1] := by
 change ResidualWord.psub 17 [1] (ProductPolynomial.product 17 [16,1] [])=[1]
 rw [qempty]
 simp only [ResidualWord.psub,ResidualWord.pneg,List.map_nil,ResidualWord.padd,ResidualWord.slots]
 decide
private theorem tvalue : w1.t1=[1,16] := by
 change ResidualWord.psub 17 [] (ProductPolynomial.product 17 [16,1] [1])=[1,16]
 rw [qprod,ResidualWord.psub,negq]
 simp only [ResidualWord.padd,ResidualWord.slots]
 decide
theorem control_nonmonic_output : finish 17 w2=some ([1],[9],[9,8]) := by
 change some (scale 17 (WideModPow.inverseGenerator 17 2) [2],scale 17 (WideModPow.inverseGenerator 17 2) w1.s1,scale 17 (WideModPow.inverseGenerator 17 2) w1.t1)=some ([1],[9],[9,8])
 rw [svalue,tvalue]
 decide
theorem control_subtraction_sign : w1.t1=[1,16] ∧ w1.t1≠[16,1] := by
 rw [tvalue]
 decide
end PegcdWord
