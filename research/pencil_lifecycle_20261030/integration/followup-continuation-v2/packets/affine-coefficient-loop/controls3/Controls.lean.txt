import Loop
set_option autoImplicit false
namespace AffineLoopControls
open Polynomial AffineCoefficientLoop MumfordNormalization

theorem half_scale : C (1/2:ℚ)*(2:ℚ[X])=1 := by
 calc
  _ = C ((1/2:ℚ)*2) := by rw [map_mul,C_ofNat]
  _ = 1 := by norm_num

theorem quadratic_row : row 2 (4:ℚ) 2 3 2=18+24*X+8*X^2 := by
 norm_num [row,term,Finset.sum_range_succ,C_ofNat]
 calc
  _ = (C (1/2:ℚ)*2)*(18+24*X+8*X^2) := by ring
  _ = _ := by rw [half_scale,one_mul]

theorem linear_row : row 1 (4:ℚ) 2 3 2=6+4*X := by
 norm_num [row,term,Finset.sum_range_succ,C_ofNat]
 calc
  _ = (C (1/2:ℚ)*2)*(6+4*X) := by ring
  _ = _ := by rw [half_scale,one_mul]

theorem constant_row : row 0 (4:ℚ) 2 3 2=2 := by
 norm_num [row,term,Finset.sum_range_succ,C_ofNat]
 calc
  _ = 2*(C (1/2:ℚ)*2) := by ring
  _ = _ := by rw [half_scale,mul_one]

theorem middle_coefficient : (row 2 (4:ℚ) 2 3 2).coeff 1=24 := by
 rw [quadratic_row]
 norm_num [coeff_add,coeff_mul,Finset.sum_range_succ]

theorem missing_binomial_rejected : (row 2 (4:ℚ) 2 3 2).coeff 1≠12 := by
 rw [middle_coefficient]; norm_num

theorem zero_row (k : ℕ) (a b s : ℚ) : row k 0 a b s=0 := by
 rw [row_equation]; simp

theorem empty_input (c : ℕ → ℚ) (a b s : ℚ) : outer c a b s 0=0 := rfl

theorem updated_slot : ((X^2:ℚ[X])+C 3*X^1).coeff 1=3 := by
 rw [slot_update]; norm_num

theorem other_slot : ((X^2:ℚ[X])+C 3*X^1).coeff 0=0 := by
 rw [slot_update]; norm_num

theorem final_slot_in_bounds : 5<6 ∧ 5≤5 ∧ 5-5+5=5 :=
 loop_index_bounds 6 5 5 (by decide) (by decide)

theorem binomial_5_2 : chooseLoop 5 2=10 := by decide

theorem max_product_attained : chooseLoop 5 2*(5-2)=30 := by decide

theorem degree6_outside_bound : 30<chooseLoop 6 3*(6-3) := by decide
end AffineLoopControls
