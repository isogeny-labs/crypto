import Composition
set_option autoImplicit false
namespace ForestReviewer
open GraphForest GraphFind GraphUnion SignedGraphAlgebra

def g : GraphFind.State ℚ 5 :=
 ⟨fun i=>if i=2 then 0 else if i=3 then 2 else if i=4 then 1 else i,
 fun i=>if i=2 then ⟨-1,5,1⟩ else if i=3 then ⟨1,-2,2⟩ else if i=4 then ⟨1,2,0⟩ else ident⟩
def s : GraphUnion.State ℚ 5:=⟨g,fun i=>if i=0 then some (1,1) else none⟩
def rank (i:Fin 5):Nat:=if i=2 then 1 else if i=3 then 2 else if i=4 then 1 else 0
def vals (i:Fin 5):ℚ:=if i=0 then 3 else if i=1 then 7 else if i=2 then 4 else if i=3 then 6 else 9

theorem initial_forest : Forest g rank := by intro i; fin_cases i <;> norm_num [g,rank]
theorem initial_valid : Valid s vals 2 := by
 constructor
 · intro i; fin_cases i <;> norm_num [Represents,s,g,vals,ident,eval]
 · intro i a b h; fin_cases i <;> simp [s] at h
   rcases h with ⟨rfl,rfl⟩; norm_num [vals]
def q2:Potential ℚ:=compose (g.pot 2) ident
def t2:GraphFind.State ℚ 5:=compress g 2 0 q2
def q3:Potential ℚ:=compose (t2.pot 3) q2
def t3:GraphFind.State ℚ 5:=compress t2 3 0 q3
def q4:Potential ℚ:=compose (t3.pot 4) ident
def t4:GraphFind.State ℚ 5:=compress t3 4 1 q4

theorem first_find : Run g 3 t3 0 q3 := by
 apply Run.step g t2 3 0 q2 (by decide)
 change Run g 2 t2 0 q2
 apply Run.step g g 2 0 ident (by decide)
 exact Run.root g 0 (by decide)
theorem second_find : Run t3 4 t4 1 q4 := by
 apply Run.step t3 t3 4 1 ident (by decide)
 exact Run.root t3 1 (by decide)
theorem both_nontrivial : g.parent 3≠3 ∧ t3.parent 4≠4 := by decide

theorem composed_both_orientations (f:Bool) :
 ∃r,Forest (branch (lifted s t4) 0 1 (reducedPotential q3 q4 1 1 15 0) f).graph r ∧
 Valid (branch (lifted s t4) 0 1 (reducedPotential q3 q4 1 1 15 0) f) vals 2 := by
 apply edge_after_two_finds s t3 t4 3 4 0 1 q3 q4 rank vals 1 1 15 0 2
 · exact initial_forest
 · exact initial_valid
 · exact first_find
 · exact second_find
 · decide
 · norm_num [q3,q2,t2,compress,compose,g,ident]
 · norm_num [q4,t3,t2,compress,compose,g,ident]
 · norm_num [vals]

theorem next_find_from_result (f:Bool) (x:Fin 5) :
 ∃t r p,Run (branch (lifted s t4) 0 1 (reducedPotential q3 q4 1 1 15 0) f).graph x t r p := by
 obtain ⟨r,hf,hv⟩:=composed_both_orientations f
 exact next_find_exists _ ⟨r,hf⟩ x

theorem first_root_survives_second : t4.parent 0=0 := by
 exact find_preserves_roots t3 t4 4 1 0 q4 second_find rank
  (run_preserves g t3 3 0 q3 first_find rank vals 2 initial_forest initial_valid.1).1
  (run_preserves g t3 3 0 q3 first_find rank vals 2 initial_forest initial_valid.1).2.2.1

theorem descendant_path : Reach g 3 0 := by
 apply Reach.step
 change Reach g 2 0
 apply Reach.step
 exact Reach.refl 0

theorem transferred_descendant : Reach (compress g 0 1 ident) 3 1 :=
 component_transferred g 0 1 3 ident descendant_path

theorem descendant_cycle_rejected (r:Fin 5→Nat) : ¬Forest (compress g 0 3 ident) r := by
 intro h
 have h0:=h 0;have h2:=h 2;have h3:=h 3
 norm_num [compress,g] at h0 h2 h3
 omega

theorem proposed_parent_not_root : g.parent 3≠3 := by decide

theorem root_components_disjoint : ¬Reach g 1 0 := by
 intro h
 have he:=root_reach g 1 0 (by decide) h
 exact (by decide : (0:Fin 5)≠1) he

def sameQ:Potential ℚ:=compose (t3.pot 2) ident
def sameT:GraphFind.State ℚ 5:=compress t3 2 0 sameQ

theorem same_component_second_find : Run t3 2 sameT 0 sameQ := by
 apply Run.step t3 t3 2 0 ident (by decide)
 exact Run.root t3 0 (by decide)

theorem same_root_branch_excluded : ¬((0:Fin 5)≠0) := by simp
end ForestReviewer
