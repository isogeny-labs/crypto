import Mathlib.LinearAlgebra.Span.Basic
set_option autoImplicit false
namespace TargetInformation
variable {K V W : Type*} [Field K] [AddCommGroup V] [Module K V]
 [AddCommGroup W] [Module K W]
def Solves (A : V →ₗ[K] W) (b c : W) (x : V) (k : K) : Prop := A x + k • b + c = 0

theorem shift_solution (A : V →ₗ[K] W) (b c : W) (x h : V) (k l : K)
 (hx : Solves A b c x k) (hh : A h = b) :
 Solves A b c (x + (k-l) • h) l := by
 unfold Solves at *
 rw [map_add,map_smul,hh]
 calc A x + (k-l) • b + l • b + c = A x + k • b + c := by rw [sub_smul, add_assoc (A x) (k • b-l • b) (l • b), sub_add_cancel]
      _ = 0 := hx

theorem in_range_all_targets (A : V →ₗ[K] W) (b c : W) (x : V) (k : K)
 (hx : Solves A b c x k) (hb : b ∈ LinearMap.range A) :
 ∀ l : K, ∃ y : V, Solves A b c y l := by
 obtain ⟨h,hh⟩ := hb
 intro l
 exact ⟨x+(k-l) • h, shift_solution A b c x h k l hx hh⟩

theorem unequal_solutions_range (A : V →ₗ[K] W) (b c : W) (x y : V) (k l : K)
 (hx : Solves A b c x k) (hy : Solves A b c y l) (hne : k ≠ l) :
 b ∈ LinearMap.range A := by
 have hdiff : A (x-y) = (l-k) • b := by
  rw [map_sub]
  unfold Solves at hx hy
  have he : A x + k • b = A y + l • b := add_right_cancel (hx.trans hy.symm)
  rw [sub_smul]
  exact (sub_eq_sub_iff_add_eq_add).2 (he.trans (add_comm _ _))
 have hn : l-k ≠ 0 := sub_ne_zero.mpr (Ne.symm hne)
 refine ⟨(l-k)⁻¹ • (x-y), ?_⟩
 rw [map_smul,hdiff,smul_smul,inv_mul_cancel₀ hn,one_smul]

theorem outside_range_unique (A : V →ₗ[K] W) (b c : W) (x y : V) (k l : K)
 (hb : b ∉ LinearMap.range A) (hx : Solves A b c x k) (hy : Solves A b c y l) : k=l := by
 by_contra hn
 exact hb (unequal_solutions_range A b c x y k l hx hy hn)

theorem unique_iff_outside_range (A : V →ₗ[K] W) (b c : W) (x : V) (k : K)
 (hx : Solves A b c x k) :
 (∀ y l, Solves A b c y l → l=k) ↔ b ∉ LinearMap.range A := by
 constructor
 · intro hu hb
   obtain ⟨y,hy⟩ := in_range_all_targets A b c x k hx hb (k+1)
   have h := hu y (k+1) hy
   exact one_ne_zero (add_left_cancel (show k+1=k+0 by simpa using h))
 · intro hb y l hy
   exact outside_range_unique A b c y x l k hb hy hx
end TargetInformation

namespace TargetInformation
variable {K V W : Type*} [Field K] [AddCommGroup V] [Module K V]
 [AddCommGroup W] [Module K W]
theorem zero_coefficient_all_targets (A : V →ₗ[K] W) (c : W) (x : V) (k : K)
 (hx : Solves A 0 c x k) : ∀ l, ∃ y, Solves A 0 c y l :=
 in_range_all_targets A 0 c x k hx (by exact ⟨0, map_zero A⟩)
theorem inconsistent_zero_system (c : W) (hc : c ≠ 0) :
 ¬ ∃ (x : V) (k : K), Solves (0 : V →ₗ[K] W) 0 c x k := by
 rintro ⟨x,k,h⟩
 exact hc (by simpa [Solves] using h)
theorem zero_map_unique (b c : W) (hb : b ≠ 0) (x : V) (k : K)
 (hx : Solves (0 : V →ₗ[K] W) b c x k) :
 ∀ y l, Solves (0 : V →ₗ[K] W) b c y l → l=k := by
 apply (unique_iff_outside_range (0 : V →ₗ[K] W) b c x k hx).2
 rintro ⟨z,hz⟩
 exact hb (by simpa using hz.symm)
end TargetInformation
