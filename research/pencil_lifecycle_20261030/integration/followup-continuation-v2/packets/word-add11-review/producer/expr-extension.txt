
inductive FieldExpr where
 | leaf : UInt64 → FieldExpr
 | mul : FieldExpr → FieldExpr → FieldExpr
 | add : FieldExpr → FieldExpr → FieldExpr
 | sub : FieldExpr → FieldExpr → FieldExpr
 | neg : FieldExpr → FieldExpr
 | pow5 : FieldExpr → FieldExpr

def FieldExpr.good (p : UInt64) : FieldExpr → Prop
 | .leaf a => a.toNat < p.toNat
 | .mul a b | .add a b | .sub a b => a.good p ∧ b.good p
 | .neg a | .pow5 a => a.good p

def FieldExpr.word (p : UInt64) : FieldExpr → UInt64
 | .leaf a => a
 | .mul a b => (a.word p*b.word p)%p
 | .add a b => wordAdd p (a.word p) (b.word p)
 | .sub a b => wordSub p (a.word p) (b.word p)
 | .neg a => wordNeg p (a.word p)
 | .pow5 a => wordPow5 p (a.word p)

def FieldExpr.field (p : UInt64) : FieldExpr → ZMod p.toNat
 | .leaf a => (a.toNat : ZMod p.toNat)
 | .mul a b => a.field p*b.field p
 | .add a b => a.field p+b.field p
 | .sub a b => a.field p-b.field p
 | .neg a => -a.field p
 | .pow5 a => (a.field p)^5

theorem FieldExpr.composition (p : UInt64) (hfive : 5 ≤ p.toNat) (hb : p.toNat < 2^31)
    (e : FieldExpr) (hg : e.good p) :
    (e.word p).toNat < p.toNat ∧ ((e.word p).toNat : ZMod p.toNat) = e.field p := by
  have hp : 0 < p.toNat := by omega
  induction e with
  | leaf a => exact ⟨hg,rfl⟩
  | mul a b ia ib =>
    have ha := ia hg.1
    have hc := ib hg.2
    exact ⟨uint64_mul_canonical p _ _ hp,by
      change (((a.word p*b.word p)%p).toNat : ZMod p.toNat) = _
      rw [uint64_mul_decode p _ _ hb ha.1 hc.1,ha.2,hc.2]; rfl⟩
  | add a b ia ib =>
    have ha := ia hg.1
    have hc := ib hg.2
    have hm := wordAdd_composition p _ _ hp hb ha.1 hc.1
    exact ⟨hm.1,by simpa only [FieldExpr.field,ha.2,hc.2] using hm.2⟩
  | sub a b ia ib =>
    have ha := ia hg.1
    have hc := ib hg.2
    have hm := wordSub_composition p _ _ hp hb ha.1 hc.1
    exact ⟨hm.1,by simpa only [FieldExpr.field,ha.2,hc.2] using hm.2⟩
  | neg a ia =>
    have ha := ia hg
    have hm := wordNeg_composition p _ hp ha.1
    exact ⟨hm.1,by simpa only [FieldExpr.field,ha.2] using hm.2⟩
  | pow5 a ia =>
    have ha := ia hg
    have hm := pow5_composition p _ hfive hb ha.1
    exact ⟨hm.1,by simpa only [FieldExpr.field,ha.2] using hm.2⟩

structure WordD1 where
 x : UInt64
 y : UInt64
 z : UInt64
 w : UInt64

def WordD1.canonical (p : UInt64) (a : WordD1) : Prop :=
 a.x.toNat < p.toNat ∧ a.y.toNat < p.toNat ∧ a.z.toNat < p.toNat ∧ a.w.toNat < p.toNat

def WordD1.decode (p : UInt64) (a : WordD1) : WeightedAdd11.Input (ZMod p.toNat) :=
 ⟨(a.x.toNat : ZMod p.toNat),(a.y.toNat : ZMod p.toNat),(a.z.toNat : ZMod p.toNat),(a.w.toNat : ZMod p.toNat)⟩

structure PayloadExpr where
 q : FieldExpr
 r : FieldExpr
 s : FieldExpr
 t : FieldExpr
 z : FieldExpr
 w : FieldExpr

def payloadExpr (a b : WordD1) : PayloadExpr :=
 let az2 := FieldExpr.mul (.leaf a.z) (.leaf a.z)
 let bz2 := FieldExpr.mul (.leaf b.z) (.leaf b.z)
 let axbz := FieldExpr.mul (.leaf a.x) bz2
 let bxaz := FieldExpr.mul (.leaf b.x) az2
 let ay := FieldExpr.mul (.leaf a.y) (.mul (.pow5 (.leaf b.z)) (.leaf b.w))
 let yb := FieldExpr.mul (.leaf b.y) (.mul (.pow5 (.leaf a.z)) (.leaf a.w))
 ⟨.neg (.add axbz bxaz),.mul axbz bxaz,.sub ay yb,
   .sub (.mul axbz yb) (.mul bxaz ay),.mul (.leaf a.z) (.leaf b.z),
   .mul (.sub axbz bxaz) (.mul (.leaf a.w) (.leaf b.w))⟩

-- Literal source operations; returned payload only, branch dispatch below.
def add11WordPayload (p : UInt64) (a b : WordD1) : WeightedWord :=
 let az2 := (a.z*a.z)%p
 let bz2 := (b.z*b.z)%p
 let axbz := (a.x*bz2)%p
 let bxaz := (b.x*az2)%p
 let ay := (a.y*((wordPow5 p b.z*b.w)%p))%p
 let yb := (b.y*((wordPow5 p a.z*a.w)%p))%p
 ⟨wordNeg p (wordAdd p axbz bxaz), (axbz*bxaz)%p,
  wordSub p ay yb,wordSub p ((axbz*yb)%p) ((bxaz*ay)%p),
  (a.z*b.z)%p,((wordSub p axbz bxaz)*((a.w*b.w)%p))%p⟩

theorem payload_decode (p : UInt64) [Fact p.toNat.Prime]
    (hfive : 5 ≤ p.toNat) (hb : p.toNat < 2^31) (a b : WordD1)
    (ha : a.canonical p) (hc : b.canonical p) :
    let c := add11WordPayload p a b
    let f := WeightedAdd11.output (a.decode p) (b.decode p)
    c.canonical p ∧
    (c.q.toNat : ZMod p.toNat)=f.q ∧ (c.r.toNat : ZMod p.toNat)=f.r ∧
    (c.s.toNat : ZMod p.toNat)=f.s ∧ (c.t.toNat : ZMod p.toNat)=f.t ∧
    (c.z.toNat : ZMod p.toNat)=f.z ∧ (c.w.toNat : ZMod p.toNat)=f.w := by
  have good : (payloadExpr a b).q.good p ∧ (payloadExpr a b).r.good p ∧
      (payloadExpr a b).s.good p ∧ (payloadExpr a b).t.good p ∧
      (payloadExpr a b).z.good p ∧ (payloadExpr a b).w.good p := by
    rcases ha with ⟨hax,hay,haz,haw⟩
    rcases hc with ⟨hbx,hby,hbz,hbw⟩
    simp [payloadExpr,FieldExpr.good,hax,hay,haz,haw,hbx,hby,hbz,hbw]
  have hq := FieldExpr.composition p hfive hb _ good.1
  have hr := FieldExpr.composition p hfive hb _ good.2.1
  have hs := FieldExpr.composition p hfive hb _ good.2.2.1
  have ht := FieldExpr.composition p hfive hb _ good.2.2.2.1
  have hz := FieldExpr.composition p hfive hb _ good.2.2.2.2.1
  have hw := FieldExpr.composition p hfive hb _ good.2.2.2.2.2
  refine ⟨⟨hq.1,hr.1,hs.1,ht.1,hz.1,hw.1⟩,?_,?_,?_,?_,?_,?_⟩
  all_goals
    first | change (((payloadExpr a b).q.word p).toNat : ZMod p.toNat) = _; rw [hq.2]
          | change (((payloadExpr a b).r.word p).toNat : ZMod p.toNat) = _; rw [hr.2]
          | change (((payloadExpr a b).s.word p).toNat : ZMod p.toNat) = _; rw [hs.2]
          | change (((payloadExpr a b).t.word p).toNat : ZMod p.toNat) = _; rw [ht.2]
          | change (((payloadExpr a b).z.word p).toNat : ZMod p.toNat) = _; rw [hz.2]
          | change (((payloadExpr a b).w.word p).toNat : ZMod p.toNat) = _; rw [hw.2]
    rfl
