use crate::fixtures::Fixture;
use crypto_lib::cryptanalysis::ecbench::canonical::sha256_hex;
use serde_json::{json,Value};
use std::{path::{Path,PathBuf},process::{Command,Output}};
fn binary()->PathBuf{PathBuf::from(std::env::var("RETAINED_EXPORT_CLI").unwrap())}
fn request(f:&Fixture)->Value{let n=&f.files;json!({"schema":"ecbench.retained-export-cli-request/v1","files":{"inventory_request":n.inventory_request,"inventory_sha256":n.inventory_sha256,"route":n.route,"host":n.host,"session":n.session,"preparation":n.preparation,"launch":n.launch,"contexts":n.contexts,"journal":n.journal,"owner":n.owner,"config":n.config,"custody":n.custody},"expected":{"fresh":f.expected.fresh,"controller_custody_sha256":f.expected.controller_custody_sha256}})}
fn put(f:&Fixture,b:&[u8])->PathBuf{let p=f.dir.join("cli-request.json");std::fs::write(&p,b).unwrap();p}
fn invoke(p:&Path,h:&str,bh:&str,out:&Path)->Output{Command::new(binary()).args(["--request",p.to_str().unwrap(),"--request-sha256",h,"--binary-sha256",bh,"--out",out.to_str().unwrap()]).output().unwrap()}
fn bpin()->String{sha256_hex(&std::fs::read(binary()).unwrap())}
fn error(o:&Output)->Value{assert!(!o.status.success());assert!(o.stdout.is_empty());let v:Value=serde_json::from_slice(&o.stderr).unwrap();assert_eq!(v["status"],"failed");assert_eq!(v["retry_performed"],false);v}
fn run(f:&Fixture,v:&Value,out:&Path)->Output{let b=serde_json::to_vec(v).unwrap();invoke(&put(f,&b),&sha256_hex(&b),&bpin(),out)}
#[test] fn cli_positive_request_binary_and_unknown_metrics(){let f=Fixture::new();let v=request(&f);let out=f.dir.join("cli-out");let o=run(&f,&v,&out);assert!(o.status.success(),"{}",String::from_utf8_lossy(&o.stderr));let r:Value=serde_json::from_slice(&o.stdout).unwrap();assert_eq!(r["request_sha256"],sha256_hex(&serde_json::to_vec(&v).unwrap()));assert_eq!(r["executable_sha256"],bpin());assert_eq!(r["owner_authentication_verified"],false);assert_eq!(r["scientific_execution_admitted"],false);assert_eq!(r["report"]["scientific_execution_admitted"],false);for k in ["worker_allocation_ns","total_active_worker_ns","exact_kernel_exit_ns"]{assert!(r[k].is_null());assert!(r["report"][k].is_null())}assert_eq!(r["report"]["structural_report"]["result"]["software_bindings_complete"],true);assert_eq!(r["export_report_sha256"],sha256_hex(&std::fs::read(out.join("export-report.json")).unwrap()));assert_eq!(std::fs::read(out.join("cli-request.json")).unwrap(),serde_json::to_vec(&v).unwrap());assert_eq!(serde_json::from_slice::<Value>(&std::fs::read(out.join("cli-invocation.json")).unwrap()).unwrap(),r);}
#[test] fn cli_request_hash_rejected(){let f=Fixture::new();let b=serde_json::to_vec(&request(&f)).unwrap();let out=f.dir.join("out");let r=error(&invoke(&put(&f,&b),&"0".repeat(64),&bpin(),&out));assert!(r["error"].as_str().unwrap().contains("request hash"));assert!(!out.exists());}
#[test] fn cli_executable_hash_rejected(){let f=Fixture::new();let b=serde_json::to_vec(&request(&f)).unwrap();let out=f.dir.join("out");let r=error(&invoke(&put(&f,&b),&sha256_hex(&b),&"0".repeat(64),&out));assert!(r["error"].as_str().unwrap().contains("executable hash"));assert!(!out.exists());}
#[test] fn cli_unknown_and_duplicate_fields_rejected(){let f=Fixture::new();let mut v=request(&f);v["authenticated"]=json!(true);let out=f.dir.join("out");error(&run(&f,&v,&out));assert!(!out.exists());for where_ in ["files","expected","fresh"]{let mut n=request(&f);if where_=="fresh"{n["expected"]["fresh"]["unknown"]=json!(1)}else{n[where_]["unknown"]=json!(1)}error(&run(&f,&n,&out));assert!(!out.exists());}let o=Command::new(binary()).args(["--request","/unused","--request","/duplicate","--binary-sha256",&bpin(),"--out",out.to_str().unwrap()]).output().unwrap();assert!(error(&o)["error"].as_str().unwrap().contains("duplicate"));assert!(!out.exists());let mut b=serde_json::to_vec(&request(&f)).unwrap();b.pop();b.extend_from_slice(b",\"schema\":\"ecbench.retained-export-cli-request/v1\"}");error(&invoke(&put(&f,&b),&sha256_hex(&b),&bpin(),&out));assert!(!out.exists());}
#[test] fn cli_external_pin_mismatch_rejected(){let f=Fixture::new();let mut v=request(&f);v["expected"]["fresh"]["pod_uid"]=json!("different");let out=f.dir.join("out");error(&run(&f,&v,&out));assert!(!out.exists());}
#[test] fn cli_output_absolute_and_fresh_required(){let f=Fixture::new();let v=request(&f);error(&run(&f,&v,Path::new("relative-output")));let out=f.dir.join("existing");std::fs::create_dir(&out).unwrap();std::fs::write(out.join("sentinel"),b"unchanged").unwrap();error(&run(&f,&v,&out));assert_eq!(std::fs::read(out.join("sentinel")).unwrap(),b"unchanged");}
#[test] fn cli_request_symlink_and_byte_bound_rejected(){let f=Fixture::new();let b=serde_json::to_vec(&request(&f)).unwrap();let p=put(&f,&b);let link=f.dir.join("link");std::os::unix::fs::symlink(&p,&link).unwrap();let out=f.dir.join("out");error(&invoke(&link,&sha256_hex(&b),&bpin(),&out));assert!(!out.exists());let large=vec![b' ';1024*1024+1];error(&invoke(&put(&f,&large),&sha256_hex(&large),&bpin(),&out));assert!(!out.exists());}
#[test] fn cli_failed_native_validation_retains_partial_output(){let f=Fixture::new();std::fs::write(&f.files.contexts[0],b"{}").unwrap();let v=request(&f);let out=f.dir.join("partial");let e=error(&run(&f,&v,&out));assert_eq!(e["output_may_contain_partial_files"],true);assert!(out.join("actual-request.json").is_file());assert!(!out.join("cli-invocation.json").exists());let before=std::fs::read(out.join("actual-request.json")).unwrap();error(&run(&f,&v,&out));assert_eq!(std::fs::read(out.join("actual-request.json")).unwrap(),before);}
