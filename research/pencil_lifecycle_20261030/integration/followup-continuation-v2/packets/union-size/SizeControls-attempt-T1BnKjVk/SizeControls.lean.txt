import SizeComposition
set_option autoImplicit false
namespace SizeControls
open UnionSize

def sizes : Sizes 3 := ⟨#[2,1,3],rfl⟩
def tied : Sizes 2 := ⟨#[1,1],rfl⟩
theorem tie_chooses_first : child tied 0 1=0 ∧ parent tied 0 1=1 := by decide
theorem smaller_first : child sizes 0 2=0 ∧ parent sizes 0 2=2 := by decide
theorem smaller_second : child sizes 2 0=0 ∧ parent sizes 2 0=2 := by decide
theorem parent_addition : sizeAt (update sizes 0 2) 2=5 := by decide
theorem child_stale_retained : sizeAt (update sizes 0 2) 0=2 := by decide
theorem unrelated_unchanged : sizeAt (update sizes 0 2) 1=1 := by decide
-- Without a strict representability bound, modular machine addition wraps.
theorem overflow_boundary : ((USize.ofNat (USize.size-1))+1).toNat=0 := by
 rw [USize.toNat_add]
 simp [USize.toNat_ofNat]

end SizeControls
