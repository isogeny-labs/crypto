 ([1] : List UInt64).length≤3 ∧ ([0] : List UInt64).length≤2 ∧
 ¬ (([0] : List UInt64).length < ([1] : List UInt64).length) := by decide

 MonicZeroTest.Normal ([1] : List UInt64) ∧ MonicZeroTest.Normal ([] : List UInt64) ∧
 ([] : List UInt64).length < ([1] : List UInt64).length := by
 simp [MonicZeroTest.Normal]
