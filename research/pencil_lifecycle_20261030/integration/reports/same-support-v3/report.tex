\[
\bar q_i=q_i/z_i^2,\quad\bar r_i=r_i/z_i^4,\quad
\bar s_i=s_i/(z_i^3w_i),\quad\bar t_i=t_i/(z_i^5w_i).
\]

\begin{align*}
Q_{c1}&=q_1z_2^2,& Q_{c2}&=q_2z_1^2,\\
R_{c1}&=r_1z_2^4,& R_{c2}&=r_2z_1^4,\\

T_{c1}&=t_1z_2^5w_2,& T_{c2}&=t_2z_1^5w_1.
\end{align*}
