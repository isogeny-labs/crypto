import InvariantBranches
import Mathlib.Data.ZMod.Basic
-- Dropping a nonzero constructor weight destroys coefficient recovery.
example : ((1 : ℚ) * 0^2) / 0^2 ≠ 1 := by norm_num
-- Zero w similarly destroys the ordinate coefficient.
example : ((1 : ℚ) * (1^3*0)) / (1^3*0) ≠ 1 := by norm_num
-- In a nonreduced residue ring the zero-dq/B guard need not mean equal support.
example : (0 : ZMod 4)-(0 : ZMod 4) = 0 ∧
    ((0 : ZMod 4)-0)*(0*2-0*0)-(2-0)^2 = 0 ∧ (2 : ZMod 4) ≠ 0 := by decide
-- Nonvacuous distinct-support B=0 branch with nonzero dq over a field.
example : ((1 : ℚ)-0)*(0*0-1*0)-(0-0)^2 = 0 ∧ (1 : ℚ)-0 ≠ 0 := by norm_num
