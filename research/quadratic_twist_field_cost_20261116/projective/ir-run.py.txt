run('cpp-build',['g++','-std=c++20','-O2','-g','-Wall','-Wextra','-I/workspace/twist-native-tooling/include','-I'+str(tool/'include'),scope/'ir.cpp','-lcrypto','-lz','-o',out/'ir'])

    run('audit-'+tag,[auditor,dest/'cases.json',prior,out/('audit-'+tag+'.json')])

    dest=out/('false-'+mode);dest.mkdir();j=json.loads((out/'A/cases.json').read_text())
    for r in j['records']:
        old=Path(r['counter_path']);new=dest/old.name;os.link(old,new);r['counter_path']=str(new)
    if mode=='cost':j['records'][0]['result']['counts'][0]=0
    elif mode=='missing':j['records'].pop(0)
    elif mode=='source':j['field']=59
    else:
        p=Path(j['records'][0]['counter_path']);old=p.read_text();lines=old.splitlines();line=next(i for i,s in enumerate(lines)if s.startswith('summary:'));lines[line]='summary: 1';new='\n'.join(lines)+'\n';assert new!=old;p.unlink();p.write_text(new)
    data=(json.dumps(j,separators=(',',':'))+'\n').encode();(dest/'cases.json').write_bytes(data);receipt=out/('false-'+mode+'-receipt.json');run('reject-'+mode,[auditor,dest/'cases.json',prior,receipt],101);assert not receipt.exists()
    controls.append({'mode':mode,'changed_payload_sha256':hashlib.sha256(data).hexdigest(),'rejected':True,'receipt_exists':False})
(out/'controls.json').write_text(json.dumps(controls,indent=2)+'\n')
