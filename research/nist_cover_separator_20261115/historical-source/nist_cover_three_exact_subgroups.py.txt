#!/usr/bin/env python3
"""Exact subgroup counts using actual elliptic and Prym Weil pairings."""
import concurrent.futures
import json
from pathlib import Path
import time
import numpy as np
from nist_three_weil_forms_local import prym_form,elliptic_module
from nist_three_isotropic_local_catalog import PAIRS,planes,stable,isotropic_counts

ROOT=Path(__file__).parent
GRASS=None


def job(path):
    global GRASS
    if GRASS is None:GRASS={k:planes(k) for k in [1,2,3]}
    dest=ROOT/'nist_cover_three_exact_subgroups';dest.mkdir(exist_ok=True);out=dest/path.name
    if out.exists():return json.loads(out.read_text())
    mat=json.loads(path.read_text());f=mat['family'];c=mat['candidate'];d=mat['cover_parameter_d']
    try:
        r=next(r for r in json.loads((ROOT/'nist_selection_audit.json').read_text())['rows'] if r['family']==f and r['candidate']==c)
        pair=prym_form(mat);ell=elliptic_module(r);E=np.array(ell['Frobenius_matrix_mod3'],dtype=np.int16);M=np.array(mat['Prym_Frobenius_matrix_mod3'],dtype=np.int16)
        A=np.zeros((6,6),dtype=np.int16);A[:2,:2]=E;A[2:,2:]=M
        scale=ell['canonical_Weil_matrix_mod3'][0][1];J=np.array(pair['canonical_Prym_Weil_matrix_mod3'],dtype=np.int16)*scale%3
        fs=np.array([[J[i,j] for i,j in PAIRS]],dtype=np.int16);counts=[]
        for k in [1,2,3]:
            B,piv=GRASS[k];BS=stable(B,piv,A);counts.append(int(isotropic_counts(BS,fs)[0]))
        z={'family':f,'candidate':c,'cover_parameter_d':d,'complete':True,
           'rational_isotropic_order_three_subgroup_count':counts[0],
           'rational_isotropic_order_nine_subgroup_count':counts[1],
           'rational_maximal_isotropic_three_kernel_count':counts[2],
           'Prym_characteristic_coefficients_mod3':mat['Prym_characteristic_coefficients_mod3'],
           'actual_Weil_pairings_computed':True,'Prym_pairing':pair,'elliptic_module_file':str(ROOT/'nist_elliptic_three_weil_modules'/f'{f}_{c}.json'),
           'primitive_alignment':'For p=1 mod3, both factors use the same Fp third root. For p=2 mod3 E[3] is an irreducible two-dimensional module whose centralizer has every pairing multiplier, so subgroup counts are independent of the relative third-root sign.'}
    except Exception as e:z={'family':f,'candidate':c,'cover_parameter_d':d,'complete':False,'error':repr(e)}
    out.write_text(json.dumps(z,indent=2)+'\n');return z


def main():
    seen=set();results=[];matrixdir=ROOT/'nist_cover_prym_three_local_matrix';start=time.monotonic()
    with concurrent.futures.ProcessPoolExecutor(max_workers=2) as pool:
        while len(seen)<306:
            paths=[p for p in matrixdir.glob('*.json') if p.name not in seen and json.loads(p.read_text()).get('complete')]
            if not paths:
                if time.monotonic()-start>900:break
                time.sleep(5);continue
            for path,z in zip(paths,pool.map(job,paths)):
                seen.add(path.name);results.append(z)
                print('SUBGROUPS',len(results),306,z['family'],z['candidate'],z['cover_parameter_d'],z['complete'],z.get('rational_isotropic_order_nine_subgroup_count'),z.get('error'),flush=True)
                (ROOT/'nist_cover_three_exact_subgroups_grid.json').write_text(json.dumps({'scope':'The common d=0..15 grid plus P384/P521 at d=1.',
                    'method':'Compute canonical order-three Weil pairings from the explicit cubic tangent functions for the Prym and tangent lines for E; enumerate all Frobenius-stable isotropic subspaces of dimensions 1,2,3.',
                    'expected_count':306,'completed_count':len(results),'rows':results},indent=2)+'\n')


if __name__=='__main__':main()
