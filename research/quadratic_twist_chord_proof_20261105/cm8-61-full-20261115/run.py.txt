    if phase=='build':
        run('cpp',['g++','-std=c++20','-O2','-Wall','-Wextra','-I/workspace/twist-native-tooling/include',str(scope/'producer.cpp'),'-lcrypto','-o',str(out/'producer')])
        run('rust',['/workspace/twist-native-tooling/cargo/bin/cargo','+1.85.1','build','--offline','--locked','--release','--manifest-path',str(scope/'auditor/Cargo.toml')])

    for tag in ['A','B']:

    aa=out/'casesA';bb=out/'casesB';assert(aa/'index.json').read_bytes()==(bb/'index.json').read_bytes()

    for row in index['cases']:assert(aa/row['path']).read_bytes()==(bb/row['path']).read_bytes()

        if name=='index.json':idx.write_bytes(altered.read_bytes())
        else:
            (false/name).unlink();(false/name).write_bytes(altered.read_bytes());changed=json.loads((aa/'index.json').read_text());record=next(x for x in changed['cases']if x['path']==name);record['sha256']=hashlib.sha256(altered.read_bytes()).hexdigest();idx.write_text(json.dumps(changed,separators=(',',':'))+'\n')

        controls.append(dict(mode=mode,altered_path=name,payload_sha256=hashlib.sha256(altered.read_bytes()).hexdigest(),index_sha256=hashlib.sha256(idx.read_bytes()).hexdigest(),receipt=False))
