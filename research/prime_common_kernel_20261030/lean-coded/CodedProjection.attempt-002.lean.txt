module

public import Mathlib.Data.Matrix.Diagonal
public import Mathlib.Algebra.BigOperators.Group.Finset.Basic
public import Mathlib.Tactic

public section

namespace CodedProjection

variable {G H R I : Type*}
variable [AddCommGroup G] [AddCommGroup H] [DecidableEq G] [DecidableEq H]
variable [Ring R]

def codeIndicator (σ : G →+ H) (x : G) : R := if σ x = 0 then 1 else 0

def translation (e : G) : Matrix G G R := fun y x => if y = x + e then 1 else 0

def projected (σ : G →+ H) (e : G) : Matrix G G R := fun y x =>
  codeIndicator (R := R) σ y * translation (R := R) e y x * codeIndicator (R := R) σ x

theorem projected_is_matrix_product [Fintype G] (σ : G →+ H) (e : G) :
    projected (R := R) σ e = Matrix.diagonal (codeIndicator (R := R) σ) * translation (R := R) e *
      Matrix.diagonal (codeIndicator (R := R) σ) := by
  ext y x
  simp [projected, Matrix.diagonal_mul, Matrix.mul_diagonal]

theorem nonzero_syndrome_annihilates (σ : G →+ H) (e : G) (he : σ e ≠ 0) :
    projected (R := R) σ e = 0 := by
  ext y x
  by_cases hx : σ x = 0
  · by_cases hxy : y = x + e
    · have hy : σ y ≠ 0 := by simpa [hxy, map_add, hx] using he
      simp [projected, codeIndicator, hy]
    · simp [projected, translation, hxy]
  · simp [projected, codeIndicator, hx]

theorem zero_syndrome_preserves (σ : G →+ H) (e : G) (he : σ e = 0) :
    projected (R := R) σ e = fun y x => translation (R := R) e y x * codeIndicator (R := R) σ x := by
  ext y x
  by_cases hx : σ x = 0
  · by_cases hxy : y = x + e
    · have hy : σ y = 0 := by simp [hxy, map_add, hx, he]
      simp [projected, codeIndicator, hx, hy]
    · simp [projected, translation, hxy]
  · simp [projected, codeIndicator, hx]

theorem mismatched_pair_annihilates [DecidableEq I]
    (σ : G →+ H) (s : I → H) (hs : Function.Injective s)
    (e₁ e₂ : I → G) (h₁ : ∀ j, σ (e₁ j) = s j)
    (h₂ : ∀ j, σ (e₂ j) = -s j) (j k : I) (hjk : j ≠ k) :
    projected (R := R) σ (e₁ j + e₂ k) = 0 := by
  apply nonzero_syndrome_annihilates (R := R)
  intro h
  have h' : s j = s k := by
    have : s j - s k = 0 := by simpa [map_add, h₁, h₂, sub_eq_add_neg] using h
    exact sub_eq_zero.mp this
  exact hjk (hs h')

theorem matched_pair_preserves
    (σ : G →+ H) (s : I → H) (e₁ e₂ : I → G)
    (h₁ : ∀ j, σ (e₁ j) = s j) (h₂ : ∀ j, σ (e₂ j) = -s j) (j : I) :
    projected (R := R) σ (e₁ j + e₂ j) =
      fun y x => translation (e₁ j + e₂ j) y x * codeIndicator (R := R) σ x := by
  apply zero_syndrome_preserves (R := R)
  simp [map_add, h₁, h₂]

theorem weighted_pair_diagonal [Fintype I] [DecidableEq I]
    (σ : G →+ H) (s : I → H) (hs : Function.Injective s)
    (e₁ e₂ : I → G) (h₁ : ∀ j, σ (e₁ j) = s j)
    (h₂ : ∀ j, σ (e₂ j) = -s j) (f : I → R) (y x : G) :
    (∑ j, ∑ k, f j * f k * projected (R := R) σ (e₁ j + e₂ k) y x) =
      ∑ j, f j * f j * projected (R := R) σ (e₁ j + e₂ j) y x := by
  apply Finset.sum_congr rfl
  intro j hj
  apply Finset.sum_eq_single j
  · intro k hk hkj
    have hz := mismatched_pair_annihilates (R := R) σ s hs e₁ e₂ h₁ h₂ j k (Ne.symm hkj)
    simp [hz]
  · simp
