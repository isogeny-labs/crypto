using Q=mpq_class;
using Poly=std::vector<Q>;
void need(bool x){if(!x)throw std::runtime_error("exact check failed");}
Poly times(Poly p,int c,int a){Poly r(p.size()+1);for(size_t k=0;k<p.size();++k){r[k]+=c*p[k];r[k+1]+=a*p[k];}return r;}
Q cp(int l){Q v=1;for(int t=-l;t<l;++t)if(t)v*=-t;return v;}
Poly chart(int l,int t,int delta,bool full){Poly p{Q(1)/cp(l)};for(int s=-l;s<l;++s)if(full||s!=delta)p=times(p,t-s,2*l);return p;}
int val(Q q){if(q==0)return 100000;mpz_class n=q.get_num();if(n<0)n=-n;return int(mpz_scan1(n.get_mpz_t(),0))-int(mpz_scan1(q.get_den().get_mpz_t(),0));}
Q eval(const Poly&p,int z){Q v=0;for(auto it=p.rbegin();it!=p.rend();++it)v=v*z+*it;return v;}
unsigned mod(Q q,unsigned m){q.canonicalize();mpz_class inv,den=q.get_den(),mm=m;need(mpz_invert(inv.get_mpz_t(),den.get_mpz_t(),mm.get_mpz_t())!=0);mpz_class v=q.get_num()*inv;return unsigned(mpz_fdiv_ui(v.get_mpz_t(),m));}
unsigned inv(int d,unsigned m){need(d%2!=0);return mod(Q(1)/Q(d),m);}
unsigned mul(unsigned a,unsigned b,unsigned m){return unsigned((uint64_t(a)*b)%m);}
unsigned power(unsigned a,unsigned b,unsigned m){unsigned r=1;while(b){if(b&1)r=mul(r,a,m);a=mul(a,a,m);b>>=1;}return r;}
