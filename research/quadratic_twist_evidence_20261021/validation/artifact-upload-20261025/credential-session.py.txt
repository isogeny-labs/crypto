import base64
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import termios

root = Path('/workspace/crypto-next')
out = Path('/workspace/twist-artifacts/upload-20261025')
proposal = root / 'research/quadratic_twist_evidence_20261021/validation/s3-yield-20261024/ARTIFACTS_UPLOAD_PENDING.json'
if hashlib.sha256(proposal.read_bytes()).hexdigest() != 'ef0f5f60214310f6f4b62747a9d270c8371fec82b89fb610564d8317a3b9d747':
    raise SystemExit('Approved proposal changed; stopped.')
data = json.loads(proposal.read_text())
payloads = data['payloads']
for entry in payloads:
    p = root / entry['path']
    if len(p.read_bytes()) != entry['bytes'] or hashlib.sha256(p.read_bytes()).hexdigest() != entry['sha256']:
        raise SystemExit('Approved payload changed; stopped.')
tool = root / 'research/quadratic_twist_evidence_20261021/validation/s3-yield-20261024/artifacts.py.txt'
if hashlib.sha256(tool.read_bytes()).hexdigest() != '0054cdc0de842c0c9efc74f1d4fed4951b0e0c0fdc81a7f9308d8b6b8a4b8fe9':
    raise SystemExit('Pinned repository tool changed; stopped.')
old = termios.tcgetattr(sys.stdin)
hidden = termios.tcgetattr(sys.stdin)
hidden[3] &= ~termios.ECHO
termios.tcsetattr(sys.stdin, termios.TCSANOW, hidden)
try:
    print('Ready for one-use credential input; terminal echo disabled.', flush=True)
    supplied = json.loads(sys.stdin.readline())
finally:
    termios.tcsetattr(sys.stdin, termios.TCSANOW, old)
env = os.environ.copy()
env.update({
    'AWS_ACCESS_KEY_ID': supplied['access_key_id'],
    'AWS_SECRET_ACCESS_KEY': supplied['secret_access_key'],
    'AWS_DEFAULT_REGION': 'us-east-2',
    'AWS_EC2_METADATA_DISABLED': 'true',
    'ARTIFACTS_REMOTE': 's3://setec-astronomy-artifacts/store',
    'ARTIFACTS_CACHE': '/workspace/twist-tooling/cache/artifacts',
    'ARTIFACTS_AWS': '/workspace/twist-tooling/awscli-venv/bin/aws',
})

def scrub(value):
    for secret in supplied.values():
        value = value.replace(secret, '[REDACTED]')
    return value

def run(command, label, cap=45):
    try:
        p = subprocess.run(command, cwd=root, env=env, capture_output=True, text=True, timeout=cap)
        log = scrub(p.stdout + p.stderr)
        code = p.returncode
    except subprocess.TimeoutExpired as e:
        def txt(s):
            return s.decode(errors='replace') if isinstance(s, bytes) else (s or '')
        log = scrub(txt(e.stdout) + txt(e.stderr)) + '\nCommand reached its configured timeout.\n'
        code = 124
    (out / (label + '.log')).write_text(log)
    (out / (label + '.exit')).write_text(str(code) + '\n')
    return code, log

code, log = run([sys.executable, str(tool), 'push', *[e['path'] for e in payloads], '--jobs', '1'], 'upload')
print(log, end='', flush=True)
if code:
    print('Upload failed with exit code ' + str(code) + '; no alternate upload attempted.', flush=True)
    raise SystemExit(code)
code, log = run([sys.executable, str(tool), 'verify', *[e['path'] for e in payloads], '--remote'], 'verify')
print(log, end='', flush=True)
if code:
    raise SystemExit(code)
verified = []
for index, entry in enumerate(payloads):
    digest = entry['sha256']
    downloaded = out / ('download-' + str(index))
    command = [env['ARTIFACTS_AWS'], 's3api', 'get-object', '--bucket', 'setec-astronomy-artifacts', '--key', 'store/sha256/' + digest[:2] + '/' + digest, '--checksum-mode', 'ENABLED', str(downloaded), '--output', 'json']
    code, log = run(command, 'get-' + str(index))
    if code:
        print(log, end='', flush=True)
        raise SystemExit(code)
    remote = downloaded.read_bytes()
    response = json.loads(log)
    checksum = response.get('ChecksumSHA256')
    want_checksum = base64.b64encode(bytes.fromhex(digest)).decode()
    if len(remote) != entry['bytes'] or hashlib.sha256(remote).hexdigest() != digest or checksum != want_checksum:
        raise SystemExit('Downloaded object or S3 checksum differs from approved payload; stopped.')
    verified.append({'path': entry['path'], 'bytes': len(remote), 'sha256': digest, 's3_checksum_sha256': checksum, 'download_hash_matches': True})
    print('Downloaded and verified: ' + entry['path'], flush=True)
(out / 'remote-verification.json').write_text(json.dumps({'files': verified, 'problems': 0}, indent=2) + '\n')
env.pop('AWS_ACCESS_KEY_ID', None)
env.pop('AWS_SECRET_ACCESS_KEY', None)
supplied.clear()
print('All four approved remote payloads verified; credentials were not persisted.', flush=True)
