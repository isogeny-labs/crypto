/-! Addition compatibility for the explicit short-model twist map.  This
module leaves the previous point-map source and certificates unchanged. -/
namespace TwistGroupMap

section Scaling

variable {F : Type*} [Field F] [DecidableEq F]

theorem slope_scale (a b d u x₁ x₂ y₁ y₂ : F) (hd : d ≠ 0)
    (hu : u ^ 2 = d) :
    (short a b).slope (x₁ / d) (x₂ / d) (y₁ / (d * u)) (y₂ / (d * u)) =
      (twist a b d).slope x₁ x₂ y₁ y₂ / u := by
  have hu0 : u ≠ 0 := by
    intro hz
    exact hd (by simpa [hz] using hu.symm)
  have hdu : d * u ≠ 0 := mul_ne_zero hd hu0
  have hx : (x₁ / d = x₂ / d) ↔ x₁ = x₂ := div_left_inj' hd
  have hy : (y₁ / (d * u) = -(y₂ / (d * u))) ↔ y₁ = -y₂ := by
    rw [← neg_div]
    exact div_left_inj' hdu
  simp only [slope, negY, short, twist, zero_mul, mul_zero, add_zero, sub_zero, hx, hy]
  split_ifs with hxx hyy
  · simp
  · have hn : 3 * (x₁ / d) ^ 2 + a = (3 * x₁ ^ 2 + d ^ 2 * a) / d ^ 2 := by
      field_simp
    have he : y₁ / (d * u) - -(y₁ / (d * u)) = (y₁ - -y₁) / (d * u) := by ring
    rw [hn, he]
    calc
      _ = ((3 * x₁ ^ 2 + d ^ 2 * a) / (y₁ - -y₁)) * ((d * u) / d ^ 2) := by
        simp only [div_eq_mul_inv, mul_inv_rev, inv_inv]
        ring
      _ = _ := by
        have hc : (d * u) / d ^ 2 = 1 / u := by
          field_simp [hd, hu0]
          all_goals rw [hu]
        rw [hc, mul_one_div]
  · simp only [div_eq_mul_inv, mul_inv_rev]
    field_simp

omit [DecidableEq F] in
theorem addX_scale (a b d u x₁ x₂ ℓ : F)
    (hu : u ^ 2 = d) :
    (short a b).addX (x₁ / d) (x₂ / d) (ℓ / u) =
      (twist a b d).addX x₁ x₂ ℓ / d := by
  simp only [addX, short, twist, zero_mul, add_zero, sub_zero, div_pow, hu]
  ring

omit [DecidableEq F] in
theorem addY_scale (a b d u x₁ x₂ y₁ ℓ : F)
    (hu : u ^ 2 = d) :
    (short a b).addY (x₁ / d) (x₂ / d) (y₁ / (d * u)) (ℓ / u) =
      (twist a b d).addY x₁ x₂ y₁ ℓ / (d * u) := by
  simp only [addY, negY, negAddY, short, twist, zero_mul, sub_zero,
    addX, add_zero, div_pow, hu]
  ring

end Scaling

/-- Cleared tangent numerator factor; the full slope theorem also handles division. -/
theorem cleared_tangent_identity {F : Type*} [CommRing F] (a d u x : F) :
    (3 * x ^ 2 + d ^ 2 * a) * u ^ 2 - d * (3 * x ^ 2 + d ^ 2 * a) =
      (u ^ 2 - d) * (3 * x ^ 2 + d ^ 2 * a) := by ring

/-- Cleared addition x-coordinate identity; u^2=d kills the error term. -/
theorem cleared_addX_identity {F : Type*} [CommRing F] (d u x₁ x₂ ℓ : F) :
    d * ℓ ^ 2 - u ^ 2 * x₁ - u ^ 2 * x₂ - u ^ 2 * (ℓ ^ 2 - x₁ - x₂) =
      (d - u ^ 2) * ℓ ^ 2 := by ring

/-- Cleared addition y-coordinate identity; all denominator cases are in addY_scale. -/
theorem cleared_addY_identity {F : Type*} [CommRing F] (d u x₁ x₂ y₁ ℓ : F) :
    ℓ * (u ^ 2 * (2 * x₁ + x₂) - d * ℓ ^ 2) - y₁ * u ^ 2 -
        u ^ 2 * (ℓ * (2 * x₁ + x₂ - ℓ ^ 2) - y₁) =
      (u ^ 2 - d) * ℓ ^ 3 := by ring

section GroupMap

variable {k K : Type*} [Field k] [Field K] [Algebra k K]
  [DecidableEq k] [DecidableEq K]

theorem mapped_slope_scale (a b d x₁ x₂ y₁ y₂ : k) (hd : d ≠ 0)
    (u : K) (hu : u ^ 2 = algebraMap k K d) :
    ((short a b).baseChange K).slope
      (algebraMap k K x₁ / algebraMap k K d)
      (algebraMap k K x₂ / algebraMap k K d)
      (algebraMap k K y₁ / (algebraMap k K d * u))
      (algebraMap k K y₂ / (algebraMap k K d * u)) =
    algebraMap k K ((twist a b d).slope x₁ x₂ y₁ y₂) / u := by
  have hw : twist (algebraMap k K a) (algebraMap k K b) (algebraMap k K d) =
      (twist a b d).map (algebraMap k K) := by
    simp [twist, short, WeierstrassCurve.map]
  have hs := slope_scale (algebraMap k K a) (algebraMap k K b)
    (algebraMap k K d) u (algebraMap k K x₁) (algebraMap k K x₂)
    (algebraMap k K y₁) (algebraMap k K y₂)
    ((map_ne_zero (algebraMap k K)).mpr hd) hu
  rw [hw, map_slope] at hs
  simpa [short, WeierstrassCurve.baseChange, WeierstrassCurve.map] using hs

theorem lift_add (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (P Q : (twist a b d).Point) :
    lift a b d hd u hu (P + Q) =
      lift a b d hd u hu P + lift a b d hd u hu Q := by
  have hdK : algebraMap k K d ≠ 0 := (map_ne_zero (algebraMap k K)).mpr hd
  have hu0 : u ≠ 0 := by
    intro hz
    exact hdK (by simpa [hz] using hu.symm)
  have hdu : algebraMap k K d * u ≠ 0 := mul_ne_zero hdK hu0
  cases P with
  | zero => rfl
  | some x₁ y₁ h₁ =>
    cases Q with
    | zero => rfl
    | some x₂ y₂ h₂ =>
      have hneg : ((short a b).baseChange K).negY
          (algebraMap k K x₂ / algebraMap k K d)
          (algebraMap k K y₂ / (algebraMap k K d * u)) =
          algebraMap k K ((twist a b d).negY x₂ y₂) / (algebraMap k K d * u) := by
        simp [negY, short, twist, WeierstrassCurve.baseChange, WeierstrassCurve.map, neg_div]
      have hxy : ((algebraMap k K x₁ / algebraMap k K d =
            algebraMap k K x₂ / algebraMap k K d) ∧
          algebraMap k K y₁ / (algebraMap k K d * u) =
            ((short a b).baseChange K).negY
              (algebraMap k K x₂ / algebraMap k K d)
              (algebraMap k K y₂ / (algebraMap k K d * u))) ↔
          (x₁ = x₂ ∧ y₁ = (twist a b d).negY x₂ y₂) := by
        rw [hneg, div_left_inj' hdK, div_left_inj' hdu]
        simp only [(algebraMap k K).injective.eq_iff]
      by_cases hv : x₁ = x₂ ∧ y₁ = (twist a b d).negY x₂ y₂
      · rw [Point.add_of_Y_eq hv.1 hv.2]
        change 0 = Point.some _ _ _ + Point.some _ _ _
        rw [Point.add_of_Y_eq (hxy.mpr hv).1 (hxy.mpr hv).2]
      · rw [Point.add_some hv]
        change Point.some _ _ _ = Point.some _ _ _ + Point.some _ _ _
        rw [Point.add_some (fun h => hv (hxy.mp h)), Point.some.injEq]
        simp only [mapped_slope_scale a b d x₁ x₂ y₁ y₂ hd u hu]
        constructor
        · have hs := addX_scale (algebraMap k K a) (algebraMap k K b)
            (algebraMap k K d) u (algebraMap k K x₁) (algebraMap k K x₂)
            (algebraMap k K ((twist a b d).slope x₁ x₂ y₁ y₂)) hu
          simpa [addX, short, twist, WeierstrassCurve.baseChange,
            WeierstrassCurve.map] using hs.symm
        · have hs := addY_scale (algebraMap k K a) (algebraMap k K b)
            (algebraMap k K d) u (algebraMap k K x₁) (algebraMap k K x₂)
            (algebraMap k K y₁)
            (algebraMap k K ((twist a b d).slope x₁ x₂ y₁ y₂)) hu
          simpa [addY, negY, negAddY, addX, short, twist,
            WeierstrassCurve.baseChange, WeierstrassCurve.map] using hs.symm

/-- The existing actual-point map, with its addition law proved rather than assumed. -/
noncomputable def liftHom (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d) :
    (twist a b d).Point →+ ((short a b).baseChange K).Point where
  toFun := lift a b d hd u hu
  map_zero' := rfl
  map_add' := lift_add a b d hd u hu

theorem lift_nsmul (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (P : (twist a b d).Point) (r : ℕ) :
    lift a b d hd u hu (r • P) = r • lift a b d hd u hu P :=
  (liftHom a b d hd u hu).map_nsmul r P

theorem lift_zsmul (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (P : (twist a b d).Point) (r : ℤ) :
    lift a b d hd u hu (r • P) = r • lift a b d hd u hu P :=
  (liftHom a b d hd u hu).map_zsmul r P

theorem source_annihilator (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (P : (twist a b d).Point) (r : ℕ) (hr : r • P = 0) :
    r • lift a b d hd u hu P = 0 := by
  rw [← lift_nsmul, hr]
  rfl

theorem lift_sum {ι : Type*} (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (s : Finset ι) (P : ι → (twist a b d).Point) :
    lift a b d hd u hu (∑ i ∈ s, P i) = ∑ i ∈ s, lift a b d hd u hu (P i) :=
  map_sum (liftHom a b d hd u hu) P s

/-- A group isomorphism onto the actual image; this alone asserts no eigenspace surjectivity. -/
noncomputable def liftEquivRange (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d) :
    (twist a b d).Point ≃+ (liftHom a b d hd u hu).range :=
  AddMonoidHom.ofInjective (f := liftHom a b d hd u hu) (lift_injective a b d hd u hu)

theorem lift_annihilator_iff (a b d : k) [(short a b).IsElliptic]
    (hd : d ≠ 0) (u : K) (hu : u ^ 2 = algebraMap k K d)
    (P : (twist a b d).Point) (r : ℕ) :
    r • lift a b d hd u hu P = 0 ↔ r • P = 0 := by
  constructor
  · intro hr
    apply lift_injective a b d hd u hu
    rw [lift_nsmul, hr]
    rfl
  · exact source_annihilator a b d hd u hu P r

end GroupMap

section Obstruction

variable {k K : Type*} [Field k] [Fintype k] [Field K] [Algebra k K]
  [DecidableEq k] [DecidableEq K]

/-- The odd annihilator is now on the source point, not assumed on its image. -/
theorem rational_relation_source_odd (a b d : k) [(short a b).IsElliptic]
    (hchar : ringChar k ≠ 2) (hd : ¬IsSquare d)
    (u : K) (hu : u ^ 2 = algebraMap k K d)
    (A R : ((short a b).baseChange k).Point) (P : (twist a b d).Point)
    (r : ℕ) (hodd : Nat.Coprime 2 r) (hr : r • P = 0)
    (hrel : FrobeniusBridge.rationalLift (short a b) A +
      lift a b d (nonsquare_ne_zero hd) u hu P =
      FrobeniusBridge.rationalLift (short a b) R) :
    P = 0 ∧ FrobeniusBridge.rationalLift (K := K) (short a b) A =
      FrobeniusBridge.rationalLift (K := K) (short a b) R :=
  rational_twist_relation_odd a b d hchar hd u hu A R P r hodd
    (source_annihilator a b d (nonsquare_ne_zero hd) u hu P r hr) hrel

/-- Only the aggregate is killed; no claim that the individual summands vanish. -/
theorem rational_sum_source_odd {ι : Type*} (a b d : k) [(short a b).IsElliptic]
    (hchar : ringChar k ≠ 2) (hd : ¬IsSquare d)
    (u : K) (hu : u ^ 2 = algebraMap k K d)
    (s : Finset ι) (P : ι → (twist a b d).Point)
    (A R : ((short a b).baseChange k).Point)
    (r : ℕ) (hodd : Nat.Coprime 2 r) (hr : r • (∑ i ∈ s, P i) = 0)
    (hrel : FrobeniusBridge.rationalLift (K := K) (short a b) A +
      (∑ i ∈ s, lift a b d (nonsquare_ne_zero hd) u hu (P i)) =
      FrobeniusBridge.rationalLift (K := K) (short a b) R) :
    (∑ i ∈ s, P i) = 0 ∧ FrobeniusBridge.rationalLift (K := K) (short a b) A =
      FrobeniusBridge.rationalLift (K := K) (short a b) R := by
  rw [← lift_sum] at hrel
  exact rational_relation_source_odd a b d hchar hd u hu A R _ r hodd hr hrel

end Obstruction
