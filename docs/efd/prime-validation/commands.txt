# Regenerate the g1p catalogue from the pinned upstream section.
# CACHE_DIR holds the fetched documents; the importer reuses them offline.
cargo run --release --manifest-path tools/efd/Cargo.toml -- \
  CACHE_DIR src/prime_ecc/efd/catalogue.json g1p

# Full prime EFD test group (catalogue hashes, chart round trip, affine-law
# comparison, checked short-Weierstrass adapter on small and large primes).
cargo test --lib prime_ecc::efd

# Emit the per-formula coverage rows checked into this directory.
EFD_COVERAGE_OUT=docs/efd/prime-validation/coverage.json \
  cargo test --lib prime_ecc::efd::tests::all_471

# Second small prime (p = 97, also 1 mod 12).

# The importer check target builds and its tests pass.
cargo test --release --manifest-path tools/efd/Cargo.toml
